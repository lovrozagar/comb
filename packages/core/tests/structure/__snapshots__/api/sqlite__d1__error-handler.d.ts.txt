// sqlite/d1/error-handler.d.ts
import { CombError } from "../../error.ts";
import { type ConstraintMap } from "../../types.ts";
/**
 * The CombError a Cloudflare D1 database error is: a constraint violation through the shared
 * SQLite handlers, then a query error (500) or a platform failure (transient 503, storage 422);
 * undefined for an error that is none of these. It throws nothing, so a caller can ask what an
 * error is (a unique violation it expects, say) without catching. The CombError's `cause` is safe
 * for clients; the original error, which holds the SQL and its bound values, is on `source`.
 */
declare function classifyD1Error(error: unknown, constraintMaps: ConstraintMap[]): CombError | undefined;
/** Throw the CombError classifyD1Error finds for `error`; return when it finds none */
declare function d1ErrorHandler(error: unknown, constraintMaps: ConstraintMap[]): void;
export { classifyD1Error, d1ErrorHandler };

// error.d.ts
import type { StatusKey } from "./types.ts";
export declare class CombError extends Error {
    readonly column?: string | undefined;
    readonly errorKey: string;
    readonly status: number;
    readonly statusKey: StatusKey;
    readonly table?: string | undefined;
    /** The error this one was made from, for logs. Never serialized: it can hold SQL and bound values. */
    readonly source: unknown;
    constructor(opts: {
        /** A message safe to show the client. */
        cause?: unknown;
        column?: string | undefined;
        errorKey: string;
        /** The original error, kept for logs and never serialized. */
        source?: unknown;
        status: StatusKey;
        table?: string | undefined;
    });
}

// types.d.ts
/** Database-relevant HTTP status keys */
export type StatusKey = "bad_request" | "conflict" | "forbidden" | "internal_server_error" | "service_unavailable" | "unprocessable_entity";
export declare const statusKeyToCode: Record<StatusKey, number>;
/** Database-related error key constants */
export declare const combErrorKeys: {
    readonly CHECK_CONSTRAINT_VIOLATION: "check_constraint_violation";
    readonly CONSTRAINT_VIOLATION: "constraint_violation";
    readonly CURSOR_NOT_WIRED: "cursor_not_wired";

    readonly DATABASE_IO_ERROR: "database_io_error";
    readonly DATABASE_READONLY: "database_readonly";
    readonly DATABASE_UNAVAILABLE: "database_unavailable";
    readonly EXCLUSION_CONSTRAINT_VIOLATION: "exclusion_constraint_violation";
    readonly FILTER_NOT_WIRED: "filter_not_wired";
    readonly FOREIGN_KEY_VIOLATION: "foreign_key_violation";
    readonly INVALID_CURSOR: "invalid_cursor";
    readonly INVALID_PAGE: "invalid_page";
    readonly INVALID_SELECT: "invalid_select";
    readonly INVALID_STATE_TRANSITION: "invalid_state_transition";
    readonly INVALID_VALUE: "invalid_value";
    readonly PAGE_TOO_DEEP: "page_too_deep";
    readonly PRIMARY_KEY_VIOLATION: "primary_key_violation";
    readonly REQUIRED_FIELD_MISSING: "required_field_missing";
    readonly SEARCH_NOT_WIRED: "search_not_wired";

    readonly VALUE_TOO_LARGE: "value_too_large";
};
/** Lowercase error key values from combErrorKeys */
export type CombErrorKey = (typeof combErrorKeys)[keyof typeof combErrorKeys];
/** Constraint map entry for mapping DB constraint violations to domain errors */
export type ConstraintMapEntry = {
    cause: string;
    errorKey: string;
    statusKey?: "bad_request" | "conflict" | "forbidden";
};
/** Per-table constraint violation mappings */
export type ConstraintMap = {
    [table: string]: {
        check?: Record<string, ConstraintMapEntry>;
        /** Postgres EXCLUDE constraints, by name */
        exclusion?: Record<string, ConstraintMapEntry>;
        foreignKey?: Record<string, ConstraintMapEntry>;
        primaryKey?: ConstraintMapEntry;
        /**
         * The columns of this table the client never sees, by SQL name: an error names none of them,
         * in its `column` or its cause (F67). Codegen writes it; a map without it names every column.
         */
        private?: readonly string[];
        /**
         * The columns of this table some foreign key points at, by SQL name: an UPDATE, or an
         * upsert's DO UPDATE, that assigns one may have failed on a row referencing the old value,
         * so SQLite's handler answers it with the neutral cause (U43). Codegen writes it.
         */
        referenced?: readonly string[];
        unique?: Record<string, ConstraintMapEntry>;
    };
};
/** Function signature for database error handlers */
export type DatabaseErrorHandler = (error: unknown, constraintMaps: ConstraintMap[]) => void;
