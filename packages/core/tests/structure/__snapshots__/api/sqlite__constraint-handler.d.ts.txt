// sqlite/constraint-handler.d.ts
/**
 * SQLite constraint violations mapped to CombErrors, one handler per extended result code.
 * Shared by the libsql, bun:sqlite and D1 handlers: the engine is the same, only the way the
 * driver carries the code differs, and `readSqliteError` absorbs that.
 */
import { type Fallback } from "../constraint-map.ts";
import { CombError } from "../error.ts";
import { type ConstraintMap } from "../types.ts";
/**
 * The CombError for an SQLite constraint violation inside `error`, or for a value its column
 * cannot hold (value-error.ts); null when it is neither.
 */
declare function constraintError(error: unknown, maps: ConstraintMap[]): CombError | null;
/**
 * Throws the mapped CombError when `error` is an SQLite constraint violation; otherwise returns.
 * `error` is the thrown error, wrapper and all, or an engine message given as text.
 */
declare function handleConstraintViolation(error: unknown, maps: ConstraintMap[]): void;
/** The outcome of an engine resource code, its extended form read as its primary; undefined for any other */
declare const resourceOutcome: (code: string | undefined) => Fallback | undefined;
/**
 * The CombError a libsql, Turso or bun:sqlite error is: a constraint violation or a resource
 * error; undefined for any other. It throws nothing, so a caller can ask what an error is.
 */
declare function classifySqliteError(error: unknown, maps: ConstraintMap[]): CombError | undefined;
/** The error handler for libsql, Turso and bun:sqlite: throws what classifySqliteError finds; any other error is left for the caller */
declare function sqliteErrorHandler(error: unknown, maps: ConstraintMap[]): void;
export { classifySqliteError, constraintError, handleConstraintViolation, resourceOutcome, sqliteErrorHandler };

// constraint-map.d.ts
/**
 * Looking a constraint violation up in the app's ConstraintMaps, shared by the SQLite and the
 * Postgres handlers: each dialect finds the table, column or constraint name its own way, and
 * both turn what they found into the same CombError.
 */
import { CombError } from "./error.ts";
import type { ConstraintMap, ConstraintMapEntry, StatusKey } from "./types.ts";
export type TableMap = NonNullable<ConstraintMap[string]>;
/** The generic error for a kind of violation, used when no map entry matches. */
export type Fallback = {
    cause: string;
    errorKey: string;
    status: StatusKey;
};
/** What a handler found: the map entry, if any, the generic error otherwise, and where. */
export type Found = {
    column?: string | undefined;
    entry: ConstraintMapEntry | undefined;
    fallback: Fallback;
    table?: string | undefined;
};
/** `record[key]` when `key` is the record's own (never an inherited name such as `constructor`). */
export declare const own: <T>(record: Record<string, T> | undefined, key: string | undefined) => T | undefined;
/**
 * The columns an error may name of those a violation reports: all of them, or none when a map lists
 * one as the table's private column, since naming the rest would point at it (F67)
 */
export declare function shown(maps: ConstraintMap[], table: string | undefined, columns: readonly string[]): readonly string[];
/** `column`, or undefined when a map lists it among `table`'s private columns (F67) */
export declare const shownColumn: (maps: ConstraintMap[], table: string | undefined, column: string | undefined) => string | undefined;
/** The first entry `pick` finds for `table` across `maps`. */
export declare function lookup(maps: ConstraintMap[], table: string | undefined, pick: (t: TableMap) => ConstraintMapEntry | undefined): ConstraintMapEntry | undefined;
/** The `kind` entry named `name` (a CHECK, or a unique index) in any table, with that table. */
export declare function findNamed(maps: ConstraintMap[], kind: "check" | "unique", name: string | undefined): {
    entry: ConstraintMapEntry;
    table: string;
} | undefined;
/** The CombError for what a handler found; `source` is the original error, kept for logs. */
export declare function toCombError({ column, entry, fallback, table }: Found, source: unknown): CombError;

// error.d.ts
import type { StatusKey } from "./types.ts";
export declare class CombError extends Error {
    readonly column?: string | undefined;
    readonly errorKey: string;
    readonly status: number;
    readonly statusKey: StatusKey;
    readonly table?: string | undefined;
    /** The error this one was made from, for logs. Never serialized: it can hold SQL and bound values. */
    readonly source: unknown;
    constructor(opts: {
        /** A message safe to show the client. */
        cause?: unknown;
        column?: string | undefined;
        errorKey: string;
        /** The original error, kept for logs and never serialized. */
        source?: unknown;
        status: StatusKey;
        table?: string | undefined;
    });
}

/** Lowercase error key values from combErrorKeys */
export type CombErrorKey = (typeof combErrorKeys)[keyof typeof combErrorKeys];
/** A key a database error classifier answers with: classifyD1Error, classifySqliteError and classifyPgError alike (C21) */
export type CombDatabaseErrorKey = (typeof combErrorKeys)["CHECK_CONSTRAINT_VIOLATION" | "CONSTRAINT_VIOLATION" | "DATABASE_IO_ERROR" | "DATABASE_READONLY" | "DATABASE_UNAVAILABLE" | "EXCLUSION_CONSTRAINT_VIOLATION" | "FOREIGN_KEY_VIOLATION" | "INVALID_VALUE" | "PRIMARY_KEY_VIOLATION" | "REQUIRED_FIELD_MISSING" | "STORAGE_LIMIT_REACHED" | "UNIQUE_CONSTRAINT_VIOLATION" | "VALUE_TOO_LARGE"];
/**
 * The keys comb's database error classifiers answer with: the subset of combErrorKeys a consumer's
 * map of database errors (a translation, a status) iterates to cover each one (C21). The query and
 * router keys are not among them.
 */
export declare const combDatabaseErrorKeys: readonly CombDatabaseErrorKey[];
/** Constraint map entry for mapping DB constraint violations to domain errors */
export type ConstraintMapEntry = {
    cause: string;
    errorKey: string;
    statusKey?: "bad_request" | "conflict" | "forbidden";
};
/** Per-table constraint violation mappings */
export type ConstraintMap = {
    [table: string]: {
        check?: Record<string, ConstraintMapEntry>;
        /** Postgres EXCLUDE constraints, by name */
        exclusion?: Record<string, ConstraintMapEntry>;
        foreignKey?: Record<string, ConstraintMapEntry>;
        primaryKey?: ConstraintMapEntry;
        /**
         * The columns of this table the client never sees, by SQL name: an error names none of them,
         * in its `column` or its cause (F67). Codegen writes it; a map without it names every column.
         */
        private?: readonly string[];
        /**
         * The columns of this table some foreign key points at, by SQL name: an UPDATE, or an
         * upsert's DO UPDATE, that assigns one may have failed on a row referencing the old value,
         * so SQLite's handler answers it with the neutral cause (U43). Codegen writes it.
         */
        referenced?: readonly string[];
        unique?: Record<string, ConstraintMapEntry>;
    };
};
/** Function signature for database error handlers */
export type DatabaseErrorHandler = (error: unknown, constraintMaps: ConstraintMap[]) => void;
