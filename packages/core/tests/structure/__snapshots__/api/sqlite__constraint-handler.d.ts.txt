// sqlite/constraint-handler.d.ts
import { CombError } from "../error.ts";
import { type ConstraintMap } from "../types.ts";
import { resourceOutcome } from "./resource-outcome.ts";
/**
 * The CombError for an SQLite constraint violation inside `error`, or for a value its column
 * cannot hold (value-error.ts); null when it is neither.
 */
declare function constraintError(error: unknown, given: readonly ConstraintMap[]): CombError | null;
/**
 * Throws the mapped CombError when `error` is an SQLite constraint violation; otherwise returns.
 * `error` is the thrown error, wrapper and all, or an engine message given as text.
 */
declare function handleConstraintViolation(error: unknown, maps: readonly ConstraintMap[]): void;
/**
 * The CombError a libsql, Turso or bun:sqlite error is: a constraint violation, a resource error,
 * or a lost connection, as on D1 and Postgres; undefined for any other. It throws nothing, so a
 * caller can ask what an error is.
 */
declare function classifySqliteError(error: unknown, maps: readonly ConstraintMap[]): CombError | undefined;
/** The error handler for libsql, Turso and bun:sqlite: throws what classifySqliteError finds; any other error is left for the caller */
declare function sqliteErrorHandler(error: unknown, maps: readonly ConstraintMap[]): void;
export { classifySqliteError, constraintError, handleConstraintViolation, resourceOutcome, sqliteErrorHandler };
export type { Fallback } from "../constraint-map.ts";

// error.d.ts
import type { StatusKey } from "./types.ts";
export declare class CombError extends Error {
    /** A message safe to show the client. The original error is on `source`. */
    readonly cause?: string | undefined;
    readonly column?: string | undefined;
    readonly errorKey: string;
    readonly status: number;
    readonly statusKey: StatusKey;
    readonly table?: string | undefined;
    /** The error this one was made from, for logs. Never serialized: it can hold SQL and bound values. */
    readonly source: unknown;
    constructor(opts: {
        /** A message safe to show the client. The original error goes on `source`. */
        cause?: string | undefined;
        column?: string | undefined;
        errorKey: string;
        /** The original error, kept for logs and never serialized. */
        source?: unknown;
        status: StatusKey;
        table?: string | undefined;
    });
    /** The body a client gets: the documented fields alone, however the engine enumerates them */
    toJSON(): {
        column?: string | undefined;
        errorKey: string;
        status: number;
        statusKey: StatusKey;
        table?: string | undefined;
    };
}

/** Lowercase error key values from combErrorKeys */
export type CombErrorKey = (typeof combErrorKeys)[keyof typeof combErrorKeys];
/** A key a database error classifier answers with: classifyD1Error, classifySqliteError and classifyPgError alike (C21) */
export type CombDatabaseErrorKey = (typeof combErrorKeys)["CHECK_CONSTRAINT_VIOLATION" | "CONSTRAINT_VIOLATION" | "DATABASE_IO_ERROR" | "DATABASE_READONLY" | "DATABASE_UNAVAILABLE" | "EXCLUSION_CONSTRAINT_VIOLATION" | "FOREIGN_KEY_VIOLATION" | "INVALID_VALUE" | "PRIMARY_KEY_VIOLATION" | "REQUIRED_FIELD_MISSING" | "STORAGE_LIMIT_REACHED" | "UNIQUE_CONSTRAINT_VIOLATION" | "VALUE_TOO_LARGE"];
/**
 * The keys comb's database error classifiers answer with: the subset of combErrorKeys a consumer's
 * map of database errors (a translation, a status) iterates to cover each one (C21). The query and
 * router keys are not among them.
 */
export declare const combDatabaseErrorKeys: readonly CombDatabaseErrorKey[];
/** Constraint map entry for mapping DB constraint violations to domain errors */
export type ConstraintMapEntry = {
    cause: string;
    errorKey: string;
    statusKey?: "bad_request" | "conflict" | "forbidden" | undefined;
};
/** Per-table constraint violation mappings */
export type ConstraintMap = {
    [table: string]: {
        check?: Record<string, ConstraintMapEntry> | undefined;
        /** Postgres EXCLUDE constraints, by name */
        exclusion?: Record<string, ConstraintMapEntry> | undefined;
        foreignKey?: Record<string, ConstraintMapEntry> | undefined;
        /**
         * The partitions of this Postgres table, keyed as the map keys a table: Postgres reports a
         * violation on the partition, which is read as this table's (Q7 of r18). Codegen writes it
         * from comb.config.ts's `partitions`; a table no entry lists or keys names no column.
         */
        partitions?: readonly string[] | undefined;
        primaryKey?: ConstraintMapEntry | undefined;
        /**
         * The columns of this table the client never sees, by SQL name: an error names none of them,
         * in its `column` or its cause (F67). Codegen writes it; a map without it names every column.
         */
        private?: readonly string[] | undefined;
        /**
         * The columns of this table some foreign key points at, by SQL name: an UPDATE, or an
         * upsert's DO UPDATE, that assigns one may have failed on a row referencing the old value,
         * so SQLite's handler answers it with the neutral cause (U43). Codegen writes it.
         */
        referenced?: readonly string[] | undefined;
        unique?: Record<string, ConstraintMapEntry> | undefined;
    };
};
/** Function signature for database error handlers */
export type DatabaseErrorHandler = (error: unknown, constraintMaps: readonly ConstraintMap[]) => void;

// constraint-map.d.ts
/**
 * Looking a constraint violation up in the app's ConstraintMaps, shared by the SQLite and the
 * Postgres handlers: each dialect finds the table, column or constraint name its own way, and
 * both turn what they found into the same CombError.
 */
import { CombError } from "./error.ts";
import type { ConstraintMap, ConstraintMapEntry, StatusKey } from "./types.ts";
export type TableMap = NonNullable<ConstraintMap[string]>;
/** The generic error for a kind of violation, used when no map entry matches. */
export type Fallback = {
    /** the message safe to show the client, which the CombError carries as its `cause` */
    cause: string;
    /** the key a client branches on, one of `combErrorKeys` */
    errorKey: string;
    /** the HTTP status the error answers with, by key: `conflict` is 409, `service_unavailable` 503 */
    status: StatusKey;
};
/** What a handler found: the map entry, if any, the generic error otherwise, and where. */
export type Found = {
    column?: string | undefined;
    entry: ConstraintMapEntry | undefined;
    fallback: Fallback;
    table?: string | undefined;
};
/** `record[key]` when `key` is the record's own (never an inherited name such as `constructor`). */
export declare const own: <T>(record: Record<string, T> | undefined, key: string | undefined) => T | undefined;
/**
 * The entry `map` holds for `table`: its own key's, else that of the longest prefix pattern it
 * starts with, a key ending in `*` (`"ut_*"`), which keys the tables an app makes at run time and
 * no generated map can name (R5 of plan 17). A key with a `*` anywhere else matches nothing
 */
export declare function entryOf(map: ConstraintMap, table: string | undefined): TableMap | undefined;
/**
 * The maps an app passes its error handlers, every key checked where it builds them, at module load:
 * a TypeError naming a key with a `*` anywhere but its end, never a lookup at the first violation
 * (Q4 of r19). The array comes back frozen, and no handler checks it again.
 */
export declare function defineConstraintMaps(...maps: ConstraintMap[]): readonly ConstraintMap[];
/**
 * `maps`, a handler's first look at each map reporting its malformed keys to the operator, through the
 * warn sink, each key once per process however many arrays or maps carry it: the violation is still classified with the well-formed
 * keys, as the client's answer must not change, nor say anything of the app's configuration (Q4 of r19)
 */
export declare function checkedOnce(maps: readonly ConstraintMap[]): readonly ConstraintMap[];
/**
 * The columns an error may name of those a violation reports: all of them, or none when a map lists
 * one as the table's private column, since naming the rest would point at it (F67), or when the
 * maps given key no such table (fail closed)
 */
export declare function shown(maps: readonly ConstraintMap[], table: string | undefined, columns: readonly string[]): readonly string[];
/** `column`, or undefined when a map lists it among `table`'s private columns (F67), or shown hides the table's */
export declare const shownColumn: (maps: readonly ConstraintMap[], table: string | undefined, column: string | undefined) => string | undefined;
/** The first entry `pick` finds for `table` across `maps`. */
export declare function lookup(maps: readonly ConstraintMap[], table: string | undefined, pick: (t: TableMap) => ConstraintMapEntry | undefined): ConstraintMapEntry | undefined;
/** The `kind` entry named `name` (a CHECK, or a unique index) in any table, with that table. */
export declare function findNamed(maps: readonly ConstraintMap[], kind: "check" | "unique", name: string | undefined): {
    entry: ConstraintMapEntry;
    table: string;
} | undefined;
/** The CombError for what a handler found; `source` is the original error, kept for logs. */
export declare function toCombError({ column, entry, fallback, table }: Found, source: unknown): CombError;
