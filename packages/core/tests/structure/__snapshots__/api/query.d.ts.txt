// query/cursor.d.ts
import type { SortDirection, SortField, SortNulls } from "./types.ts";
/**
 * Column keyset pagination breaks ties on.
 *
 * This is the single source of truth for the tiebreak: the SQL builders resolve
 * the column by this name, and `createListQuerySchema` publishes it as
 * `stableTiebreak`. Keeping one constant is what stops the published fact from
 * drifting away from the query that is actually run.
 *
 * A table whose primary-key property is named something else has no cursor
 * predicate applied at all — rows then duplicate and skip across pages — so
 * `validateTables()` rejects that shape rather than letting it fail silently.
 */
declare const CURSOR_TIEBREAK_COLUMN = "id";
/**
 * The longest cursor a list takes, and so the longest comb mints: it fits a URL. A cursor carries
 * its sort values, so a long text key could mint one past it, which the next page would refuse.
 */
declare const MAX_CURSOR_LENGTH = 8192;
/**
 * Wire payload. `s` is the sort signature the cursor was minted under, `v` one
 * value per sort key in order, `i` the tiebreak id. A cursor is only valid for
 * the exact sort that produced it.
 */
type CursorPayload = {
    i: CursorId;
    s: string;
    v: unknown[];
};
/** The tiebreak id as the row held it: a number stays a number, a bigint a bigint, so it compares as one */
type CursorId = bigint | number | string;
type DecodedCursor = {
    id: CursorId;
    sort: string;
    values: CursorValue[];
};
/** What a sort key can hold and its database can compare */
type CursorValue = bigint | boolean | Date | number | string | null;
type PaginationMeta = {
    /** Rows the filter matches across all pages; null when the response cannot know */
    count: number | null;
    hasMore: boolean;
    limit: number;
    nextCursor: string | null;
    page: number | null;
};
/** PostgreSQL semantics: nulls sort last ascending and first descending. */
declare function effectiveNulls(direction: SortDirection, nulls?: SortNulls): SortNulls;
/** Canonical form of a sort — what a cursor is bound to. */
declare function sortSignature(parsedSort: readonly SortField[]): string;
/** A signed cursor for `payload` */
declare function encodeCursor(payload: CursorPayload, secret: string): string;
/** Mint the cursor for the row after which the next page starts. */
declare function createCursor(record: Record<string, unknown>, parsedSort: readonly SortField[], opts: {
    cursorSecret: string;
    idField?: string | undefined;
}): string;
/**
 * Why a cursor cannot be used under this sort, or null when its shape can: the list schema's
 * check, a 400 at validation. Its tag is verified by readCursor, which holds the secret.
 */
declare function cursorProblem(cursor: string, parsedSort: readonly SortField[]): string | null;
/** The cursor `secret` signed, decoded; null when its tag is not that secret's or it is no cursor */
declare function verifiedCursor(cursor: string, secret: string): DecodedCursor | null;
/** Verify a cursor's tag under `secret` and decode it for this sort, or throw a 400 CombError. */
declare function readCursor(cursor: string, parsedSort: readonly SortField[], secret: string): DecodedCursor;
export { createCursor, CURSOR_TIEBREAK_COLUMN, MAX_CURSOR_LENGTH, cursorProblem, effectiveNulls, encodeCursor, readCursor, sortSignature, type CursorId, type CursorPayload, type CursorValue, type DecodedCursor, type PaginationMeta, verifiedCursor, };

    limit: number;

 * Closed on purpose: a bare `{ cursor? }` took any object, so a wrong argument compiled and counted.
 */

    cursor?: string | null | undefined;
    limit: number;
    parsedSort: readonly SortField[];
} | {
    meta: {
        type: "cursor" | "offset";
    };
};
/**
 * The `select` extra that counts every row the filter matches, for `paginate` to report. An
 * offset request gets it; a cursor request gets none, because the window would make the
 * database read every remaining row on each page, which is what keyset pagination avoids.
 */
declare function countExtra(query: CountQuery): {
    _total?: SQL.Aliased<number>;
};
/** Extract a single filter value from FilterAST */
declare function extractFilterValue<T = unknown>(filterAst: FilterAST | null, field: string): T | undefined;
/** Extract multiple filter values from FilterAST */
declare function extractFilterValues<T extends Record<string, unknown>>(filterAst: FilterAST | null, fields: (keyof T)[]): Partial<T>;
/** What signs and verifies the cursors of a list: the app's secret, of at least 32 UTF-8 bytes */
type CursorOptions = {
    cursorSecret: string;
};
/**
 * limit, offset, order and the cursor predicate for drizzle's relational `findMany`, from the
 * list query schema's output, the cursor's tag verified under `cursorSecret`. The limit is one
 * over the page so `paginate` can tell there is more.
 */
declare function drizzleListOptions(query: {
    cursor?: string | undefined;
    limit: number;
    page?: number | undefined;
    parsedSort: SortField[];
    sortOrderBy: Record<string, "asc" | "desc">;
    validated?: true | undefined;
}, { cursorSecret }: CursorOptions): DrizzleListOptions;

    _total?: number;
};
/** paginate's options: the cursor secret, and the field a row's tiebreak id is in (default `id`) */
type PaginateOptions = CursorOptions & {
    idField?: string | undefined;
};
/**
 * Create pagination result from drizzle query results with _total extra, the next cursor signed
 * under `cursorSecret` and keyed by `idField`, as buildDerivedListQuery's idColumn keys the page
 */
declare function drizzlePaginationResult<T extends object>(items: DrizzleListResult<T>[], query: {
    cursor?: string;
    limit: number;
    page?: number;
    parsedFields?: ParsedFields | null | undefined;
    parsedSort: SortField[];
    validated?: true | undefined;
}, options: PaginateOptions): readonly [T[], PaginationMeta];
declare function drizzleProject<T extends Record<string, unknown>>(row: T, parsedFields: ParsedFields | null): T;
/** List-query helpers for drizzle queries: count, options, pagination, projection */
declare const queryHelpers: {
    countExtra: typeof countExtra;
    filterValue: typeof extractFilterValue;
    filterValues: typeof extractFilterValues;
    listOptions: typeof drizzleListOptions;
    paginate: typeof drizzlePaginationResult;
    project: typeof drizzleProject;
};
export { type CursorOptions, type DrizzleListOptions, type PaginateOptions, queryHelpers };

// query/filter.d.ts
/**
 * PostgREST-style filters and sort orders: parse and validate. DB-agnostic: no SQL, only
 * the AST and its checks. The grammar lives in filter-parse.ts, the checks in filter-validate.ts.
 */
import { FILTER_OPERATORS, parseFilter } from "./filter-parse.ts";
import { FILTER_LIMITS, OPERATORS_BY_TYPE, type FilterLimits, type FilterValidationResult, validateFilter } from "./filter-validate.ts";
import type { SortField } from "./types.ts";
/**
 * Parse PostgREST-style `order` string into SortField array.
 *
 * Grammar:
 *   order  := entry ("," entry)*
 *   entry  := field ("." direction)? ("." nulls)?
 *   field  := any string not containing "." at the direction/nulls boundary
 *            (dotted relation paths OK: "author.name.asc")
 *   direction := "asc" | "desc"        (default "asc" when omitted)
 *   nulls     := "nullsfirst" | "nullslast"
 *
 * Examples:
 *   "name"                         → [{ field: "name", direction: "asc" }]
 *   "name.desc"                    → [{ field: "name", direction: "desc" }]
 *   "name.asc.nullslast"           → [{ field: "name", direction: "asc", nulls: "last" }]
 *   "col1.desc,col2.asc.nullsfirst"
 *     → [{field:"col1",direction:"desc"}, {field:"col2",direction:"asc",nulls:"first"}]
 *
 * Malformed entries are dropped silently — downstream schema validator
 * rejects unknown fields, which catches most mistakes.
 */
declare function parseOrder(input: string | null | undefined): SortField[];
export { FILTER_LIMITS, FILTER_OPERATORS, OPERATORS_BY_TYPE, parseFilter, parseOrder, validateFilter, type FilterLimits, type FilterValidationResult, };

// query/like.d.ts
/**
 * PostgREST like/ilike: `*` is the wildcard, every other character is literal.
 *
 * `ilike` lowers to SQL LIKE (ASCII case-insensitive in SQLite), so LIKE's own
 * specials `%` and `_` are escaped before `*` expands to `%`.
 */
declare function likePattern(value: string): string;
/**
 * `like` lowers to SQLite GLOB, which is case-sensitive. GLOB has no escape
 * character: `?` and `[` are made literal by wrapping them in a one-character
 * class, and `*` stays the wildcard.
 */
declare function globPattern(value: string): string;
/**
 * `text` as a literal inside a LIKE pattern written with `ESCAPE '\\'`: the escape character and
 * LIKE's own specials `%` and `_` each escaped, so `%${escapeLike(term)}%` finds `term` as written
 */
declare function escapeLike(text: string): string;
export { escapeLike, globPattern, likePattern };

 * - `in.(a,b)`, `nin.(a,b)`: lists

 */
