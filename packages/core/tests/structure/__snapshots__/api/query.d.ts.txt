/**
 * Column keyset pagination breaks ties on.
 *
 * This is the single source of truth for the tiebreak: the SQL builders resolve
 * the column by this name, and `createListQuerySchema` publishes it as
 * `stableTiebreak`. Keeping one constant is what stops the published fact from
 * drifting away from the query that is actually run.
 *
 * A table whose primary-key property is named something else has no cursor
 * predicate applied at all — rows then duplicate and skip across pages — so
 * `validateTables()` rejects that shape rather than letting it fail silently.
 */
declare const CURSOR_TIEBREAK_COLUMN = "id";
/**
 * The longest cursor a list takes, and so the longest comb mints: it fits a URL. A cursor carries
 * its sort values, so a long text key could mint one past it, which the next page would refuse.
 */
declare const MAX_CURSOR_LENGTH = 8192;
/**
 * Wire payload. `s` is the sort signature the cursor was minted under, `v` one
 * value per sort key in order, `i` the tiebreak id. A cursor is only valid for
 * the exact sort that produced it.
 */
type CursorPayload = {
    c?: boolean[] | undefined;
    d?: Array<[unknown] | number | "invalid" | null> | undefined;
    i: CursorId;
    s: string;
    v: unknown[];
};
/** An opened cursor, as `readCursor` returns it */
type DecodedCursor = {
    /** per sort value, true where it is its key's stored text, which the next page binds as itself; false where it is the value as the row read it */
    carried: boolean[];
    /** the tiebreak id of the row the next page starts after */
    id: CursorId;
    /** the signature of the sort it was minted under, as `sortSignature` writes it */
    sort: string;
    /** one value per sort key, in order */
    values: CursorValue[];
};
type OpenedCursor = DecodedCursor & {
    read?: Array<readonly [CursorValue] | null> | undefined;
};
/** PostgreSQL semantics: nulls sort last ascending and first descending. */
declare function effectiveNulls(direction: SortDirection, nulls?: SortNulls): SortNulls;
/** Canonical form of a sort — what a cursor is bound to. */
declare function sortSignature(parsedSort: readonly SortField[]): string;
/** A sealed cursor for `payload`, opened only under the same `cursorSecret` and `cursorScope` */
declare function encodeCursor(payload: CursorPayload, options: CursorOptions): string;
/**
 * Mint the cursor for the row after which the next page starts, sealed under `options` and keyed
 * by `idField`, whole: the public `createCursor` (cursor-mint.ts) keeps it within the cap (U37). A
 * row it cannot resume after is a 500 here, on the page that reads it (refuseUncarried). `read` is
 * the row as the driver read it, whose value readCursor returns for a key `record` binds otherwise
 */
declare function mintCursor(record: Record<string, unknown>, parsedSort: readonly SortField[], options: CursorOptions & MintOptions): string;
/** What a mint is told beside the seal's options: the carried keys, the id's field, and the row as read */
type MintOptions = {
    carried?: ReadonlySet<string> | undefined;
    idField?: string | undefined;
    read?: Record<string, unknown> | undefined;
};
/**
 * What mintCursor seals for `record`, every refusal of it run: the cursor's length, which the bytes'
 * count alone sets (sealedLength), is known before any seal, so a page checks each row it serves
 */
declare function cursorBytes(record: Record<string, unknown>, parsedSort: readonly SortField[], options: MintOptions | undefined): Uint8Array;
/**
 * Why `cursor` cannot be one, or null when its shape can: the list schema's check, a 400 at
 * validation. What it holds, and the sort it was minted under, only readCursor can open.
 */
declare function cursorProblem(cursor: string): string | null;
/** The cursor sealed under `options`, opened and decoded; null when it was not, or it is no cursor */
declare function openCursor(cursor: string, options: CursorOptions): OpenedCursor | null;
/** The cursor as the keyset binds it, each key's stored value, for this sort; a 400 CombError as readCursor throws */
declare function readKeyset(cursor: string, parsedSort: readonly SortField[], options: CursorOptions): OpenedCursor;
/**
 * Open a cursor under `options` and decode it for this sort, or throw a 400 CombError. Each key is
 * as the cursor's row read it, where the list pages by another value: a time key is the Date the
 * driver read, an Invalid one included, and bytes are a Uint8Array. A value may be a CursorRef: a long sort value
 * carried by reference, its prefix and length, which a keyset of its own must read from the

 */
declare function readCursor(cursor: string, parsedSort: readonly SortField[], options: CursorOptions): DecodedCursor;

/**
 * How a built list pages what it fetched (Q6, C3, C19, C2 of the 0.14.1 review). Each builder
 * (`buildListQuery`, its `merged` list included, `buildDerivedListQuery`, `queryHelpers.listOptions`)
 * hands over a `pagination` descriptor: the page it read, whether a cursor can resume it, how a
 * long key is carried, and how many rows it asked for. `paginate` reads it, so nothing about the
 * list is told twice. The cursor secret is never a value of it: the descriptor seals and opens
 * cursors through closures over the options its builder was given, so logging or serializing a
 * built query shows no part of the secret.
 */
import { type DecodedCursor } from "./cursor.ts";
import { type KeysetKind } from "./cursor-mint.ts";
import { type CursorOptions } from "./cursor-seal.ts";
import { COMB_ID } from "./list-extras.ts";
import type { paginationResponseSchema } from "./schema.ts";
import type { ListQueryInput, ParsedFields, SortField } from "./types.ts";
import type * as z from "zod";
/**
 * The name of one column a builder selects for `paginate` to read, which no row it returns keeps: the
 * tiebreak, the gone-anchor flag, the count, each sort key's storage class, or its stored text
 */
type ListExtra = "_combAnchorGone" | "_combId" | "_combKeyTexts" | "_combKeyTypes" | "_combTextKeys" | "_total";
/**
 * A row as paginate and paginateMerged return it, without the `ListExtra` columns. A union row type
 * loses them from each member, so narrowing on its discriminant still reaches each variant's fields
 */
type ListRow<T> = T extends unknown ? Omit<T, ListExtra> : never;
/** The pagination meta paginate returns: the list response's own schema */
type PaginationResponse = z.output<typeof paginationResponseSchema>;
/**
 * How a built list pages what it fetched, as a builder hands it over on `pagination`; pass it on to
 * `paginate` whole. It seals and opens cursors without holding the secret as a value, so logging or
 * serializing it shows none of it
 */
type ListPagination = {
    /** the cursor the request resumed from; none on a page by number */
    cursor: string | undefined;
    /** false when no cursor can resume the list: its order holds a computed sort, or it has no tiebreak */
    cursorable: boolean;
    /** the scope its cursors are sealed under, as the builder was given it */
    cursorScope: string;
    /** the extras the builder selected, which every row it reads must carry, or `paginate` answers a 500 naming the one missing */
    expects: readonly ListExtra[];
    /** how a key too long for a cursor is carried: by reference to its row ("table"), or not at all ("derived") */
    keyset: KeysetKind;
    /** the page's row count, the request's limit */
    limit: number;
    /** true on a list built with `merged`, which `paginateMerged` pages and `paginate` refuses */
    merged: boolean;
    /** "exact" where the builder was told its driver reads ±Infinity back as itself, so a null cursor value is a real NULL */
    nonFinite: "exact" | undefined;
    /** the `page` param as sent; a cursor, when the request has one, wins over it */
    page: number | undefined;
    /** the request's `select`, which `paginate` cuts each row to; null keeps every field */
    parsedFields: ParsedFields | null;
    /** the order the list was read in, which the next cursor is minted under */
    parsedSort: readonly SortField[];
    /** the most rows the built query reads: one past the page, so paginate can tell there is more, or on a merged page by number each shard's rows to the page's end and one more */
    window: number;
    /**
     * Every check the cursor for the page after `row` passes, run now, so a row no cursor holds is a 500
     * on the page that serves it; and what seals that cursor under the builder's options
     */
    fit(row: object): () => string;
    /** `cursor` opened under the builder's options, for this list's sort; a 400 when it is no cursor of it */
    read(cursor: string): DecodedCursor;
};
/** What a builder knows of its list, beside the request it parsed */
type PaginationFacts = {
    /** the sort fields whose keys the list binds as read, with no row to read a bound from (QL1 of r22b); every one on a derived list */
    asRead?: ReadonlySet<string> | undefined;
    /** the sort fields whose keys read a reference back from the list's table (readsBack); every one when not given */
    byReference?: ReadonlySet<string> | undefined;
    cursorable: boolean;
    /** the extras the builder selected, which every row must carry */
    expects: readonly ListExtra[];
    keyset?: KeysetKind | undefined;
    query: ListQueryInput;
    window: number;
};
/** The descriptor a builder hands over: its facts, and the options it seals under closed over, never held */
declare function paginationOf<TMerged extends boolean = false>(options: CursorOptions, facts: PaginationFacts & {
    merged?: TMerged | undefined;
}): ListPagination & {
    merged: TMerged;
};
/** The count an offset page's row carries, in any encoding isCount takes, which refuseUnwired holds every such row to */
declare const totalOf: (row: object) => number;
/**
 * Last mutation of item shape. Query stays fat; the returned object is the
 * public subset. No-op (minus `_total`) when parsedFields is null.
 */
declare function applySelect<T extends Record<string, unknown>>(row: T, parsedFields: ParsedFields | null): T;
/**
 * The page of `items`, read with `built`, and its pagination: count, whether there is more, and
 * the next cursor, sealed under the builder's options; none for a list no cursor can resume
 */
declare function paginate<T extends object>(items: readonly T[], built: {
    /** the built list's pagination, as its builder returned it */
    pagination: ListPagination & {
        /** false: a merged list goes to `paginateMerged` */
        merged: false;
    };
}): readonly [Array<ListRow<T>>, PaginationResponse];
export { applySelect, COMB_ID, type ListExtra, type ListPagination, type ListRow, paginate, type PaginationFacts, type PaginationResponse, paginationOf, totalOf, };

// query/cursor-mint.d.ts
import { type CursorOptions } from "./cursor-seal.ts";

/** paginate's choice for an overflowing key: by reference on a table, refused on a derived or a merged list */
type KeysetKind = "derived" | "merged" | "table";
/**
 * Mint the cursor for the row after which the next page starts, as `paginate` does: its longest
 * string values carried by reference while it passes the cap that `readCursor` holds; a 500 on a
 * derived list (`keyset: "derived"`), or when no reference makes it fit; a 500 on a derived list's
 * integer key past ±(2^53 − 1), which it binds as the driver rounded it
 */
declare function createCursor(row: Record<string, unknown>, sort: readonly SortField[], options: CursorOptions & {
    /** the sort fields the list binds as the row read them, with no row to read a bound from; every one on a derived list */
    asRead?: ReadonlySet<string> | undefined;
    /** the sort fields a long value of may go by reference to its row; every one when not given */
    byReference?: ReadonlySet<string> | undefined;
    /** the sort fields whose value in `row` is the key's stored text, which the next page binds as itself */
    carried?: ReadonlySet<string> | undefined;
    /** the field of `row` that holds the tiebreak id; default `id` */
    idField?: string | undefined;
    /** how a value too long for the cap is carried: by reference on "table", the default; refused with a 500 on "derived" or "merged" */
    keyset?: KeysetKind | undefined;
}): string;
/**
 * createCursor for the row paginate typed, bound by its stored values: `read` is the row as the driver
 * read it, whose value readCursor returns for a key bound otherwise
 */
declare function fitCursor(row: Record<string, unknown>, sort: readonly SortField[], options: Parameters<typeof createCursor>[2] & {
    read?: Record<string, unknown> | undefined;
}): string;
/**
 * fitCursor's every refusal of `row`, run with no seal, and what seals its cursor: the cap is measured
 * on the plaintext (sealedLength), so a page checks each row it serves and seals the last alone
 */
declare function fittedCursor(row: Record<string, unknown>, sort: readonly SortField[], options: Parameters<typeof fitCursor>[2]): () => string;
export { createCursor, fitCursor, fittedCursor, type KeysetKind };

// query/cursor-seal.d.ts
/** What seals and opens the cursors of one list: the app's secret, and the list's scope (route plus tenant) */
type CursorOptions = {
    /** the list's name, its route and tenant, such as `members:${orgId}`; a cursor sealed under one opens under no other. Empty is a 500 */
    cursorScope: string;
    /** the app's secret, at least `MIN_CURSOR_SECRET_BYTES` bytes of UTF-8, the same on every worker; a shorter one is a 500 */
    cursorSecret: string;
    /**
     * "exact" when the driver reads a SQLite REAL's ±Infinity back as Infinity (a local libsql file,
     * bun:sqlite), so a null cursor is a real NULL and a list whose NULL anchor is deleted resumes
     * right after it. Unset, as on D1, which reads ±Infinity as null, such a list resumes from the
     * earliest value the null may stand for, repeating rows and skipping none.
     */
    nonFinite?: "exact" | undefined;
};
/** The shortest secret a cursor is sealed under, in UTF-8 bytes: the key's length */
declare const MIN_CURSOR_SECRET_BYTES = 32;
/** The 32-byte key HKDF-SHA256 derives from `secret`, with no salt and comb's info */
declare function cursorKey(secret: string): Uint8Array;
/** Throw the 500 sealing throws when `options` cannot seal a cursor */
declare function assertCursorWired(options: CursorOptions): void;
/** The length of the cursor seal makes of `length` bytes of plaintext, which their count alone sets */
declare const sealedLength: (length: number) => number;
/**
 * `plaintext` sealed under `options`, as base64url, padded with spaces up to a multiple of 256
 * bytes first (U9): AEAD hides the bytes, not their count, and a cursor carries a hidden column's
 * value too. openSealed strips them: what seal takes ends in no space of its own.
 */
declare function seal(plaintext: Uint8Array, options: CursorOptions): string;
/** Whether `cursor` has a sealed cursor's shape: what a check without the secret can know */
declare const isSealedShape: (cursor: string) => boolean;
/** What `cursor` holds, opened under `options`; null when it was not sealed under both */
declare function openSealed(cursor: string, options: CursorOptions): Uint8Array | null;
export { assertCursorWired, cursorKey, type CursorOptions, isSealedShape, MIN_CURSOR_SECRET_BYTES, openSealed, seal, sealedLength };

 * query-helpers with drizzle's relational query API.

/** A table as drizzle's relational callbacks hand it over, holding its columns as own properties */
export type RelationalTable = object;
/**
 * What drizzle's relational `findMany` reads one list page with: spread `limit`, `offset`, `orderBy`
 * and `extras` into it, AND `where` with your own, and hand the rows to `paginate` with these options
 */
type DrizzleListOptions = {
    /** one over the page's limit, so `paginate` can tell there is a next page */
    limit: number;
    /** the rows before the page; 0 on a cursor request, which resumes by `where` */
    offset: number;
    /** The sort, its null placement and the id tiebreak, for `findMany({ orderBy })` */
    orderBy: (table: RelationalTable) => SQL[];
    /**
     * For `findMany({ extras })`: the columns `paginate` reads from each row, and refuses a page
     * without. `project` strips them from the rows it returns
     */
    extras: {
        /** the row's `id`, which the next cursor resumes from */
        _combId: (table: RelationalTable) => SQL;
    } & Partial<{
        /** on a cursor whose row may be gone: 1 when it is, which `paginate` answers with a 400 asking to start again */
        _combAnchorGone: (table: RelationalTable) => SQL<number>;
        /** each sort key's stored text where reading it back rounds it; on every list with a sort */
        _combKeyTexts: (table: RelationalTable) => SQL<string>;
        /** each SQLite timestamp sort key's stored TEXT, which the next cursor binds as stored; on every list with a sort */
        _combTextKeys: (table: RelationalTable) => SQL<string>;
        /** on a page read by number: every row the filter matches, which `paginate` reports as the count */
        _total: (table: RelationalTable) => SQL<number>;
    }>;
    /** How `paginate` pages the rows these options read */
    pagination: ListPagination & {
        /** always false: these options read one database, which `paginate` pages */
        merged: false;
    };
} & Partial<{
    /** The keyset predicate of a cursor request, for `findMany({ where })` (AND it with your own) */
    where: {
        /** the predicate on the queried table, for drizzle to inline as raw SQL */
        RAW: (table: RelationalTable) => SQL;
    };
}>;
/**
 * limit, offset, order and the cursor predicate for drizzle's relational `findMany`, from the
 * list query schema's output, the cursor opened under `options`: the app's secret and the list's
 * scope. The limit is one over the page so `paginate` can tell there is more.
 */
declare function drizzleListOptions(query: ListQueryInput, options: CursorOptions): DrizzleListOptions;
/** `row` with only the fields `select` asked for, and none of the list's own extras; every field when `parsedFields` is null */
declare function drizzleProject<T extends Record<string, unknown>>(row: T, parsedFields: ParsedFields | null): T;
/** List-query helpers for drizzle queries: options, pagination, projection */
declare const queryHelpers: {
    /** the value the filter holds every row's field to, as `extractFilterValue` reads it */
    filterValue: typeof extractFilterValue;
    /** the values the filter holds every row's fields to, as `extractFilterValues` reads them */
    filterValues: typeof extractFilterValues;
    /** the `findMany` options of a list page, as `drizzleListOptions` builds them */
    listOptions: typeof drizzleListOptions;
    /** the page and its pagination from the rows `listOptions` read */
    paginate: typeof paginate;
    /** a row cut to its `select`, as `drizzleProject` cuts it */
    project: typeof drizzleProject;
};
export { type CursorOptions, drizzleListOptions, type DrizzleListOptions, drizzleProject, queryHelpers };

// query/filter.d.ts
/**
 * PostgREST-style filters and sort orders: parse and validate. DB-agnostic: no SQL, only
 * the AST and its checks. The grammar lives in filter-parse.ts, the checks in filter-validate.ts.
 */
import { FILTER_OPERATORS, parseFilter } from "./filter-parse.ts";
import { parseFilterNumber } from "./filter-typed.ts";
import { FILTER_LIMITS, OPERATORS_BY_TYPE, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, type LimitsGiven, validateFilter } from "./filter-validate.ts";
import type { SortField } from "./types.ts";
/**
 * Parse PostgREST-style `order` string into SortField array.
 *
 * Grammar:
 *   order  := entry ("," entry)*
 *   entry  := field ("." direction)? ("." nulls)?
 *   field  := any string not containing "." at the direction/nulls boundary
 *            (dotted relation paths OK: "author.name.asc")
 *   direction := "asc" | "desc"        (default "asc" when omitted)
 *   nulls     := "nullsfirst" | "nullslast"
 *
 * Examples:
 *   "name"                         → [{ field: "name", direction: "asc" }]
 *   "name.desc"                    → [{ field: "name", direction: "desc" }]
 *   "name.asc.nullslast"           → [{ field: "name", direction: "asc", nulls: "last" }]
 *   "col1.desc,col2.asc.nullsfirst"
 *     → [{field:"col1",direction:"desc"}, {field:"col2",direction:"asc",nulls:"first"}]
 *
 * An entry that is empty, names no field (`asc`, `.desc`, a trailing comma) or has whitespace around
 * it, and a field named twice, refuse the whole order as null, as parseFilter refuses a filter, rather
 * than drop it and widen the request to the default order.
 */
declare function parseOrder(input: string | null | undefined): SortField[] | null;
/** Each field `fields` names more than once, once each */
declare function namedTwice(fields: readonly SortField[]): string[];
/**
 * Each entry of `order` read as a sort field, or kept as its text when it is empty, names no field or
 * has whitespace around it, which the published pattern refuses too
 */
declare function orderEntries(input: string | null | undefined): Array<SortField | string>;
export { FILTER_LIMITS, FILTER_OPERATORS, OPERATORS_BY_TYPE, parseFilter, parseFilterNumber, namedTwice, orderEntries, parseOrder, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, type LimitsGiven, };

// query/like.d.ts
/**
 * PostgREST like/ilike: `*` is the wildcard, every other character is literal.
 *
 * `ilike` lowers to SQL LIKE (ASCII case-insensitive in SQLite), so LIKE's own
 * specials `%` and `_` are escaped before `*` expands to `%`.
 */
declare function likePattern(value: string): string;
/**
 * `like` lowers to SQLite GLOB, which is case-sensitive. GLOB has no escape
 * character: `?` and `[` are made literal by wrapping them in a one-character
 * class, and `*` stays the wildcard.
 */

// query/types.d.ts
/**
 * Query helper types — DB-agnostic.
 * Shared by parsers, validators, and DB-specific SQL generators.
 */
import type { SQL } from "drizzle-orm";
/** The fields one level of a `select` names, as `parseSelect` reads it */
type FieldSelection = {
    /** each relation embedded as `name(...)`, by name, with what it selects; null selects every field of it */
    relations: Record<string, FieldSelection | null>;
    /** the field names at this level, each once, in the order written; `*` stands for every field, a leading `@` for a computed one */
    scalars: string[];
};
/** A parsed `select`; `filterBySelect` projects a row by it */
type ParsedFields = {
    /** what the top level selects */
    root: FieldSelection;
};
/**
 * A filter condition's operator, as written between the field and the value: `ne` and `neq` both
 * mean not equal, `in` and `nin` take a parenthesized list, `like` is case-sensitive and `ilike`
 * ignores ASCII case (both with `*` as the wildcard), and `is` takes `null`, `notnull`, `true` or `false`
 */
type FilterOperator = "eq" | "ne" | "neq" | "gt" | "gte" | "lt" | "lte" | "in" | "nin" | "like" | "ilike" | "is"
/** Parsed for callers that validate and run it themselves (JSON-array membership); comb's own validator and SQL take none */
 | "contains";
/** A filterable field's type, which decides the operators it takes and how its value is read: a `date` as epoch milliseconds unless its spec says `iso` */
type FieldType = "string" | "number" | "boolean" | "date" | "enum";
/** One `field.operator.value` term of a filter */
type FilterCondition = {
    /** the field as written: a column, `relation.column`, or a `@`-prefixed computed field */
    field: string;
    /** the operator as written; `ne` and `neq` are one */
    operator: FilterOperator;
    /** the value as parsed: a string, null for `is.null`, or an array of strings for a list; `validateFilter` types it as its field */
    value: unknown;
};
/** Conditions and nested groups joined by one logic */
type FilterGroup = {
    /** the group's own terms, in the order written */
    conditions: FilterCondition[];
    /** how the group's conditions and subgroups combine; the top level is `and` */
    logic: "and" | "or";
    /** each nested `and(...)` or `or(...)`, in the order written */
    subgroups: FilterGroup[];
};
/** A parsed filter, as `parseFilter` and `validateFilter` return it */
type FilterAST = {
    /** the top-level group, joined by its own `logic`; `parseFilter` and `validateFilter` make it an `and` */
    root: FilterGroup;
};
/** A sort key's direction */
type SortDirection = "asc" | "desc";
/** Where a sort key puts nulls; left out, they sort last ascending and first descending, as on Postgres */
type SortNulls = "first" | "last";
/** One key of an `order`, as `parseOrder` returns it */
type SortField<T extends string = string> = {
    /** the direction as written; `asc` when the key names none */
    direction: SortDirection;
    /** the field sorted on: a column, or a `@`-prefixed computed field */
    field: T;
    /** null placement as written; left out, the direction decides it (see SortNulls) */
    nulls?: SortNulls | undefined;
};
/**
 * What every list builder takes (buildListQuery, buildDerivedListQuery, queryHelpers.listOptions):
 * the list schema's output, or a query the app builds itself. The builders never change its arrays
 */
type ListQueryInput<TSortField extends string = string> = {
    /** the previous page's next cursor, to read the page after it; null or absent reads the first page */
    cursor?: string | null | undefined;
    /** the parsed filter; null or absent filters nothing */
    filterAst?: FilterAST | null | undefined;
    /** the most rows the page holds, a whole number from 1 */
    limit: number;
    /** the 1-based page to read by number; null or absent is page 1. A cursor, when given, wins */
    page?: number | null | undefined;
    /** the deepest page this query may read; absent, the default maxPage. A deeper page is a 400 */
    pageBound?: number | undefined;
    /** the largest limit this query may take; absent, the default maxLimit. A larger limit is a 400 */
    limitBound?: number | undefined;
    /** the column the list schema breaks ties on; a derived list's idColumn must name it */
    tiebreak?: string | undefined;
    /** the parsed `select`; null or absent selects every field */
    parsedFields?: ParsedFields | null | undefined;
    /** the sort keys in order; empty sorts by the tiebreak column alone */
    parsedSort: readonly SortField<TSortField>[];
    /** the search text, trimmed; null, absent or blank searches nothing */
    q?: string | null | undefined;
    /**
     * The list schema sets it. A builder throws a 500 on a filter or an order without it, as neither
     * was checked against the fields the endpoint declares; set it only on ones your code checked
     */
    validated?: true | undefined;
};
/** Lowers a computed filter field's condition to SQL; null when it does not take the operator, which the builder answers with a 500 */
type ComputedFilterResolver = (operator: FilterOperator, value: unknown) => SQL | null;
/** The ORDER BY term of a computed sort field in `direction`; a list sorted on one pages by number, and a cursor on it is a 400 */
type ComputedSortResolver = (direction: SortDirection) => SQL;
export type { ComputedFilterResolver, ComputedSortResolver, FieldSelection, FieldType, FilterAST, FilterCondition, FilterGroup, FilterOperator, ListQueryInput, ParsedFields, SortDirection, SortField, SortNulls, };

    /** the action the database takes on this row when the referenced row is deleted, as drizzle declares it (`cascade`, `set null`); absent for the database's default */
    onDelete?: string | undefined;
    /** the action the database takes on this row when the referenced key changes, as drizzle declares it; absent for the database's default */
    onUpdate?: string | undefined;
    /** the referenced column's property name in its table */
    refColumn: string;
    /** the referenced table's export name in the tables file */
    refTable: string;
    /** whether the referenced column is a c.id key, so the reference takes the id rule, 1 to 48 characters and no blank; its prefix is not checked */
    toId?: boolean | undefined;
    /** a composite foreign key's columns; absent for a single-column key */
    key?: {
        /** the referencing property names, in key order */
        columns: string[];
        /** the referenced property names, each paired by position with `columns` */
        refColumns: string[];
    } | undefined;
};
/** A single column/field in a table, as drizzle built it and its `c` helper recorded it */
type FieldMeta = {
    /** SQLite's AUTOINCREMENT on an integer key */
    autoIncrement: boolean;
    /** drizzle's column class: `SQLiteText`, `PgVarchar` */
    columnType: string;
    /** the `c` helper that built the column; null for a plain drizzle column */
    combKind: CombColumnKind | null;
    /** the constraints the `c` helper declared, with comb's defaults applied; for a plain drizzle column, every flag off but `private` on a name starting with `_` and `autogenerate` on a filled timestamp key */
    constraints: FieldConstraintsMeta;
    /** drizzle's data type: `string`, `number int53`, `object json` */
    dataType: string;
    /** the `c` helper's name, else drizzle's column class */
    drizzleHelper: string;
    /** the values of a c.enum, a pgEnum or an enum text column; null for any other */
    enumValues: string[] | null;
    /** State machine declared as c.enum's third argument, null when absent */
    states: StateMachineMeta | null;
    /** FK reference if .references(() => table.column) */
    foreignKey: ForeignKeyRef | null;
    /** generated always as an expression: the database writes it */
    generated: boolean;
    /** a default value, default SQL or default function fills it when a write leaves it out */
    hasDefault: boolean;
    /** an identity column, always or by default */
    identity: boolean;
    /** a column of the table's composite primary key */
    inCompositeKey: boolean;
    /** the column is NOT NULL: a write must give it a value or leave it to a default */
    isNotNull: boolean;
    /** the table's single-column primary key */
    isPrimaryKey: boolean;
    /** the name the generated DTO imports a c.json column's schema under, null for any other column */
    jsonSchemaName: string | null;
    /** the names the field's generated schema uses and must import (a c.json schema) */
    schemaImports: ImportRef[];
    /** a Postgres array's declared dimensions; 0 or absent for a column of one value */
    dimensions?: number | undefined;
    /** whether a Postgres array's elements may be null, as its drizzle type declares; false or absent otherwise */
    nullElements?: boolean | undefined;
    /** the length the column itself declares (a varchar(n), a SQLite text({ length })); null when none. A `c` helper's declared max is in `constraints` */
    length: number | null;
    /** Field name as defined in the table schema object */
    name: string;
    /** c.id's prefix; null for any other column */
    prefix: string | null;
    /** an update hook fills it on every update */
    onUpdate: boolean;
    /** The column's SQL name: what the engine reports in errors and what DDL and triggers name */
    sqlName: string;
    /** the column's SQL type, as DDL writes it */
    sqlType: string;
    /** the name of the column's own `.unique()`, drizzle's when it gives none; null when it has none */
    unique: string | null;
};
/** Unique index from table options (3rd arg arrow function) */
type UniqueIndexMeta = {
    /** the indexed columns' property names, in index order */
    columns: string[];
    /** the index's name, which a violation of it reports */
    name: string;
};
/** A partial (`.where()`) unique index: its columns are unique only among the rows `where` matches */
type PartialUniqueIndexMeta = UniqueIndexMeta & {
    /** the predicate, as the index's SQL holds it */
    where: string;
};
/** A unique index whose keys are not all plain columns, keyed in the constraint map by its name */
type ExpressionIndexMeta = {
    /** its keys as SQL */
    expression: string;
    /** the index's name, which a violation of it reports */
    name: string;
};
/** Check constraint from table options */
type CheckConstraintMeta = {
    /** the condition every row must meet, as SQL */
    expression: string;
    /** the constraint's name, which a violation of it reports */
    name: string;
};
/** Timestamp tracking for a table */
type TimestampsMeta = {
    /** the creation time is comb's c.createdAt(), read back and NOT NULL, so the generated schemas use comb's shared timestamp fields for it */
    createdAt: boolean;
    /** the soft-delete time is comb's c.deletedAt(), read back and nullable, so the generated schemas use comb's shared timestamp fields for it */
    deletedAt: boolean;
    /** the property name of each epoch-millisecond column named createdAt, updatedAt or deletedAt in either casing; null when the table has none */
    names: {
        /** the creation time's property name, `createdAt` or `created_at`; null when none */
        createdAt: string | null;
        /** the soft-delete time's property name, `deletedAt` or `deleted_at`; null when none */
        deletedAt: string | null;
        /** the update time's property name, `updatedAt` or `updated_at`; null when none */
        updatedAt: string | null;
    };
    /** the casing every timestamp shares; mixed when they differ, which no shared spread fits */
    style: "camelCase" | "mixed" | "snake_case";
    /** the update time is comb's c.updatedAt(), read back and NOT NULL, so the generated schemas use comb's shared timestamp fields for it */
    updatedAt: boolean;
};
/** Complete metadata for a single table */
type TableMeta = {
    /** Check constraints from table options */
    checkConstraints: CheckConstraintMeta[];
    /** All fields in the table */
    fields: FieldMeta[];
    /** unique indexes on an expression */
    expressionIndexes: ExpressionIndexMeta[];
    /** Whether composite PK exists in table options */
    hasCompositePrimaryKey: boolean;
    /** Entity ID prefix from c.id("prefix"), null if c.serialId() or missing */
    idPrefix: string | null;
    /** the Postgres schema of a pgSchema's table; null for the default */
    schema: string | null;
    /** SQL table name (first arg to createTable) */
    sqlName: string;
    /** Detected timestamp fields and style */
    timestamps: TimestampsMeta;
    /** Partial (`.where()`) unique indexes on plain columns: the constraint map names their violations, and they are listed apart from the table's unique indexes */
    partialUniqueIndexes: PartialUniqueIndexMeta[];
    /** Unique indexes from table options */
    uniqueIndexes: UniqueIndexMeta[];
    /** Variable name (export const X = createTable...) */
    varName: string;
};
/** Imports discovered during analysis */
type ImportsMeta = {
    /** Enum constant name -> module path */
    enumImports: Map<string, string>;
    /** All imported identifiers -> module path */
    identifiers: Map<string, string>;
    /** JSON schema names referenced by c.json() */
    jsonSchemaImports: Set<string>;
};
/** What a constraint can hold */
type ConstantValue = boolean | number | string;
/** Full analysis result */
type AnalysisResult = {
    /** Database name derived from tables file path */
    dbName: string;
    /** The engine the tables are built for, by the drizzle builder each one's file imports; null when none says */
    dialect: "postgres" | "sqlite" | null;
    /** the names the tables file imports, which generated files import from the same modules */
    imports: ImportsMeta;
    /** every table the tables file exports, in export order */
    tables: TableMeta[];
    /** Relative import path for the tables file */
    tablesImportPath: string;
};
export type { ConstantValue, ExpressionIndexMeta, ImportRef, AnalysisResult, CheckConstraintMeta, FieldConstraintsMeta, FieldMeta, ForeignKeyRef, ImportsMeta, StateMachineMeta, TableMeta, TimestampsMeta, PartialUniqueIndexMeta, UniqueIndexMeta, };

// meta/read-indexes.d.ts
/** The unique indexes an entity payload publishes, read as readCombEntityMeta reads them: absent, none; malformed, refused */
import type { PartialUniqueIndexMeta, UniqueIndexMeta } from "./contract.ts";
declare const isRecord: (value: unknown) => value is Record<string, unknown>;
declare const isStringArray: (value: unknown) => value is string[];
/**
 * `uniqueIndexes` arrived in a later comb than the contract's v1, so a payload
 * without it is normal and reads as [] — absence is not an error, only a
 * malformed value is.
 */
declare function readUniqueIndexes(raw: unknown): UniqueIndexMeta[] | "invalid";
/**
 * `partialUniqueIndexes` arrived in 0.15, after `uniqueIndexes`: absent, it reads as []; each needs
 * its predicate, without which it would read as the uniqueness it is not (C7)
 */
declare function readPartialUniqueIndexes(raw: unknown): PartialUniqueIndexMeta[] | "invalid";
export { isRecord, isStringArray, readPartialUniqueIndexes, readUniqueIndexes };

// meta/column.d.ts
/**
 * What a `c` helper records on the column it builds, for codegen to read at runtime: the helper's
 * kind and the very values it was given. Kept under a registered symbol on drizzle's builder
 * config, which the built column keeps, so a second copy of comb reads what the first wrote.
 */
/** The `c` helpers, by name */
type CombColumnKind = "boolean" | "createdAt" | "deletedAt" | "enum" | "id" | "integer" | "json" | "real" | "ref" | "serialId" | "text" | "timestamp" | "updatedAt";
/** A column's comb metadata: its helper, and the constraints, enum values, states, schema or prefix it took */
type CombColumnMeta = {
    constraints?: Readonly<Record<string, unknown>> | undefined;
    kind: CombColumnKind;
    prefix?: string | undefined;
    schema?: unknown;
    states?: unknown;
    values?: readonly string[] | undefined;
};
/** `builder` with `meta` recorded on its config; the options a helper was not given are left out */
declare function withCombColumn<B>(builder: B, meta: {
    [K in keyof CombColumnMeta]: CombColumnMeta[K] | undefined;
}): B;
/** The comb metadata a built column (or its builder) carries; undefined for a column no `c` helper built */
declare function combColumnMeta(column: unknown): CombColumnMeta | undefined;
/** A c.enum's options as its column's record: the flags its constraints, the rest its state machine, none when it names no state (F9) */
declare function enumMeta(values: readonly string[], options?: {
    nomutate?: boolean | undefined;
    private?: boolean | undefined;
    tenant?: boolean | undefined;
}): {
    [K in keyof CombColumnMeta]: CombColumnMeta[K] | undefined;
};
export { combColumnMeta, enumMeta, withCombColumn, type CombColumnKind, type CombColumnMeta };
