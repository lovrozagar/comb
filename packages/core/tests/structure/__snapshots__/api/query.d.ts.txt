// query/cursor.d.ts
import { type CursorOptions } from "./cursor-seal.ts";

/**
 * Column keyset pagination breaks ties on.
 *
 * This is the single source of truth for the tiebreak: the SQL builders resolve
 * the column by this name, and `createListQuerySchema` publishes it as
 * `stableTiebreak`. Keeping one constant is what stops the published fact from
 * drifting away from the query that is actually run.
 *
 * A table whose primary-key property is named something else has no cursor
 * predicate applied at all — rows then duplicate and skip across pages — so
 * `validateTables()` rejects that shape rather than letting it fail silently.
 */
declare const CURSOR_TIEBREAK_COLUMN = "id";
/**
 * The longest cursor a list takes, and so the longest comb mints: it fits a URL. A cursor carries
 * its sort values, so a long text key could mint one past it, which the next page would refuse.
 */
declare const MAX_CURSOR_LENGTH = 8192;
/**
 * Wire payload. `s` is the sort signature the cursor was minted under, `v` one
 * value per sort key in order, `i` the tiebreak id. A cursor is only valid for
 * the exact sort that produced it.
 */
type CursorPayload = {
    i: CursorId;
    s: string;
    v: unknown[];
};
/** The tiebreak id as the row held it: a number stays a number, a bigint a bigint, so it compares as one */
type CursorId = bigint | number | string;
type DecodedCursor = {
    id: CursorId;
    sort: string;
    values: CursorValue[];
};
/**
 * A sort value carried by reference (cursor-ref.ts): the prefix of it that fit and the whole
 * value's length, both in code points as SQLite's substr() and Postgres's left() count them; the
 * rest is read from the row while it still has that prefix and length
 */
declare class CursorRef {
    readonly length: number;
    readonly prefix: string;
    constructor(prefix: string, length: number);
}
/** What a sort key can hold and its database can compare, or a long one carried by reference */
type CursorValue = bigint | boolean | CursorRef | Date | number | string | null;
type PaginationMeta = {
    /** Rows the filter matches across all pages; null when the response cannot know */
    count: number | null;
    hasMore: boolean;
    limit: number;
    nextCursor: string | null;
    page: number | null;
};
/** PostgreSQL semantics: nulls sort last ascending and first descending. */
declare function effectiveNulls(direction: SortDirection, nulls?: SortNulls): SortNulls;
/** Canonical form of a sort — what a cursor is bound to. */
declare function sortSignature(parsedSort: readonly SortField[]): string;
/** A sealed cursor for `payload`, opened only under the same `options` */
declare function encodeCursor(payload: CursorPayload, options: CursorOptions): string;
/** Whether `value` is an integer a JS number rounds: past ±(2^53 − 1), a lossy read of the row's key (U22) */
declare const lossy: (value: unknown) => value is number;
/**
 * Mint the cursor for the row after which the next page starts, sealed under `options` and keyed
 * by `idField`, whole: the public `createCursor` (cursor-mint.ts) keeps it within the cap (U37)
 */
declare function mintCursor(record: Record<string, unknown>, parsedSort: readonly SortField[], options: CursorOptions & {
    idField?: string | undefined;
}): string;
/**
 * Why `cursor` cannot be one, or null when its shape can: the list schema's check, a 400 at
 * validation. What it holds, and the sort it was minted under, only readCursor can open.
 */
declare function cursorProblem(cursor: string): string | null;
/** The cursor sealed under `options`, opened and decoded; null when it was not, or it is no cursor */
declare function openCursor(cursor: string, options: CursorOptions): DecodedCursor | null;
/**
 * Open a cursor under `options` and decode it for this sort, or throw a 400 CombError. A value may
 * be a CursorRef: a long sort value carried by reference, its prefix and length, which a keyset
 * of its own must read from the cursor's row (cursor-ref.ts), never bind as it is.
 */
declare function readCursor(cursor: string, parsedSort: readonly SortField[], options: CursorOptions): DecodedCursor;
export { CursorRef, CURSOR_TIEBREAK_COLUMN, MAX_CURSOR_LENGTH, cursorProblem, effectiveNulls, encodeCursor, lossy, mintCursor, readCursor, sortSignature, type CursorId, type CursorOptions, type CursorPayload, type CursorValue, type DecodedCursor, openCursor, type PaginationMeta, };

// query/cursor-seal.d.ts
/** What seals and opens the cursors of one list: the app's secret, and the list's scope (route plus tenant) */
type CursorOptions = {
    cursorScope: string;
    cursorSecret: string;
    /**
     * "exact" when the driver reads a SQLite REAL's ±Infinity back as Infinity (a local libsql file,
     * bun:sqlite), so a null cursor is a real NULL and a list whose NULL anchor is deleted resumes
     * right after it. Unset, as on D1, which reads ±Infinity as null, such a list resumes from the
     * earliest value the null may stand for, repeating rows and skipping none (U24).
     */
    nonFinite?: "exact" | undefined;
};
/** The shortest secret a cursor is sealed under, in UTF-8 bytes: the key's length */
declare const MIN_CURSOR_SECRET_BYTES = 32;
/** The 32-byte key HKDF-SHA256 derives from `secret`, with no salt and comb's info */
declare function cursorKey(secret: string): Uint8Array;
/** Throw the 500 sealing throws when `options` cannot seal a cursor */
declare function assertCursorWired(options: CursorOptions): void;
/**
 * `plaintext` sealed under `options`, as base64url, padded with spaces up to a multiple of 256
 * bytes first (U9): AEAD hides the bytes, not their count, and a cursor carries a hidden column's
 * value too. openSealed strips them: what seal takes ends in no space of its own.
 */
declare function seal(plaintext: Uint8Array, options: CursorOptions): string;
/** Whether `cursor` has a sealed cursor's shape: what a check without the secret can know */
declare const isSealedShape: (cursor: string) => boolean;
/** What `cursor` holds, opened under `options`; null when it was not sealed under both */
declare function openSealed(cursor: string, options: CursorOptions): Uint8Array | null;
export { assertCursorWired, cursorKey, type CursorOptions, isSealedShape, MIN_CURSOR_SECRET_BYTES, openSealed, seal };

// query/filter.d.ts
/**
 * PostgREST-style filters and sort orders: parse and validate. DB-agnostic: no SQL, only
 * the AST and its checks. The grammar lives in filter-parse.ts, the checks in filter-validate.ts.
 */
import { FILTER_OPERATORS, parseFilter } from "./filter-parse.ts";
import { FILTER_LIMITS, OPERATORS_BY_TYPE, parseFilterNumber, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, validateFilter } from "./filter-validate.ts";
import type { SortField } from "./types.ts";
/**
 * Parse PostgREST-style `order` string into SortField array.
 *
 * Grammar:
 *   order  := entry ("," entry)*
 *   entry  := field ("." direction)? ("." nulls)?
 *   field  := any string not containing "." at the direction/nulls boundary
 *            (dotted relation paths OK: "author.name.asc")
 *   direction := "asc" | "desc"        (default "asc" when omitted)
 *   nulls     := "nullsfirst" | "nullslast"
 *
 * Examples:
 *   "name"                         → [{ field: "name", direction: "asc" }]
 *   "name.desc"                    → [{ field: "name", direction: "desc" }]
 *   "name.asc.nullslast"           → [{ field: "name", direction: "asc", nulls: "last" }]
 *   "col1.desc,col2.asc.nullsfirst"
 *     → [{field:"col1",direction:"desc"}, {field:"col2",direction:"asc",nulls:"first"}]
 *
 * Malformed entries are dropped silently — downstream schema validator
 * rejects unknown fields, which catches most mistakes.
 */
declare function parseOrder(input: string | null | undefined): SortField[];
export { FILTER_LIMITS, FILTER_OPERATORS, OPERATORS_BY_TYPE, parseFilter, parseFilterNumber, parseOrder, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, };

// query/like.d.ts
/**
 * PostgREST like/ilike: `*` is the wildcard, every other character is literal.
 *
 * `ilike` lowers to SQL LIKE (ASCII case-insensitive in SQLite), so LIKE's own
 * specials `%` and `_` are escaped before `*` expands to `%`.
 */
declare function likePattern(value: string): string;
/**
 * `like` lowers to SQLite GLOB, which is case-sensitive. GLOB has no escape
 * character: `?` and `[` are made literal by wrapping them in a one-character
 * class, and `*` stays the wildcard.
 */
declare function globPattern(value: string): string;
/**
 * `text` as a literal inside a LIKE pattern written with `ESCAPE '\\'`: the escape character and
 * LIKE's own specials `%` and `_` each escaped, so `%${escapeLike(term)}%` finds `term` as written
 */
declare function escapeLike(text: string): string;
/** A value's stars only at its ends: whether each end is open, and the text between */
declare const FIXED: RegExp;
/** The pattern SQLite matches `value` with under `operator`, or null for a fixed text, which is found with none (U21) */
declare const patternOf: (operator: "like" | "ilike", value: string) => string | null;
export { escapeLike, FIXED, globPattern, likePattern, patternOf };

// meta/column.d.ts
/**
 * What a `c` helper records on the column it builds, for codegen to read at runtime: the helper's
 * kind and the very values it was given. Kept under a registered symbol on drizzle's builder
 * config, which the built column keeps, so a second copy of comb reads what the first wrote.
 */
/** The `c` helpers, by name */
type CombColumnKind = "boolean" | "createdAt" | "deletedAt" | "enum" | "id" | "integer" | "json" | "real" | "ref" | "serialId" | "text" | "timestamp" | "updatedAt";
/** A column's comb metadata: its helper, and the constraints, enum values, states, schema or prefix it took */
type CombColumnMeta = {
    constraints?: Readonly<Record<string, unknown>>;
    kind: CombColumnKind;
    prefix?: string;
    schema?: unknown;
    states?: unknown;
    values?: readonly string[];
};
/** `builder` with `meta` recorded on its config; the options a helper was not given are left out */
declare function withCombColumn<B>(builder: B, meta: {
    [K in keyof CombColumnMeta]: CombColumnMeta[K] | undefined;
}): B;
/** The comb metadata a built column (or its builder) carries; undefined for a column no `c` helper built */
declare function combColumnMeta(column: unknown): CombColumnMeta | undefined;
/** A c.enum's options as its column's record: the flags its constraints, the rest its state machine, none when it names no state (F9) */
declare function enumMeta(values: readonly string[], options?: {
    nomutate?: boolean | undefined;
    private?: boolean | undefined;
    tenant?: boolean | undefined;
}): {
    [K in keyof CombColumnMeta]: CombColumnMeta[K] | undefined;
};
export { combColumnMeta, enumMeta, withCombColumn, type CombColumnKind, type CombColumnMeta };
