// codegen/analyzer.d.ts
/**
 * The analysis every generator consumes: the tables module loaded at runtime in codegen's own
 * child (runtime/load.ts), each table read as drizzle built it and as comb's `c` helpers recorded
 * it (runtime/read-tables.ts), then named as the generators ask (runtime/to-meta.ts). Generators
 * never load or read source themselves.
 */
import type { AnalysisResult, TableMeta } from "./analyzer-types.ts";
/**
 * Load and analyze a tables file. Loading runs the module in a child process, so it must be safe to
 * import: no top-level connection, server or side effect
 */
declare function analyze(tablesPath: string, { cwd, }?: {
    /** the working directory the module runs in, the one the app runs in; the current process's when omitted */
    cwd?: string | undefined;
}): Promise<AnalysisResult>;
export { analyze };
export type { AnalysisResult, TableMeta };

/** One database's codegen: its tables file and what is generated from it */
export type DatabaseCodegen = {
    /** the database's dialect; the tables' own when omitted, else `sqlite`. One the tables contradict throws */
    dialect?: OrmConfig["dialect"] | undefined;
    /** the driver the orm client is written for; the orm generator writes nothing without it */
    driver?: OrmConfig["driver"] | undefined;
    /** The stem of every generated file name; `db.<name>` when omitted */
    filePrefix?: string | undefined;
    /** SQLite full-text search: each table, by its SQL name, to its index; the fts generator writes nothing without it */
    fts?: Record<string, FtsTableConfig> | undefined;

    /** Where each generator writes; `default` is the directory for every one not named */

    } | undefined;
    /**
     * Postgres only: each partitioned table, keyed as its constraint map keys it, to its partitions'
     * names, which live in its schema. Postgres reports a violation on the partition, and the map
     * reads it as the parent's; a violation on a partition not listed names no column
     */
    partitions?: Record<string, readonly string[]> | undefined;
    /** The tables file, relative to comb.config.ts */
    tables: string;
};

export type CodegenConfig = {
    /** each database, by the name `comb codegen` and `comb migrate diff` select it by and its generated files take (`db.<name>`) */

/**
 * TableMeta — serializable intermediate format produced by TableAnalyzer.
 * No ts-morph or Drizzle dependencies. All generators consume this.
 */
/** Constraint annotations parsed from field type arguments */
type FieldConstraintsMeta = {
    /** comb or the database fills the column, so the generated create and update DTOs leave it out; on by default for a column a default or hook fills */
    autogenerate: boolean;
    /** the generated DTOs take only an email address */
    email: boolean;
    /** an https URL or a data:image/ URI */
    imageUri: boolean;
    /** the generated DTOs lowercase the text before they check it */
    lowercase: boolean;
    /** the largest value the generated DTOs take: a text column's length in characters, a timestamp's epoch milliseconds, any other column's value; null when unbounded */
    max: number | null;
    /** the most bytes a c.json value may take once stringified; null when unbounded */
    maxBytes: number | null;
    /** the smallest value the generated DTOs take, measured as `max` is; null when unbounded */
    min: number | null;
    /** a name: letters, marks, numbers, spaces and punctuation */
    name: boolean;
    /** a foreign key the generated update DTO takes, so a row can move to another parent; every other reference is set once, at creation */
    mutable: boolean;
    /** set at creation and never changed: the generated update DTO leaves it out */
    nomutate: boolean;
    /** a stored password hash: kept as written and held only to its bounds and pattern */
    password: boolean;
    /** the regex the generated DTOs hold the text to, as a `/source/flags` literal with the `u` flag; null when none */
    pattern: string | null;
    /** never read back: the generated read DTOs and the meta stamp leave it out and the manifest flags it `private: true`; also set for a field whose name starts with `_` */
    private: boolean;
    /** the name the generated relation of this foreign key takes, where the column's role would collide with another; null or absent for the role's own name */
    relation?: string | null | undefined;
    /** the column that scopes the table's rows to a tenant, set once at creation; declared, never inferred, and at most one per table */
    tenant: boolean;
    /** the generated DTOs trim surrounding whitespace before they check the text */
    trim: boolean;
    /** the generated DTOs uppercase the text before they check it */
    uppercase: boolean;
    /** the generated DTOs take only a URL */
    url: boolean;
};
/** State machine declared on an enum column */
type StateMachineMeta = {
    /** the state a new row starts in; null when the declaration names none */
    initial: string | null;
    /** the states a row never leaves; null when the declaration names none */
    terminal: string[] | null;
    /** each state, to the states a row may move to from it; null when the declaration names none */
    transitions: Record<string, string[]> | null;
};
/** A name a generated file must import, with the module the tables file gets it from. */
type ImportRef = {
    /** the module specifier as the tables file writes it */
    from: string;
    /** the exported name (`*` for a namespace import, `default` for a default import) */
    imported: string;
    /** the local name the tables file uses */
    name: string;
    /** the module's file when it is in the project; null for a package */
    path: string | null;
};
/** Foreign key reference from this field to another table */
type ForeignKeyRef = {
    /** the referencing column's property name in its table */
    column: string;
    /** the action the database takes on this row when the referenced row is deleted, as drizzle declares it (`cascade`, `set null`); absent for the database's default */
    onDelete?: string | undefined;
    /** the action the database takes on this row when the referenced key changes, as drizzle declares it; absent for the database's default */
    onUpdate?: string | undefined;
    /** the referenced column's property name in its table */
    refColumn: string;
    /** the referenced table's export name in the tables file */
    refTable: string;
    /** whether the referenced column is a c.id key, so the reference takes the same id format and prefix */
    toId?: boolean | undefined;
    /** a composite foreign key's columns; absent for a single-column key */
    key?: {
        /** the referencing property names, in key order */
        columns: string[];
        /** the referenced property names, each paired by position with `columns` */
        refColumns: string[];
    } | undefined;
};
/** A single column/field in a table, as drizzle built it and its `c` helper recorded it */
type FieldMeta = {
    /** SQLite's AUTOINCREMENT on an integer key */
    autoIncrement: boolean;
    /** drizzle's column class: `SQLiteText`, `PgVarchar` */
    columnType: string;
    /** the `c` helper that built the column; null for a plain drizzle column */
    combKind: CombColumnKind | null;
    /** the constraints the `c` helper declared, with comb's defaults applied; every flag off for a plain drizzle column */
    constraints: FieldConstraintsMeta;
    /** drizzle's data type: `string`, `number int53`, `object json` */
    dataType: string;
    /** the `c` helper's name, else drizzle's column class */
    drizzleHelper: string;
    /** the values of a c.enum, a pgEnum or an enum text column; null for any other */
    enumValues: string[] | null;
    /** State machine declared as c.enum's third argument, null when absent */
    states: StateMachineMeta | null;
    /** FK reference if .references(() => table.column) */
    foreignKey: ForeignKeyRef | null;
    /** generated always as an expression: the database writes it */
    generated: boolean;
    /** a default value, default SQL or default function fills it when a write leaves it out */
    hasDefault: boolean;
    /** an identity column, always or by default */
    identity: boolean;
    /** a column of the table's composite primary key */
    inCompositeKey: boolean;
    /** the column is NOT NULL: a write must give it a value or leave it to a default */
    isNotNull: boolean;
    /** the table's single-column primary key */
    isPrimaryKey: boolean;
    /** the name the generated DTO imports a c.json column's schema under, null for any other column */
    jsonSchemaName: string | null;
    /** the names the field's generated schema uses and must import (a c.json schema) */
    schemaImports: ImportRef[];
    /** a Postgres array's declared dimensions; 0 or absent for a column of one value */
    dimensions?: number | undefined;
    /** whether a Postgres array's elements may be null, as its drizzle type declares; false or absent otherwise */
    nullElements?: boolean | undefined;
    /** the length the column itself declares (a varchar(n), a SQLite text({ length })); null when none. A `c` helper's declared max is in `constraints` */
    length: number | null;
    /** Field name as defined in the table schema object */
    name: string;
    /** c.id's prefix; null for any other column */
    prefix: string | null;
    /** an update hook fills it on every update */
    onUpdate: boolean;
    /** The column's SQL name: what the engine reports in errors and what DDL and triggers name */
    sqlName: string;
    /** the column's SQL type, as DDL writes it */
    sqlType: string;
    /** the name of the column's own `.unique()`, drizzle's when it gives none; null when it has none */
    unique: string | null;
};
/** Unique index from table options (3rd arg arrow function) */
type UniqueIndexMeta = {
    /** the indexed columns' property names, in index order */
    columns: string[];
    /** the index's name, which a violation of it reports */
    name: string;
};
/** A partial (`.where()`) unique index: its columns are unique only among the rows `where` matches */
type PartialUniqueIndexMeta = UniqueIndexMeta & {
    /** the predicate, as the index's SQL holds it */
    where: string;
};
/** A unique index whose keys are not all plain columns, keyed in the constraint map by its name */
type ExpressionIndexMeta = {
    /** its keys as SQL */
    expression: string;
    /** the index's name, which a violation of it reports */
    name: string;
};
/** Check constraint from table options */
type CheckConstraintMeta = {
    /** the condition every row must meet, as SQL */
    expression: string;
    /** the constraint's name, which a violation of it reports */
    name: string;
};
/** Timestamp tracking for a table */
type TimestampsMeta = {
    /** the creation time is comb's c.createdAt(), read back and NOT NULL, so the generated schemas use comb's shared timestamp fields for it */
    createdAt: boolean;
    /** the soft-delete time is comb's c.deletedAt(), read back and nullable, so the generated schemas use comb's shared timestamp fields for it */
    deletedAt: boolean;
    /** the property name of each epoch-millisecond column named createdAt, updatedAt or deletedAt in either casing; null when the table has none */
    names: {
        /** the creation time's property name, `createdAt` or `created_at`; null when none */
        createdAt: string | null;
        /** the soft-delete time's property name, `deletedAt` or `deleted_at`; null when none */
        deletedAt: string | null;
        /** the update time's property name, `updatedAt` or `updated_at`; null when none */
        updatedAt: string | null;
    };
    /** the casing every timestamp shares; mixed when they differ, which no shared spread fits */
    style: "camelCase" | "mixed" | "snake_case";
    /** the update time is comb's c.updatedAt(), read back and NOT NULL, so the generated schemas use comb's shared timestamp fields for it */
    updatedAt: boolean;
};
/** Complete metadata for a single table */
type TableMeta = {
    /** Check constraints from table options */
    checkConstraints: CheckConstraintMeta[];
    /** All fields in the table */
    fields: FieldMeta[];
    /** unique indexes on an expression */
    expressionIndexes: ExpressionIndexMeta[];
    /** Whether composite PK exists in table options */
    hasCompositePrimaryKey: boolean;
    /** Entity ID prefix from c.id("prefix"), null if c.serialId() or missing */
    idPrefix: string | null;
    /** the Postgres schema of a pgSchema's table; null for the default */
    schema: string | null;
    /** SQL table name (first arg to createTable) */
    sqlName: string;
    /** Detected timestamp fields and style */
    timestamps: TimestampsMeta;
    /** Partial (`.where()`) unique indexes on plain columns: the constraint map names their violations, and they are listed apart from the table's unique indexes */
    partialUniqueIndexes: PartialUniqueIndexMeta[];
    /** Unique indexes from table options */
    uniqueIndexes: UniqueIndexMeta[];
    /** Variable name (export const X = createTable...) */
    varName: string;
};
/** Imports discovered during analysis */
type ImportsMeta = {
    /** Enum constant name -> module path */
    enumImports: Map<string, string>;
    /** All imported identifiers -> module path */
    identifiers: Map<string, string>;
    /** JSON schema names referenced by c.json() */
    jsonSchemaImports: Set<string>;
};
/** What a constraint can hold */
type ConstantValue = boolean | number | string;
/** Full analysis result */
type AnalysisResult = {
    /** Database name derived from tables file path */
    dbName: string;
    /** The engine the tables are built for, by the drizzle builder each one's file imports; null when none says */
    dialect: "postgres" | "sqlite" | null;
    /** the names the tables file imports, which generated files import from the same modules */
    imports: ImportsMeta;
    /** every table the tables file exports, in export order */
    tables: TableMeta[];
    /** Relative import path for the tables file */
    tablesImportPath: string;
};
export type { ConstantValue, ExpressionIndexMeta, ImportRef, AnalysisResult, CheckConstraintMeta, FieldConstraintsMeta, FieldMeta, ForeignKeyRef, ImportsMeta, StateMachineMeta, TableMeta, TimestampsMeta, PartialUniqueIndexMeta, UniqueIndexMeta, };

// codegen/annotations.d.ts
/**
 * A column's constraints, from the record its `c` helper was given: every key one the column
 * types allow, a pattern one comb can emit, and `autogenerate` defaulting to true on a column
 * something other than the caller fills.
 */
import type { CombColumnKind } from "../meta/column.ts";
import type { ConstantValue, FieldConstraintsMeta } from "./analyzer-types.ts";
/** Field constraints extracted from type arguments */
export type FieldConstraints = FieldConstraintsMeta;
export declare const NUMBER_KEYS: readonly ["autogenerate", "max", "min", "mutable", "nomutate", "private", "relation", "tenant"];
export declare const JSON_KEYS: readonly ["autogenerate", "maxBytes", "mutable", "nomutate", "private", "relation", "tenant"];
export declare const ENUM_KEYS: readonly ["nomutate", "private", "tenant"];
export declare const TEXT_KEYS: readonly ["autogenerate", "email", "imageUri", "lowercase", "max", "min", "mutable", "name", "nomutate", "password", "pattern", "private", "relation", "tenant", "trim", "uppercase", "url"];
type Keys = typeof NUMBER_KEYS;
type None = readonly [];
/** The constraint keys each c helper takes, its constraint type's; a state machine is c.enum's own (CD9 of r22b) */
export type ConstraintKeys = {
    readonly boolean: None;
    readonly createdAt: Keys;
    readonly deletedAt: Keys;
    readonly enum: typeof ENUM_KEYS;
    readonly id: None;
    readonly integer: Keys;
    readonly json: typeof JSON_KEYS;
    readonly real: Keys;
    readonly ref: typeof TEXT_KEYS;
    readonly serialId: None;
    readonly text: typeof TEXT_KEYS;
    readonly timestamp: Keys;
    readonly updatedAt: Keys;
};
export declare const CONSTRAINT_KEYS: ConstraintKeys;
/**
 * A column's declared constraints, checked and with comb's defaults applied. Throws on a key the
 * helper does not take, a value of the wrong type, two keys that contradict, or bounds no value
 * meets, each message prefixed with `place`
 */
export declare function fieldConstraintsOf(constraints: Readonly<Record<string, ConstantValue>>, { fieldName, filled, helper, measure, place, }: {
    /** the column's property name; one starting with `_` is private */
    fieldName: string;
    /** whether a default or hook fills the column, which turns `autogenerate` on unless it is declared false */
    filled: boolean;
    /** the c helper that built the column, whose constraint type names the keys it takes; null for a drizzle column, which takes none */
    helper: CombColumnKind | null;
    /** what `min` and `max` bound: a text's length, or the value itself */
    measure: "length" | "value";
    /** where the column is declared, as each error message names it */
    place: string;
}): FieldConstraints;
export {};

/**

/**

    schema?: unknown;

/** The SQL columns of an FTS index configured on `table` as `columns`, each once */
declare const ftsColumns: (table: TableMeta, columns: readonly string[]) => string[];

// meta.d.ts
/**
 * The comb meta contract — facts about tables and queries, stamped onto Zod
 * schemas so they survive the trip to JSON Schema and into an OpenAPI document.
 *
 * comb publishes facts in comb's vocabulary. Mapping them to a consumer's tag
 * names is the consumer's job. See docs/meta-contract.md.
 */
import { COMB_FILTER_GRAMMAR, COMB_META_KEY, COMB_META_VERSION, type CombEntityMeta, type CombEntityMetaInput, type CombFilterField, type CombFilterGrammar, type CombMeta, type CombMetaInput, type CombMetaKind, type CombMetaStamp, type CombQueryMeta, type CombQueryMetaInput, type CombStatesMeta, type UniqueIndexMeta } from "./meta/contract.ts";
import { readCombEntityMeta, readCombMeta, readCombQueryMeta } from "./meta/read.ts";
/**
 * Build the stamp for a Zod `.meta()` call.
 *
 * Stamp the schema you RETURN, not the one you build from — a transform/pipe
 * does not forward the base's metadata to the output view. See docs §3.1.
 */
declare function combMeta(meta: CombEntityMetaInput): CombMetaStamp<CombEntityMeta>;
/** Build the stamp for a list query schema's `.meta()` call; `v` is stamped for you */
declare function combMeta(meta: CombQueryMetaInput): CombMetaStamp<CombQueryMeta>;
/** Minimal shape of a Zod schema, so this module needs no zod import. */
export type MetaCarrier<T> = {
    /** Zod's `.meta()`: returns a copy of the schema carrying `m` */
    meta: (m: Record<string, unknown>) => T;
};
/**
 * Re-stamp `to` with the facts `from` carries.
 *
 * `.extend()` drops registry metadata silently (docs §5), so a consumer who
 * extends a generated read schema loses the entity facts with no diagnostic.
 * This is the recovery path.
 */
declare function carryCombMeta<T>(from: unknown, to: MetaCarrier<T>): T | null;

// meta/contract.d.ts
/**
 * The comb meta contract — facts about tables and queries, stamped onto Zod
 * schemas so they survive the trip to JSON Schema and into an OpenAPI document.
 *
 * comb publishes facts in comb's vocabulary. Mapping them to a consumer's tag
 * names is the consumer's job. See docs/meta-contract.md.
 */
import type { PartialUniqueIndexMeta, UniqueIndexMeta } from "../codegen/analyzer-types.ts";
/** Reserved key. One collision surface, one version field. */
declare const COMB_META_KEY = "x-comb";
/**
 * Contract version. Bumped ONLY on a breaking change — a field removed, or a
 * field whose meaning changed. Adding a field never bumps it, which is what
 * lets comb and its consumers ship independently.
 */
declare const COMB_META_VERSION = 1;
/** Which payload a stamp carries: facts about a table (`entity`) or about a list query (`query`) */
type CombMetaKind = "entity" | "query";
/** Filter grammar comb's parser implements — `field.op.value`. */
declare const COMB_FILTER_GRAMMAR = "postgrest";
/** The filter grammar a query meta names, `postgrest`: `field.op.value` */
type CombFilterGrammar = typeof COMB_FILTER_GRAMMAR;
/** Facts about an entity, derived from its table definition. */
type CombEntityMeta = {
    /** The contract version the payload was stamped with, `COMB_META_VERSION` */
    v: number;
    /** Always `entity` */
    kind: "entity";
    /** Entity name — the SQL table name */
    name: string;
    /** Primary key / stable identifier column */
    identity: string;
    /** Server-assigned: omitted from create bodies, expected in responses */
    generated: string[];
    /** Must reject or ignore a mutation */
    immutable: string[];
    /** Tombstone column, or null when the entity is hard-deleted */
    softDelete: string | null;
    /** Column scoping rows to a tenant, or null when comb cannot know (see docs §6.3) */
    tenantColumn: string | null;
    /**
     * Lifecycle of the entity's state column, or null when none is declared.
     *
     * `transitions` is deliberately absent: it is a write-side rule, and
     * publishing the graph would invite a consumer to drive state by walking it.
     * See docs/state-machines.md §5.
     */
    states: CombStatesMeta | null;
    /**
     * The uniqueness that holds over every row: a `uniqueIndex()`, a `unique().on()`
     * constraint and a column's `.unique()`, in the order the analyzer extracted
     * them. Empty when none. Never inferred from PKs or column names, and never a
     * partial index, which `partialUniqueIndexes` holds.
     */
    uniqueIndexes: UniqueIndexMeta[];
    /**
     * Partial (`.where()`) unique indexes: their columns are unique only among the
     * rows `where` matches. Apart from `uniqueIndexes`, so a reader that knows no
     * predicate claims no uniqueness (docs §6.8). Added within v1, so a producer
     * may leave it out; readCombEntityMeta reads its absence as [].
     */
    partialUniqueIndexes?: PartialUniqueIndexMeta[] | undefined;
};
/** The published subset of a column state machine — describes a lifecycle, not how to drive it. */
type CombStatesMeta = {
    /** The column the machine governs */
    column: string;
    /** Every legal value, in declaration order */
    values: string[];
    /** The value a row starts at, or null when the machine declares none */
    initial: string | null;
    /** The values no transition may leave; empty when none are declared */
    terminal: string[];
};
/** One filterable field with the operators its type accepts. */
type CombFilterField = {
    /** The field's name, as a filter names it */
    field: string;
    /** The field's filter type: `string`, `number`, `boolean`, `date` or `enum` */
    type: string;
    /** The operators a filter may apply to the field, such as `eq` or `in` */
    ops: string[];
};
/**
 * Facts about a list query, derived from the config that parses the request.
 * Fields marked optional were added after v1 shipped; a v1 reader built before
 * them ignores them, which is why adding them did not bump `v`.
 */
type CombQueryMeta = {
    /** The contract version the payload was stamped with, `COMB_META_VERSION` */
    v: number;
    /** Always `query` */
    kind: "query";
    /** The fields a `filter` may name */
    filterable: string[];
    /** `filterable` with types and operators */
    filterFields?: CombFilterField[] | undefined;
    /** The fields an `order` may name */
    sortable: string[];
    /** Null placements every sortable field accepts */
    nulls?: Array<"first" | "last"> | undefined;
    /** Conditions one `filter` may hold */
    maxFilterConditions?: number | undefined;
    /** Nested `or()` and `and()` groups one `filter` may hold below its top level */
    maxFilterDepth?: number | undefined;
    /** Values one filter binds in all, every in/nin member counted */
    maxFilterValues?: number | undefined;
    /** Values one `in` or `nin` list may hold */
    maxInValues?: number | undefined;
    /**
     * UTF-8 bytes of the escaped pattern of a `like` or `ilike` value with a `*` inside it, as SQLite
     * reads it: a `[` in a `like` is the three bytes `[[]`, a `%` or `_` in an `ilike` two. A value whose
     * only `*` are at its ends is no pattern and takes no cap. Not the raw value's characters.
     */
    maxPatternBytes?: number | undefined;
    /** Bytes one `filter` may take however a client encodes it, beside its character cap */
    maxFilterBytes?: number | undefined;
    /** Bytes the whole query string comb accepts takes at most, however a client encodes it: 16 KB less a 2 KB path budget */
    maxQueryBytes?: number | undefined;
    /** Whitespace-separated words one `q` may hold */
    maxSearchTerms?: number | undefined;
    /** Bytes one `q` may take however a client encodes it */
    maxSearchBytes?: number | undefined;
    /** Keys one `order` may hold */
    maxSortKeys?: number | undefined;
    /** null = not knowable at this layer, which is not the same as "none" (docs §6.1) */
    searchable: string[] | null;
    /** The fields a `select` may name */
    selectable: string[];
    /** The largest `limit` a request may ask for */
    maxLimit: number;
    /** The last page offset pagination serves; past it, a client pages by cursor */
    maxPage?: number | undefined;
    /** e.g. "created_at.desc" — the order applied when the request names none */
    defaultOrder: string;
    /** Tiebreak column for keyset pagination */
    stableTiebreak: string;
    /** The filter grammar `filter` is parsed with */
    grammar: CombFilterGrammar;
};
/** A comb meta payload: an entity's or a list query's, told apart by `kind` */
type CombMeta = CombEntityMeta | CombQueryMeta;
/** Payload accepted by combMeta() — the version is stamped for you. */
type CombEntityMetaInput = Omit<CombEntityMeta, "v">;
/** A query payload as combMeta() takes it, without `v` */
type CombQueryMetaInput = Omit<CombQueryMeta, "v">;
/** A payload as combMeta() takes it, entity or query, without `v` */
type CombMetaInput = CombEntityMetaInput | CombQueryMetaInput;
/** The object combMeta() produces, ready for `.meta()`. */
type CombMetaStamp<T extends CombMeta = CombMeta> = {
    [COMB_META_KEY]: T;
};
export { COMB_FILTER_GRAMMAR, COMB_META_KEY, COMB_META_VERSION, type CombEntityMeta, type CombEntityMetaInput, type CombFilterField, type CombFilterGrammar, type CombMeta, type CombMetaInput, type CombMetaKind, type CombMetaStamp, type CombQueryMeta, type CombQueryMetaInput, type CombStatesMeta, type PartialUniqueIndexMeta, type UniqueIndexMeta, };

// meta/read.d.ts
/**
 * Reading comb's facts off a JSON Schema. Returns null, never throws: a refused or malformed
 * payload degrades to "no facts", never to a confident wrong answer. See docs/meta-contract.md §4.
 */
import { isRecord } from "./read-indexes.ts";
import { type CombEntityMeta, type CombMeta, type CombQueryMeta } from "./contract.ts";
/** How readCombMeta, readCombEntityMeta and readCombQueryMeta read a payload */
type ReadOptions = {
    /**
     * Highest contract version this reader understands. A payload newer than
     * this is refused rather than partially read — see docs §4.2.
     * Defaults to the version of the comb build doing the reading.
     */
    maxVersion?: number | undefined;
    /** Called instead of throwing when a payload is refused or malformed. */
    onDiagnostic?: ((message: string) => void) | undefined;
};
/**
 * Read comb's facts off the ROOT of a JSON Schema.
 *
 * Does not unwrap `items` for array schemas — the consumer that already
 * resolves which schema to read owns that step, so the two cannot disagree.
 *
 * Returns null, never throws: a refused or malformed payload must degrade to
 * "no facts" (which downstream handles as a visible coverage gap), not to a
 * confident wrong answer.
 */
declare function readCombMeta(jsonSchema: unknown, options?: ReadOptions): CombMeta | null;
/** Narrowing convenience — null when absent, refused, or the other kind. */
declare function readCombEntityMeta(jsonSchema: unknown, options?: ReadOptions): CombEntityMeta | null;
/** Narrowing convenience — null when absent, refused, or the other kind. */
declare function readCombQueryMeta(jsonSchema: unknown, options?: ReadOptions): CombQueryMeta | null;
export { isRecord, type ReadOptions, readCombEntityMeta, readCombMeta, readCombQueryMeta };
