// sqlite/constraints.d.ts
/**
 * Type-safe column constraint definitions
 * These exist only in the type system and are completely erased at runtime
 */
type BaseConstraints = {
    /** Field is filled by comb or the database: generated create and update DTOs leave it out */
    autogenerate?: boolean | undefined;
    /**
     * A foreign key that may move after creation (a person to another team): generated update DTOs
     * take it. Every other reference is set once, at creation. Refused on a column that references
     * nothing, and together with `nomutate`.
     */
    mutable?: boolean | undefined;
    /** Field cannot be mutated after creation */
    nomutate?: boolean | undefined;
    /**
     * Field is never read back: generated read DTOs and the meta stamp leave it out, and the manifest
     * lists it flagged `private: true`, on both dialects
     */
    private?: boolean | undefined;
    /**
     * the name this foreign key's relation and its way back take, where the column's role would take a
     * column's own name (`currency` toward `currency.code`, a `parent` self-reference): an identifier,
     * refused on a column that references nothing
     */
    relation?: string | undefined;
    /**
     * This column scopes rows to a tenant.
     *
     * Declared, never inferred: comb can see that a column is a foreign key but
     * not that the table it points at is the tenant, and a wrong tenant boundary
     * is worse than a missing one. Saying so here makes it a hand-reviewed fact.
     * At most one column per table may carry it.
     */
    tenant?: boolean | undefined;
};
/**
 * Rewrites a written text takes in the generated DTOs before its checks, so its bounds and pattern
 * hold the rewritten value: `trim` first, then `lowercase`, or `uppercase` when `lowercase` is unset.
 * An email is always trimmed and lowercased, a url trimmed and stripped of tabs and line breaks, and
 * codegen refuses a rewrite either would drop, or any on a password, which is kept as written
 */
type StringTransformConstraints = {
    /** Apply lowercase transformation */
    lowercase?: boolean | undefined;
    /** Trim whitespace */
    trim?: boolean | undefined;
    /** Apply uppercase transformation */
    uppercase?: boolean | undefined;
};
/** Bounds a written value is held to: a text's length, or a number's or timestamp's value */
type NumericRangeConstraints = {
    /** Maximum value */
    max?: number | undefined;
    /** Minimum value */
    min?: number | undefined;
};
/**
 * What a written text must hold, checked by the generated DTOs: one kind at most, of `email`, `url`,
 * `password`, `imageUri` and `name`, which codegen refuses a second of, and a `pattern` beside it
 */
type StringValidationConstraints = {
    /** Email validation */
    email?: boolean | undefined;
    /** Image URI validation (accepts https URLs and data:image/ URIs) */
    imageUri?: boolean | undefined;
    /** Human/entity name validation (letters, marks, numbers, spaces, punctuation) */
    name?: boolean | undefined;
    /** A stored password hash: kept as written, held to the declared bounds and pattern alone; validate the plaintext with `passwordText()` */
    password?: boolean | undefined;
    /** Regex pattern validation */
    pattern?: string | undefined;
    /** URL validation */
    url?: boolean | undefined;
};
/**
 * Text column constraints
 * Closed record - only these properties allowed
 */
export type TextConstraints = BaseConstraints & StringTransformConstraints & NumericRangeConstraints & StringValidationConstraints;
/**
 * Integer/numeric column constraints
 * Closed record - only these properties allowed
 */
export type IntegerConstraints = BaseConstraints & NumericRangeConstraints;
/**
 * JSON column constraints
 * Closed record - only these properties allowed
 */
export type JsonConstraints = BaseConstraints & {
    /** Maximum JSON stringified byte length */
    maxBytes?: number | undefined;
};
/**
 * Timestamp column constraints
 * Closed record - only these properties allowed
 */
export type TimestampConstraints = BaseConstraints & NumericRangeConstraints;
/**
 * State machine declared alongside an enum column's values.
 *
 * Type-level only, like every other constraint here — the analyzer reads it
 * out of the source. Hold the same object as a const and pass it to both
 * `c.enum` and `assertTransition`. See docs/state-machines.md.
 */
export type EnumStates<TState extends string = string> = {
    /** The value a row starts at */
    initial?: TState | undefined;
    /** Values from which no transition is legal */
    terminal?: readonly TState[] | undefined;
    /** Legal moves, `from` → `to[]`. Omitted states are unconstrained. */
    transitions?: {
        readonly [K in TState]?: readonly TState[] | undefined;
    } | undefined;
};
/** What `c.enum` takes beside its values: its state machine, and the `nomutate`, `private` and `tenant` flags every helper takes */
export type EnumOptions<TState extends string = string> = EnumStates<TState> & Pick<BaseConstraints, "nomutate" | "private" | "tenant">;
export {};
