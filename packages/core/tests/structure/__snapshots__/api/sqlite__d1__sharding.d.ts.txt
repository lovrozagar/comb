// query/merged.d.ts
import { type ListPagination, type ListRow, type PaginationResponse } from "./pagination.ts";
/**
 * The page of a list merged from `rowsPerShard`, each shard's rows read with `built`,
 * and its pagination, as `paginate` gives one database's: ordered as the list orders, windowed by
 * page or cursor, counted across shards, its next cursor sealed under `options`. Ids must be
 * unique across the shards, as a cursor names its row by id
 */
declare function paginateMerged<T extends object>(rowsPerShard: ReadonlyArray<readonly T[]>, built: {
    /** the pagination buildListQuery returned for the list */
    pagination: ListPagination & {
        /** set when buildListQuery was given the `merged` option, so each shard reads the rows a merged page needs */
        merged: true;
    };
}): readonly [Array<ListRow<T>>, PaginationResponse];
export { paginateMerged };

// query/sqlite/build-list-query-types.d.ts
/** The options the SQLite list builders take, and the query they build. Declarations only. */
import type { SQL, SQLWrapper } from "drizzle-orm";
import type { SQLiteColumn, SQLiteTable } from "drizzle-orm/sqlite-core";
import type { ListPagination } from "../pagination.ts";
import type { ComputedFilterResolver, ComputedSortResolver, ListQueryInput } from "../types.ts";
import type { LikeSearchResolver } from "./search.ts";
import type { RelationConfig } from "./types.ts";
/** What `buildListQuery` builds a table's list from */
type BuildListQueryOpts<TTable extends SQLiteTable> = {
    /** a WHERE clause ANDed with the filter, search and cursor, for joins, access rules or soft deletes; a function is called with drizzle's `sql` */
    baseWhere?: SQL | ((sql: (strings: TemplateStringsArray, ...values: unknown[]) => SQL) => SQL) | undefined;
    /** the resolver of each `@`-prefixed filter field, by its name with the `@`; a filter on one without a resolver is a 500 */
    computedFilters?: Record<string, ComputedFilterResolver> | undefined;
    /** the resolver of each `@`-prefixed sort field, by its name with the `@`; a sort on one without a resolver is a 500 */
    computedSorts?: Record<string, ComputedSortResolver> | undefined;
    /** the list's scope, its route and tenant: a cursor minted in another opens in none here */
    cursorScope: string;
    /** the app's secret a cursor is sealed under, of at least 32 UTF-8 bytes */
    cursorSecret: string;
    /** "exact" when the driver reads ±Infinity back as Infinity, so a null cursor is a real NULL, as on `CursorOptions` */
    nonFinite?: "exact" | undefined;
    /**
     * Present for a list whose rows sit on several D1 shards: every shard reads it with this one
     * query, and `paginateMerged` merges what they read. A page by number reads every row to its end
     * from each shard. It sorts by columns only, and the table needs an `id` column to break ties across shards
     */
    merged?: {
        /** the most rows a shard reads for one page by number, a whole number from 1; default 1001, and a deeper page is a 400 `page_too_deep` */
        maxRowsPerShard?: number | undefined;
    } | undefined;
    /** the list schema's output; a filter or order it did not check is a 500 */
    parsed: ListQueryInput;
    /** the many-to-many relation each `relation.column` filter field reaches, by the relation's name */
    relations?: Record<string, RelationConfig> | undefined;
    /** the search the `q` param runs, such as `likeSearch(column)`; a `q` without one is a 500 */
    search?: LikeSearchResolver | undefined;
    /** the table the list reads; each sort and filter field is one of its columns by property name, and its `id` column breaks ties */
    table: TTable;
};
/** What `buildDerivedListQuery` builds a list from whose rows are no one table's: a join, a union or a view */
type BuildDerivedListQueryOpts<TSortField extends string = string> = {
    /** a WHERE clause ANDed with the filter, search and cursor; a function is called with drizzle's `sql` */
    baseWhere?: SQL | ((sql: (strings: TemplateStringsArray, ...values: unknown[]) => SQL) => SQL) | undefined;
    /** the list's scope, its route and tenant: a cursor minted in another opens in none here */
    cursorScope: string;
    /** the app's secret a cursor is sealed under, of at least 32 UTF-8 bytes */
    cursorSecret: string;
    /**
     * the database the list reads, needed only where its id and keys hold no column: a column says it
     * wherever it sits, and a list of SQL expressions alone (a union, say) cannot
     */
    dialect?: "postgres" | "sqlite" | undefined;
    /** the request's filter as SQL, lowered by your code; null or absent filters nothing */
    filterWhere?: SQL | null | undefined;
    /**
     * Present when the rows come from several shards, which `paginateMerged` merges, as on
     * `buildListQuery`: each shard reads its window, an offset page its count. A merged derived list
     * carries no sort key by reference, so it takes `idColumn`, and its rows carry each sort field under its name
     */
    merged?: {
        /** the most rows a shard reads for one page by number, a whole number from 1; default 1001, and a deeper page is a 400 `page_too_deep` */
        maxRowsPerShard?: number | undefined;
    } | undefined;
    /** the list schema's output; its sort fields are the keys of `sortColumns` */
    parsed: ListQueryInput<TSortField>;
    /** the request's search as SQL, built by your code; null or absent searches nothing */
    searchWhere?: SQL | null | undefined;
    /** each sort field's expression; one that yields epoch milliseconds is `{ sql, type: "date" }`, and its cursor value a Date. A sort on a field with none is a 500 */
    sortColumns: Record<TSortField, SQLWrapper | {
        /** the expression, yielding epoch milliseconds */
        sql: SQLWrapper;
        /** "date": the cursor carries the key as a Date */
        type: "date";
    }>;
} & ({
    /**
     * the one table the rows are, by its id column: a sort key too long for a cursor is
     * carried by reference and read back from it, as on a table's own list
     */
    anchor: {
        /** the table's id column, which the order breaks ties on */
        idColumn: SQLiteColumn;
        /** the table each row is one of */
        table: SQLiteTable;
    };
    /** absent: `anchor` names the id */
    idColumn?: never | undefined;
} | {
    /** absent: the rows are no one table's */
    anchor?: never | undefined;
    /** the id the order breaks ties on; with no table to read a key back from, a key too long for a cursor is a 500 */
    idColumn: SQLWrapper;
});
/**
 * The columns a built list's rows must carry beside the app's own, for `paginate` to read: select
 * them with the rest (`db.select({ ...cols, ...built.extras })`), and paginate strips them
 */
type ListExtras = Partial<{
    /** 1 when the cursor's anchor row is gone where no bound is safe, which paginate answers with a 400 invalid_cursor */
    _combAnchorGone: SQL.Aliased<number>;
    /** the value the list breaks ties on, which the next cursor resumes from; none when no cursor can */
    _combId: SQL.Aliased<unknown>;
    /** the class of each sort key and the tiebreak, comma-joined, where the list binds them as read or one is an undeclared SQLite expression */
    _combKeyTypes: SQL.Aliased<string>;
    /** each sort key's stored time where the list binds it as read, comma-joined, empty for any other key: a Postgres time key's text, a SQLite REAL seconds */
    _combKeyTexts: SQL.Aliased<string>;
    /** each SQLite timestamp key's TEXT where the list reads it back and the row holds TEXT, a JSON array with null for any other key */
    _combTextKeys: SQL.Aliased<string>;
    /** an offset page's count of every row the filter matches; a cursor page selects none */
    _total: SQL.Aliased<number>;
}>;
/** A built list query: apply it with `applyListQuery`, then page its rows with `paginate` */
type ListQueryResult = {
    /** select these beside the row's own columns */
    extras: ListExtras;
    /** the rows to read: one more than the page, so `paginate` can tell a next page exists */
    limit: number;
    /** How `paginate` pages the rows this query reads */
    pagination: ListPagination & {
        /** false: `paginate` pages it */
        merged: false;
    };
    /** the page as the request asked for it */
    meta: {
        /** the page size */
        limit: number;
        /** the page number; 1 on a cursor page */
        page: number;
        /** how the page is read: past a cursor, or by number */
        type: "cursor" | "offset";
    };
    /** the rows to skip: 0 on a cursor page, which the WHERE clause seeks past */
    offset: number;
    /** the ORDER BY: each sort key with its null placement stated, then the tiebreak */
    orderBy: SQL[];
    /** the trimmed search text; null when the request searches nothing */
    search: string | null;
    /** the base, filter, search and cursor clauses ANDed; undefined when there are none */
    where: SQL | undefined;
};
/** A merged list's query, which every shard reads and `paginateMerged` alone pages; each shard reads from its first row, so `offset` is 0 */
type MergedListQueryResult = Omit<ListQueryResult, "pagination"> & {
    /** How `paginateMerged` pages the rows every shard reads */
    pagination: ListPagination & {
        /** true: `paginateMerged` pages it, and `paginate` refuses it */
        merged: true;
    };
};
export type { BuildDerivedListQueryOpts, BuildListQueryOpts, ListExtras, ListQueryResult, MergedListQueryResult };

// sqlite/d1/sharding/router.d.ts
import type { D1DatabaseLike } from "../d1-types.ts";
import type { ShardBindingOf, ShardDbHandle, ShardEnv, ShardMapping, ShardMove, ShardPlacement, ShardResolverDeps, ShardRouterConfig, ShardVerifyDeps, WaitUntilCtx } from "./types.ts";
/**
 * Shard router — resolves org -> D1 database handle.
 *
 * Instantiate once per isolate (module-level) so the LRU cache persists.
 * 3-tier lookup: LRU -> KV -> Core D1; a new org goes to the shard `pick` names.
 */
declare class ShardRouter {
    private readonly cache;
    private readonly log;
    private readonly pick;
    private readonly totalShards;
    /**
     * Check `config` and fill its defaults: a RangeError when `totalShards` or `maxOrgsPerShard` is
     * no positive integer, or `kvTtlSeconds` is no integer of at least 60. Without `pick`, a new org
     * goes to the shard with the most room under `maxOrgsPerShard`; without `log`, only failures the
     * router absorbs reach console.warn.
     */
    constructor(config: ShardRouterConfig);
    /**
     * Resolve existing mapping, or allocate a new shard if org is unknown.
     * This is the primary entry point for middleware.
     */
    resolveOrAllocate(orgId: string, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /**
     * Resolve shard mapping via LRU (SWR) -> KV -> Core D1.
     * Returns undefined if org has no mapping.
     * The core read goes to the nearest replica, and to the primary when the replica has no
     * mapping, which a fresh placement may not have reached yet; a miss there is remembered for
     * missTtlMs, so a run of requests for an unknown id cannot load the primary.
     */
    resolve(orgId: string, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping | undefined>;
    /**
     * Resolve as `resolve` does, then check with `deps.present` that the org's rows are on the shard
     * named: a mapping cached here can predate a `move` in another isolate, since invalidate
     * reaches only this isolate's LRU and KV. When they are not, the primary is read past every
     * cache. When it names the shard just probed, the org's rows may not be there yet: that mapping
     * is returned, and nothing is written or logged. When it names another, or none, the org is
     * invalidated, the primary read again under the tombstone, and that mapping, or undefined when
     * the org is gone, returned without a second probe; it is cached only while no invalidate here,
     * a move's among them, came after the tombstone.
     */
    resolveVerified(orgId: string, deps: ShardVerifyDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping | undefined>;
    /**
     * Allocate a shard for an org: the one `pick` names, told each shard's live count. A
     * soft-deleted org is revived on the shard it had, where its data is, without asking pick. The
     * count is read before the insert, so two isolates allocating at once can both take a shard's
     * last place: maxOrgsPerShard is a soft cap. Writes go to the primary.
     */
    allocate(orgId: string, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /**
     * Place an org on a shard the app chose. The first writer wins: an org that has a row keeps
     * its shard, a soft-deleted one revived there, and the mapping that stands is returned, so two
     * isolates placing one new org at once agree. To put a placed org elsewhere, `move` it.
     */
    place(placement: ShardPlacement, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /**
     * Move an org to `shardId` and invalidate it here. Moving its rows is the app's; its writes on
     * the core database for the move (`with`) batch with the mapping's, landing together or not at
     * all. Another isolate routes to the old shard until its caches expire or it verifies. A
     * RangeError when `orgId` is over 64 UTF-16 code units or `shardId` is not a shard.
     */
    move(placement: ShardMove, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /** Soft-delete an org's mapping and invalidate it here; a later allocate or place revives it on its shard */
    unplace(orgId: string, deps: ShardResolverDeps): Promise<void>;
    /**
     * Resolve a shard mapping to a concrete D1 database handle: the env's `SHARD_DB_{target}`
     * binding, in the type the env holds its shard bindings as. An env typed by a string index
     * signature gives the D1 binding it holds, else D1DatabaseLike; one whose known keys hold no
     * `SHARD_DB_*` is refused. Throws when the env holds no binding under that key at run time.
     */
    getHandle<E extends object>(mapping: ShardMapping, env: E & ShardEnv<E>): ShardDbHandle<ShardBindingOf<E>>;
    /**
     * Resolve a shard mapping for an env TypeScript cannot read the keys of, such as a helper's type
     * parameter bounded by an env interface whose `SHARD_DB_0` is required: the binding type that key
     * holds, since shard 0 always exists, with every other `SHARD_DB_*` key it names a D1 binding.
     * An env bounded by no shard key is refused. So is a bound whose `SHARD_DB_0` is optional, as
     * `wrangler types` writes it when no shard is bound at the top level: the helper widens its
     * parameter to that env first, `const env: Env = given`, and the first overload then gives
     * shard 0's binding. Throws when the env holds no binding under the mapping's key at run time.
     */
    getHandle<E extends {
        /** shard 0's D1 binding, which every sharded env holds */
        readonly SHARD_DB_0: D1DatabaseLike;
    }>(mapping: ShardMapping, env: E & {
        readonly [K in keyof E & `SHARD_DB_${string}`]?: D1DatabaseLike | undefined;
    }): ShardDbHandle<E["SHARD_DB_0"]>;
    /**
     * Invalidate both cache tiers. Use after whale graduation or reassignment. It reaches this
     * isolate's LRU and the KV; another isolate's LRU keeps the mapping until it expires.
     */
    invalidate(orgId: string, kv: ShardResolverDeps["kv"]): Promise<void>;
    /** Places an org as `place` does, unchecked: an org being revived keeps the shard it had */
    private placeFirst;
    /**
     * Caches the row a write left standing, this isolate's miss for the org forgotten, and returns
     * its mapping; logs the placement only when the write placed it, so the events count rows (U40).
     * Cached only while no invalidate here came after `generation`, which was taken before the
     * write read the row: one that did, a move's among them, may have remapped it (Q6 of r18)
     */
    private settle;
}
export { ShardRouter };

// sqlite/d1/sharding/shard-map.d.ts
import { type HasDefault, type IsPrimaryKey, type NotNull } from "drizzle-orm";
import { sqliteTable as createTable } from "drizzle-orm/sqlite-core";
import { c, type SqliteColumns } from "../../columns.ts";
/** The longest org id `shard_map.org_id` holds; a longer one is no org's */
export declare const ORG_ID_MAX_LENGTH = 64;
/**
 * Refuse an org id longer than the column holds, measured in UTF-16 code units as `resolve`
 * measures it, so no row is written that `resolve` would answer as a miss (U41). The id is the
 * caller's input: a route that takes it from a path validates it first.
 */
export declare function assertOrgId(orgId: string): void;
/** Each column's builder, from the `c` factory that makes it */
export type ShardMapColumns = {
    /** when the org was first placed, in epoch milliseconds, filled on insert */
    created_at: ReturnType<SqliteColumns["createdAt"]>;
    /** when the org was unplaced, in epoch milliseconds; null while its row is live */
    deleted_at: ReturnType<SqliteColumns["deletedAt"]>;
    /** the kind of database the shard is, `"d1"` by default */
    driver: HasDefault<NotNull<ReturnType<typeof c.enum<readonly ["d1"]>>>>;
    /** the org's id, the primary key, at most 64 UTF-16 code units */
    org_id: IsPrimaryKey<ReturnType<SqliteColumns["text"]>>;
    /** the shard the org's rows are on */
    shard_id: NotNull<ReturnType<SqliteColumns["integer"]>>;
    /** the suffix of the `SHARD_DB_<target>` binding the router opens, at most 128 characters; comb writes String(shard_id), so a hand-written move of shard_id writes it too */
    target: NotNull<ReturnType<SqliteColumns["text"]>>;
    /** when the row last changed, in epoch milliseconds */
    updated_at: ReturnType<SqliteColumns["updatedAt"]>;
};
/** The drizzle table type of `shard_map`, for a query or a migration schema that names it */
export type ShardMapTable = ReturnType<typeof createTable<"shard_map", ShardMapColumns>>;
/**
 * The core database's `shard_map` table, one row per org placed: which shard its rows are on.
 * Include it in the core database's schema so its migrations create it, with an index on
 * `shard_id` and one on `driver` over live rows. The router writes it; read it for an operator's
 * view, and change a row through `router.move` so its caches are invalidated.
 */
export declare const shardMap: ShardMapTable;

// sqlite/d1/sharding/types.d.ts
import type { D1DatabaseLike, D1StatementLike } from "../d1-types.ts";
/** The kind of database a shard is; Cloudflare D1 is the only one */
type ShardDriver = "d1";
/** Where an org's rows live: its `shard_map` row, as the router reads and caches it */
type ShardMapping = {
    /** when the org was first placed, in epoch milliseconds */
    createdAt: number;
    /** when the org was unplaced, in epoch milliseconds; null while its mapping is live */
    deletedAt: number | null;
    /** the kind of database the shard is */
    driver: ShardDriver;
    /** the org's id, at most 64 UTF-16 code units */
    orgId: string;
    /** the shard the org's rows are on, from 0 up to `totalShards - 1` */
    shardId: number;
    /** the suffix of the env binding getHandle opens, `SHARD_DB_<target>`; comb writes it as String(shardId) */
    target: string;
    /** when the mapping last changed, in epoch milliseconds */
    updatedAt: number;
};
/** A shard's binding, in the type getHandle reads from the env it is passed */
type ShardDbHandle<D extends D1DatabaseLike = D1DatabaseLike> = {
    /** the kind of database `handle` is */
    driver: "d1";
    /** the shard's D1 binding, taken from the env as it is */
    handle: D;
};
/**
 * The binding getHandle hands back for an env of type `E`: the type its `SHARD_DB_*` keys hold,
 * as `wrangler types` writes them, a key bound in only some environments among them; an env with
 * a string index signature gives the D1 binding its index holds, and the structural
 * D1DatabaseLike when it holds none
 */
type ShardBindingOf<E> = string extends keyof E ? Extract<NonNullable<E[string & keyof E]>, D1DatabaseLike> extends infer D extends D1DatabaseLike ? [D] extends [never] ? D1DatabaseLike : D : never : Extract<NonNullable<E[Extract<keyof E, `SHARD_DB_${string}`>]>, D1DatabaseLike>;
/**
 * What getHandle asks of an env of type `E` beyond `E` itself: each `SHARD_DB_*` key holds a D1
 * binding, and an env whose known keys hold no such key is refused, naming the key it lacks. A key
 * may be optional, as `wrangler types` writes one bound in only some environments: getHandle
 * throws on an absent one at run time. An env with a string index signature is taken when its
 * index may hold a D1 binding, and refused when it provably holds none
 */
type ShardEnv<E> = string extends keyof E ? unknown extends E[string & keyof E] ? unknown : [Extract<NonNullable<E[string & keyof E]>, D1DatabaseLike>] extends [never] ? {
    /** the key the env lacks: a D1 binding for each shard, `SHARD_DB_0` upward; it names the fault in the type error and no env holds it */
    "SHARD_DB_<target>": D1DatabaseLike;
} : unknown : [Extract<keyof E, `SHARD_DB_${string}`>] extends [never] ? {
    /** the key the env lacks: a D1 binding for each shard, `SHARD_DB_0` upward; it names the fault in the type error and no env holds it */
    "SHARD_DB_<target>": D1DatabaseLike;
} : {
    [K in Extract<keyof E, `SHARD_DB_${string}`>]?: D1DatabaseLike | undefined;
};
/**
 * The KV the router caches mappings in: the three calls it makes. Structural, so a Worker's global
 * `KVNamespace` and the one `@cloudflare/workers-types` exports both fit; the two are distinct types.
 */
type ShardKv = {
    /** Remove `key`; the router drops an org's cached mapping this way when it invalidates the org */
    delete(key: string): Promise<void>;
    /** The JSON value stored under `key`, parsed, or null when none is */
    get(key: string, type: "json"): Promise<unknown>;
    /** Store `value`, a JSON text, under `key` */
    put(key: string, value: string, options?: {
        /** seconds until the key expires: `kvTtlSeconds` for a mapping, never under KV's 60-second floor */
        expirationTtl?: number | undefined;
    }): Promise<void>;
};
/** The bindings a lookup or a write reads, passed per request since a Worker gets them from its env */
type ShardResolverDeps = {
    /** the core D1 database holding `shard_map`: reads go to the nearest replica, writes to the primary */
    coreDb: D1DatabaseLike;
    /** the KV namespace mappings are cached in, under `shard:<orgId>`, for `kvTtlSeconds` */
    kv: ShardKv;
};
/** What `pick` is told about a new org: how many live orgs each D1 shard holds, absent when none */
type ShardPickInput = {
    /** each shard's live orgs, by shard id; a shard holding none has no entry */
    counts: ReadonlyMap<number, number>;
    /** the org being placed */
    orgId: string;
    /** the router's shard count: a pick answers a shard from 0 up to `totalShards - 1`, or a RangeError is thrown */
    totalShards: number;
};
/** The shard a new org goes to, or undefined when none should take it (`shards_full`) */
type ShardPick = (input: ShardPickInput) => number | undefined | Promise<number | undefined>;
/** Structural type so both `@cloudflare/workers-types` and `cloudflare:workers` satisfy it */
type WaitUntilCtx = {
    /** Keep the Worker alive until `promise` settles; the router hands it KV writes and background revalidation */
    waitUntil: (promise: Promise<unknown>) => void;
};
/** A logger for the router's events: the event's name, such as `shard-router.miss`, and its data */
type LogFn = (event: string, data: Record<string, unknown>) => void;
/** An org and the shard to put it on */
type ShardPlacement = {
    /** the org, at most 64 UTF-16 code units, or a RangeError is thrown */
    orgId: string;
    /** the shard, from 0 up to `totalShards - 1`, or a RangeError is thrown */
    shardId: number;
};
/** A move: the placement, and the app's own statements on the core database to batch with it */
type ShardMove = ShardPlacement & {
    /** statements on the core database run in one batch after the mapping's write, landing with it or not at all */
    with?: readonly D1StatementLike[] | undefined;
};
/** The deps of `resolveVerified`: those of `resolve`, and how the app checks an org's rows are on the shard a mapping names */
type ShardVerifyDeps = ShardResolverDeps & {
    /** Whether the org's rows are on the shard `mapping` names, as the app checks it, such as by reading one row there */
    present: (mapping: ShardMapping) => Promise<boolean>;
};
/** A ShardRouter's settings; the constructor throws a RangeError on one it cannot run on */
interface ShardRouterConfig {
    /** Seconds a mapping stays in KV, an integer of at least 60, KV's floor; 60 when omitted */
    kvTtlSeconds?: number | undefined;
    /** Where the router's events go, such as `shard-router.hit`, `shard-router.allocate` and `shard-router.kv.error`; when omitted, each `*.error` event, a failure the router absorbed, goes to console.warn and the rest are dropped */
    log?: LogFn | undefined;
    /** Milliseconds a mapping stays in this isolate's memory at all; 900000 (15 minutes) when omitted */
    lruHardTtlMs?: number | undefined;
    /** The most orgs this isolate keeps in memory, the least recently used dropped first; 1000 when omitted */
    lruMaxSize?: number | undefined;
    /** Milliseconds a mapping in memory is served fresh; after it, served stale while one background read refreshes it; 300000 (5 minutes) when omitted */
    lruTtlMs?: number | undefined;
    /** The live orgs the default `pick` puts on one shard, a positive integer; 500 when omitted. A soft cap: two allocations at once can both take a shard's last place */
    maxOrgsPerShard?: number | undefined;
    /** How long this isolate remembers that the primary has no mapping for an org, in milliseconds; 5000 when omitted */
    missTtlMs?: number | undefined;
    /** The shard a new org goes to; the default takes the one with the most headroom under maxOrgsPerShard */
    pick?: ShardPick | undefined;
    /** How many shards there are, a positive integer: shards 0 up to `totalShards - 1`, each bound as `SHARD_DB_<n>` */
    totalShards: number;
}

/**

 * (`buildListQuery`, its `merged` list included, `buildDerivedListQuery`, `queryHelpers.listOptions`)
 * hands over a `pagination` descriptor: the page it read, whether a cursor can resume it, how a
 * long key is carried, and how many rows it asked for. `paginate` reads it, so nothing about the
 * list is told twice. The cursor secret is never a value of it: the descriptor seals and opens
 * cursors through closures over the options its builder was given, so logging or serializing a

/**

 * `paginate` whole. It seals and opens cursors without holding the secret as a value, so logging or
 * serializing it shows none of it
 */
type ListPagination = {

    keyset: KeysetKind;
    /** the page's row count, the request's limit */
    limit: number;
    /** true on a list built with `merged`, which `paginateMerged` pages and `paginate` refuses */
    merged: boolean;

    nonFinite: "exact" | undefined;

    page: number | undefined;

    parsedFields: ParsedFields | null;
    /** the order the list was read in, which the next cursor is minted under */
    parsedSort: readonly SortField[];
    /** the most rows the built query reads: one past the page, so paginate can tell there is more, or on a merged page by number each shard's rows to the page's end and one more */
    window: number;
    /** the cursor for the page after `row`, sealed under the builder's options */
    mint(row: object): string;
    /** `cursor` opened under the builder's options, for this list's sort; a 400 when it is no cursor of it */
    read(cursor: string): DecodedCursor;
};
/** What a builder knows of its list, beside the request it parsed */

    byReference?: ReadonlySet<string> | undefined;

    window: number;

    merged: TMerged;
};

/**
 * A filter condition's operator, as written between the field and the value: `ne` and `neq` both
 * mean not equal, `in` and `nin` take a parenthesized list, `like` is case-sensitive and `ilike`
 * ignores ASCII case (both with `*` as the wildcard), and `is` takes `null`, `notnull`, `true` or `false`

type FilterOperator = "eq" | "ne" | "neq" | "gt" | "gte" | "lt" | "lte" | "in" | "nin" | "like" | "ilike" | "is"

    field: string;

    /** the value as parsed: a string, null for `is.null`, or an array of strings for a list; `validateFilter` types it as its field */
    value: unknown;
};

type FilterGroup = {
    /** the group's own terms, in the order written */
    conditions: FilterCondition[];
    /** how the group's conditions and subgroups combine; the top level is `and` */
    logic: "and" | "or";
    /** each nested `and(...)` or `or(...)`, in the order written */
    subgroups: FilterGroup[];
};
/** A parsed filter, as `parseFilter` and `validateFilter` return it */

    /** the top-level group, joined by its own `logic`; `parseFilter` and `validateFilter` make it an `and` */
    root: FilterGroup;
};

/** Where a sort key puts nulls; left out, they sort last ascending and first descending, as on Postgres */
type SortNulls = "first" | "last";

type SortField<T extends string = string> = {
    /** the direction as written; `asc` when the key names none */
    direction: SortDirection;
    /** the field sorted on: a column, or a `@`-prefixed computed field */
    field: T;
    /** null placement as written; left out, the direction decides it (see SortNulls) */
    nulls?: SortNulls | undefined;
};

    /** the most rows the page holds, a whole number from 1 */
    limit: number;
    /** the 1-based page to read by number; null or absent is page 1. A cursor, when given, wins */
    page?: number | null | undefined;
    /** the deepest page this query may read; absent, the default maxPage. A deeper page is a 400 */
    pageBound?: number | undefined;
    /** the largest limit this query may take; absent, the default maxLimit. A larger limit is a 400 */
    limitBound?: number | undefined;
    /** the column the list schema breaks ties on; a derived list's idColumn must name it */
    tiebreak?: string | undefined;
    /** the parsed `select`; null or absent selects every field */
    parsedFields?: ParsedFields | null | undefined;
    /** the sort keys in order; empty sorts by the tiebreak column alone */
    parsedSort: readonly SortField<TSortField>[];
    /** the search text, trimmed; null, absent or blank searches nothing */
    q?: string | null | undefined;
    /**
     * The list schema sets it. A builder throws a 500 on a filter or an order without it, as neither
     * was checked against the fields the endpoint declares; set it only on ones your code checked
     */
    validated?: true | undefined;
};

type ComputedFilterResolver = (operator: FilterOperator, value: unknown) => SQL | null;

type ComputedSortResolver = (direction: SortDirection) => SQL;

// sqlite/constraints.d.ts
/**
 * Type-safe column constraint definitions
 * These exist only in the type system and are completely erased at runtime
 */
type BaseConstraints = {
    /** Field is filled by comb or the database: generated create and update DTOs leave it out */
    autogenerate?: boolean | undefined;
    /**
     * A foreign key that may move after creation (a person to another team): generated update DTOs
     * take it. Every other reference is set once, at creation. Refused on a column that references
     * nothing, and together with `nomutate`.
     */
    mutable?: boolean | undefined;
    /** Field cannot be mutated after creation */
    nomutate?: boolean | undefined;
    /**
     * Field is never read back: generated read DTOs and the meta stamp leave it out, and the manifest
     * lists it flagged `private: true`, on both dialects
     */
    private?: boolean | undefined;
    /**
     * the name this foreign key's relation and its way back take, where the column's role would take a
     * column's own name (`currency` toward `currency.code`, a `parent` self-reference): an identifier,
     * refused on a column that references nothing
     */
    relation?: string | undefined;
    /**
     * This column scopes rows to a tenant.
     *
     * Declared, never inferred: comb can see that a column is a foreign key but
     * not that the table it points at is the tenant, and a wrong tenant boundary
     * is worse than a missing one. Saying so here makes it a hand-reviewed fact.
     * At most one column per table may carry it.
     */
    tenant?: boolean | undefined;
};
/**
 * Rewrites a written text takes in the generated DTOs before its checks, so its bounds and pattern
 * hold the rewritten value: `trim` first, then `lowercase`, or `uppercase` when `lowercase` is unset.
 * An email is always trimmed and lowercased, a url trimmed and stripped of tabs and line breaks, and
 * codegen refuses a rewrite either would drop, or any on a password, which is kept as written
 */
type StringTransformConstraints = {
    /** Apply lowercase transformation */
    lowercase?: boolean | undefined;
    /** Trim whitespace */
    trim?: boolean | undefined;
    /** Apply uppercase transformation */
    uppercase?: boolean | undefined;
};
/** Bounds a written value is held to: a text's length, or a number's or timestamp's value */
type NumericRangeConstraints = {
    /** Maximum value */
    max?: number | undefined;
    /** Minimum value */
    min?: number | undefined;
};
/**
 * What a written text must hold, checked by the generated DTOs: one kind at most, of `email`, `url`,
 * `password`, `imageUri` and `name`, which codegen refuses a second of, and a `pattern` beside it
 */
type StringValidationConstraints = {
    /** Email validation */
    email?: boolean | undefined;
    /** Image URI validation (accepts https URLs and data:image/ URIs) */
    imageUri?: boolean | undefined;
    /** Human/entity name validation (letters, marks, numbers, spaces, punctuation) */
    name?: boolean | undefined;
    /** A stored password hash: kept as written, held to the declared bounds and pattern alone; validate the plaintext with `passwordText()` */
    password?: boolean | undefined;
    /** Regex pattern validation */
    pattern?: string | undefined;
    /** URL validation */
    url?: boolean | undefined;
};
/**
 * Text column constraints
 * Closed record - only these properties allowed
 */
export type TextConstraints = BaseConstraints & StringTransformConstraints & NumericRangeConstraints & StringValidationConstraints;
/**
 * Integer/numeric column constraints
 * Closed record - only these properties allowed
 */
export type IntegerConstraints = BaseConstraints & NumericRangeConstraints;
/**
 * JSON column constraints
 * Closed record - only these properties allowed
 */
export type JsonConstraints = BaseConstraints & {
    /** Maximum JSON stringified byte length */
    maxBytes?: number | undefined;
};
/**
 * Timestamp column constraints
 * Closed record - only these properties allowed
 */
export type TimestampConstraints = BaseConstraints & NumericRangeConstraints;
/**
 * State machine declared alongside an enum column's values.
 *
 * Type-level only, like every other constraint here — the analyzer reads it
 * out of the source. Hold the same object as a const and pass it to both
 * `c.enum` and `assertTransition`. See docs/state-machines.md.
 */
export type EnumStates<TState extends string = string> = {
    /** The value a row starts at */
    initial?: TState | undefined;
    /** Values from which no transition is legal */
    terminal?: readonly TState[] | undefined;
    /** Legal moves, `from` → `to[]`. Omitted states are unconstrained. */
    transitions?: {
        readonly [K in TState]?: readonly TState[] | undefined;
    } | undefined;
};
/** What `c.enum` takes beside its values: its state machine, and the `nomutate`, `private` and `tenant` flags every helper takes */
export type EnumOptions<TState extends string = string> = EnumStates<TState> & Pick<BaseConstraints, "nomutate" | "private" | "tenant">;

    parsedFields: ParsedFields | null;
    /** the `order` keys in order; empty without an `order` */
    parsedSort: Array<SortField<TSortField>>;
    /** the `q` param as sent */

    /** each sorted field's direction */
    sortOrderBy: Partial<Record<TSortField, SortDirection>>;
    /**
     * The schema's own maxPage, which holds the page wherever this object is read, a spread of it
     * included. A query built by hand leaves it out for the default maxPage, or states the bound it means
     */
    pageBound?: number | undefined;
    /** The schema's own maxLimit, which holds the limit as `pageBound` holds the page */
    limitBound?: number | undefined;
    /** The column the schema says the list breaks ties on, `id` by default, which a derived list's idColumn names */
    tiebreak?: string | undefined;
    /** Always true: the filter and the order were checked against the endpoint's fields */
    validated: true;
};
/** A retrieve query schema's parsed output */
type RetrieveQueryOutput = {
    /** the `lang` param as sent */
    lang: string | undefined;
    /** the checked `select`, `*` expanded to the declared fields; null without one, which selects every field */
    parsedFields: ParsedFields | null;

// query/filter.d.ts
/**
 * PostgREST-style filters and sort orders: parse and validate. DB-agnostic: no SQL, only
 * the AST and its checks. The grammar lives in filter-parse.ts, the checks in filter-validate.ts.
 */
import { FILTER_OPERATORS, parseFilter } from "./filter-parse.ts";
import { parseFilterNumber } from "./filter-typed.ts";
import { FILTER_LIMITS, OPERATORS_BY_TYPE, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, type LimitsGiven, validateFilter } from "./filter-validate.ts";
import type { SortField } from "./types.ts";
/**
 * Parse PostgREST-style `order` string into SortField array.
 *
 * Grammar:
 *   order  := entry ("," entry)*
 *   entry  := field ("." direction)? ("." nulls)?
 *   field  := any string not containing "." at the direction/nulls boundary
 *            (dotted relation paths OK: "author.name.asc")
 *   direction := "asc" | "desc"        (default "asc" when omitted)
 *   nulls     := "nullsfirst" | "nullslast"
 *
 * Examples:
 *   "name"                         → [{ field: "name", direction: "asc" }]
 *   "name.desc"                    → [{ field: "name", direction: "desc" }]
 *   "name.asc.nullslast"           → [{ field: "name", direction: "asc", nulls: "last" }]
 *   "col1.desc,col2.asc.nullsfirst"
 *     → [{field:"col1",direction:"desc"}, {field:"col2",direction:"asc",nulls:"first"}]
 *
 * An entry that is empty, names no field (`asc`, `.desc`, a trailing comma) or has whitespace around
 * it, and a field named twice, refuse the whole order as null, as parseFilter refuses a filter, rather
 * than drop it and widen the request to the default order.
 */
declare function parseOrder(input: string | null | undefined): SortField[] | null;
/** Each field `fields` names more than once, once each */
declare function namedTwice(fields: readonly SortField[]): string[];
/**
 * Each entry of `order` read as a sort field, or kept as its text when it is empty, names no field or
 * has whitespace around it, which the published pattern refuses too
 */
declare function orderEntries(input: string | null | undefined): Array<SortField | string>;
export { FILTER_LIMITS, FILTER_OPERATORS, OPERATORS_BY_TYPE, parseFilter, parseFilterNumber, namedTwice, orderEntries, parseOrder, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, type LimitsGiven, };

// query/filter-validate.d.ts
import type { FieldType, FilterAST, FilterOperator } from "./types.ts";
/** The operators each field type takes when its spec lists none; `is` takes null and notnull on every type */
declare const OPERATORS_BY_TYPE: Record<FieldType, readonly FilterOperator[]>;
/**
 * A field as a runtime field map declares it: its type, the operators it takes when not its
 * type's (`OPERATORS_BY_TYPE`), and for a date, `format: "iso"` to take ISO 8601 text, as a
 * column that stores it holds, in place of epoch milliseconds
 */
type FilterFieldSpec = {
    /** "iso", on a date field only: values are ISO 8601 text, not epoch milliseconds */
    format?: "iso" | undefined;
    /** the operators the field takes in place of its type's; `contains` on any type, and gt to lte on a string, beyond them */
    operators?: readonly FilterOperator[] | undefined;
    /** the type each value is checked and typed as */
    type: FieldType;
};
/** A field of `validateFilter`'s map: a bare type, or a spec */
type FilterField = FieldType | FilterFieldSpec;
/** A checked filter, or every reason it was refused */
type FilterValidationResult = {
    /** the filter, each value typed as its field declares */
    ast: FilterAST;
    /** true: the filter passed every check */
    valid: true;
} | {
    /** one message per problem, each naming its field, for a 400's detail */
    errors: string[];
    /** false: the filter was refused */
    valid: false;
};
/** Bounds on one filter expression. Each condition is a predicate SQLite evaluates per row. */
type FilterLimits = {
    /** Conditions in the whole filter, every group counted */
    maxConditions: number;
    /** Nested or()/and() groups below the top level */
    maxDepth: number;
    /** Members of one in/nin list */
    maxInValues: number;
    /** Bytes of a like or ilike pattern, a value with an inner `*`, once escaped: D1 caps a pattern at 50 */
    maxPatternBytes: number;
    /** Values the whole filter binds, every in/nin member counted: D1 binds at most 100 per statement */
    maxValues: number;
};
/** The default filter bounds: 20 conditions, 3 nested groups, 50 in/nin members, 50 pattern bytes and 60 values */
declare const FILTER_LIMITS: FilterLimits;
/** Bounds as a caller gives them: a key left out or set to undefined keeps its default, and any other is a whole number from 0 */
type LimitsGiven<T> = {
    [K in keyof T]?: number | undefined;
};
/**
 * `given` over `defaults`, a key set to undefined left out as every other option bag reads it, each
 * other a whole number from 0, a TypeError naming it and `owner` otherwise (KT1 of r22b): a cap of
 * NaN or undefined compared false, which turned it off. No cap is a large whole number
 */
declare function limitsOver<T extends Record<string, number>>(owner: string, defaults: T, given: LimitsGiven<T> | undefined): T;
/**
 * Parse `filterString` and check it against `allowedFields`: known fields, operators their type
 * or spec allows, values of their type, within `limits`. Null when there is no filter. A spec
 * that lists an operator its type cannot take, or a format on no date, throws a TypeError.
 */
declare function validateFilter(filterString: string | null | undefined, allowedFields: Record<string, FilterField>, limits?: LimitsGiven<FilterLimits> | undefined): FilterValidationResult | null;
export { FILTER_LIMITS, limitsOver, OPERATORS_BY_TYPE, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type LimitsGiven, type FilterValidationResult, };

// query/filter-parse.d.ts
/**
 * The PostgREST-style filter grammar: a filter string to its AST. Values stay as the request
 * wrote them; `validateFilter` checks them against the declared field types.
 *
 * - `field.op.value`: one condition; `,` joins conditions with AND
 * - `or(...)`, `and(...)`: groups, nestable
 * - `in.(a,b)`, `nin.(a,b)`: lists
 * - `"..."`: a quoted value may hold `,` `(` `)` and `.`; `\"` and `\\` escape inside it
 * - `@name`: a computed field
 */
import type { FilterAST, FilterOperator } from "./types.ts";
/** Every operator the filter grammar parses; `validateFilter` holds each field to its type's */
declare const FILTER_OPERATORS: readonly [FilterOperator, ...FilterOperator[]];
/**
 * The AST of a filter string, or null when there is none or any part of it cannot be read: a
 * filter the parser does not fully understand is refused, never reduced to "no filter", which
 * would widen the result to the whole table on a typo.
 */
declare function parseFilter(input: string | null | undefined): FilterAST | null;
export { FILTER_OPERATORS, parseFilter };

/**
