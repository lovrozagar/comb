// query/merged.d.ts
/**
 * A list whose rows sit on several D1 shards, merged in memory (L2 of the 0.13.0 consumers review).
 * Each shard reads its part through `mergedListOptions`: past the cursor, the page's rows and one
 * more, as `listOptions` reads them; by page number, every row up to the page's end, as any shard
 * may hold any of them. `paginateMerged` then orders the merge as SQLite's ORDER BY does, windows
 * it as the request asks, counts what every shard counted, and mints the next cursor from the
 * page's last row, so the merged list pages as one database's would, by number or by cursor.
 */
import { type SQL } from "drizzle-orm";
import type { CursorOptions } from "./cursor-seal.ts";
import { type DrizzleListOptions, queryHelpers } from "./drizzle.ts";
import { type ListPagination, type PaginationMeta } from "./pagination.ts";
type MergedQuery = Parameters<typeof queryHelpers.listOptions>[0];
/** listOptions for one shard: its extras may carry the shard's count beside the gone-anchor flag */
type MergedListOptions = Omit<DrizzleListOptions, "extras"> & {
    extras?: Record<string, (table: object) => SQL>;
};
/**
 * `listOptions` for one shard of a merged list: a cursor page as `listOptions` reads it, an offset
 * page from the list's first row to the page's last and one more, each row carrying the shard's
 * count, so no countExtra is needed. A page whose window passes
 * `maxRowsPerShard` (1001 by default) is `page_too_deep`: every shard would read every row before it
 */
declare function mergedListOptions(query: MergedQuery, options: CursorOptions & {
    maxRowsPerShard?: number | undefined;
}): MergedListOptions;
/**
 * The page of a list merged from `rowsPerShard`, each shard's rows read with `mergedListOptions`,
 * and its pagination, as `paginate` gives one database's: ordered as the list orders, windowed by
 * page or cursor, counted across shards, its next cursor sealed under `options`. Ids must be
 * unique across the shards, as a cursor names its row by id
 */
declare function paginateMerged<T extends object>(rowsPerShard: ReadonlyArray<ReadonlyArray<T & {
    _total?: number;
}>>, built: {
    pagination: ListPagination;
}): readonly [T[], PaginationMeta];
export { type MergedListOptions, mergedListOptions, paginateMerged };

// sqlite/d1/sharding/router.d.ts
import type { ShardDbHandle, ShardMapping, ShardMove, ShardPlacement, ShardResolverDeps, ShardRouterConfig, ShardVerifyDeps, WaitUntilCtx } from "./types.ts";
/**
 * Shard router — resolves org -> D1 database handle.
 *
 * Instantiate once per isolate (module-level) so the LRU cache persists.
 * 3-tier lookup: LRU -> KV -> Core D1; a new org goes to the shard `pick` names.
 */
declare class ShardRouter {
    private readonly cache;
    private readonly log;
    private readonly pick;
    private readonly totalShards;
    constructor(config: ShardRouterConfig);
    /**
     * Resolve existing mapping, or allocate a new shard if org is unknown.
     * This is the primary entry point for middleware.
     */
    resolveOrAllocate(orgId: string, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /**
     * Resolve shard mapping via LRU (SWR) -> KV -> Core D1.
     * Returns undefined if org has no mapping.
     * The core read goes to the nearest replica, and to the primary when the replica has no
     * mapping, which a fresh placement may not have reached yet; a miss there is remembered for
     * missTtlMs, so a run of requests for an unknown id cannot load the primary.
     */
    resolve(orgId: string, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping | undefined>;
    /**
     * Resolve as `resolve` does, then check with `deps.present` that the org's rows are on the shard
     * named: a mapping cached here can predate a `move` in another isolate, since invalidate
     * reaches only this isolate's LRU and KV. When they are not, the primary is read past every
     * cache and its mapping, or undefined when the org is gone, is cached and returned without a
     * second probe: when it names the shard just probed, the org's rows may not be there yet.
     */
    resolveVerified(orgId: string, deps: ShardVerifyDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping | undefined>;
    /**
     * Allocate a shard for an org: the one `pick` names, told each shard's live count. A
     * soft-deleted org is revived on the shard it had, where its data is, without asking pick. The
     * count is read before the insert, so two isolates allocating at once can both take a shard's
     * last place: maxOrgsPerShard is a soft cap. Writes go to the primary.
     */
    allocate(orgId: string, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /**
     * Place an org on a shard the app chose. The first writer wins: an org that has a row keeps
     * its shard, a soft-deleted one revived there, and the mapping that stands is returned, so two
     * isolates placing one new org at once agree. To put a placed org elsewhere, `move` it.
     */
    place(placement: ShardPlacement, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /**
     * Move an org to `shardId` and invalidate it here. Moving its rows is the app's; its writes on
     * the core database for the move (`with`) batch with the mapping's, landing together or not at
     * all (R4). Another isolate routes to the old shard until its caches expire or it verifies.
     */
    move(placement: ShardMove, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /** Soft-delete an org's mapping and invalidate it here; a later allocate or place revives it on its shard */
    unplace(orgId: string, deps: ShardResolverDeps): Promise<void>;
    /**
     * Resolve a shard mapping to a concrete D1 database handle.
     * Reads `SHARD_DB_{target}` binding from env.
     */
    getHandle(mapping: ShardMapping, env: Record<string, unknown>): ShardDbHandle;
    /**
     * Invalidate both cache tiers. Use after whale graduation or reassignment. It reaches this
     * isolate's LRU and the KV; another isolate's LRU keeps the mapping until it expires.
     */
    invalidate(orgId: string, kv: ShardResolverDeps["kv"]): Promise<void>;
    /** Places an org as `place` does, unchecked: an org being revived keeps the shard it had */
    private placeFirst;
    /**
     * Caches the row a write left standing, this isolate's miss for the org forgotten, and returns
     * its mapping; logs the placement only when the write placed it, so the events count rows (U40)
     */
    private settle;
}
export { ShardRouter };

// query/cursor-seal.d.ts
/** What seals and opens the cursors of one list: the app's secret, and the list's scope (route plus tenant) */
type CursorOptions = {
    cursorScope: string;
    cursorSecret: string;
    /**
     * "exact" when the driver reads a SQLite REAL's ±Infinity back as Infinity (a local libsql file,
     * bun:sqlite), so a null cursor is a real NULL and a list whose NULL anchor is deleted resumes
     * right after it. Unset, as on D1, which reads ±Infinity as null, such a list resumes from the
     * earliest value the null may stand for, repeating rows and skipping none (U24).
     */
    nonFinite?: "exact" | undefined;
};
/** The shortest secret a cursor is sealed under, in UTF-8 bytes: the key's length */
declare const MIN_CURSOR_SECRET_BYTES = 32;
/** The 32-byte key HKDF-SHA256 derives from `secret`, with no salt and comb's info */
declare function cursorKey(secret: string): Uint8Array;
/** Throw the 500 sealing throws when `options` cannot seal a cursor */
declare function assertCursorWired(options: CursorOptions): void;
/**
 * `plaintext` sealed under `options`, as base64url, padded with spaces up to a multiple of 256
 * bytes first (U9): AEAD hides the bytes, not their count, and a cursor carries a hidden column's
 * value too. openSealed strips them: what seal takes ends in no space of its own.
 */
declare function seal(plaintext: Uint8Array, options: CursorOptions): string;
/** Whether `cursor` has a sealed cursor's shape: what a check without the secret can know */
declare const isSealedShape: (cursor: string) => boolean;
/** What `cursor` holds, opened under `options`; null when it was not sealed under both */
declare function openSealed(cursor: string, options: CursorOptions): Uint8Array | null;
export { assertCursorWired, cursorKey, type CursorOptions, isSealedShape, MIN_CURSOR_SECRET_BYTES, openSealed, seal };

// query/cursor.d.ts
import { type CursorOptions } from "./cursor-seal.ts";
import type { SortDirection, SortField, SortNulls } from "./types.ts";
/**
 * Column keyset pagination breaks ties on.
 *
 * This is the single source of truth for the tiebreak: the SQL builders resolve
 * the column by this name, and `createListQuerySchema` publishes it as
 * `stableTiebreak`. Keeping one constant is what stops the published fact from
 * drifting away from the query that is actually run.
 *
 * A table whose primary-key property is named something else has no cursor
 * predicate applied at all — rows then duplicate and skip across pages — so
 * `validateTables()` rejects that shape rather than letting it fail silently.
 */
declare const CURSOR_TIEBREAK_COLUMN = "id";
/**
 * The longest cursor a list takes, and so the longest comb mints: it fits a URL. A cursor carries
 * its sort values, so a long text key could mint one past it, which the next page would refuse.
 */
declare const MAX_CURSOR_LENGTH = 8192;
/**
 * Wire payload. `s` is the sort signature the cursor was minted under, `v` one
 * value per sort key in order, `i` the tiebreak id. A cursor is only valid for
 * the exact sort that produced it.
 */
type CursorPayload = {
    i: CursorId;
    s: string;
    v: unknown[];
};
/** The tiebreak id as the row held it: a number stays a number, a bigint a bigint, so it compares as one */
type CursorId = bigint | number | string;
type DecodedCursor = {
    id: CursorId;
    sort: string;
    values: CursorValue[];
};
/**
 * A sort value carried by reference (cursor-ref.ts): the prefix of it that fit and the whole
 * value's length, both in code points as SQLite's substr() and Postgres's left() count them; the
 * rest is read from the row while it still has that prefix and length
 */
declare class CursorRef {

/**
 * Type-safe column constraint definitions
 * These exist only in the type system and are completely erased at runtime
 */
type BaseConstraints = {
    /** Field is filled by comb or the database: generated create and update DTOs leave it out */
    autogenerate?: boolean;
    /**
     * A foreign key that may move after creation (a person to another team): generated update DTOs
     * take it. Every other reference is set once, at creation. Refused on a column that references
     * nothing, and together with `nomutate`.
     */
    mutable?: boolean;
    /** Field cannot be mutated after creation */
    nomutate?: boolean;
    /**
     * Field is never read back: generated read DTOs and the meta stamp leave it out, and the manifest
     * lists it flagged `private: true`, on both dialects
     */
    private?: boolean;
    /** the name this key's relation takes, where its column's role would take the column's own name (F57) */
    relation?: string;
    /**
     * This column scopes rows to a tenant.
     *
     * Declared, never inferred: comb can see that a column is a foreign key but
     * not that the table it points at is the tenant, and a wrong tenant boundary
     * is worse than a missing one. Saying so here makes it a hand-reviewed fact.
     * At most one column per table may carry it.
     */
    tenant?: boolean;
};
type StringTransformConstraints = {
    /** Apply lowercase transformation */
    lowercase?: boolean;
    /** Trim whitespace */
    trim?: boolean;
    /** Apply uppercase transformation */
    uppercase?: boolean;
};
type NumericRangeConstraints = {
    /** Maximum value */
    max?: number;
    /** Minimum value */
    min?: number;
};
type StringValidationConstraints = {
    /** Email validation */
    email?: boolean;
    /** Image URI validation (accepts https URLs and data:image/ URIs) */
    imageUri?: boolean;
    /** Human/entity name validation (letters, marks, numbers, spaces, punctuation) */
    name?: boolean;
    /** Password validation (min 8, max 72, printable ASCII) */
    password?: boolean;
    /** Regex pattern validation */
    pattern?: string;
    /** URL validation */
    url?: boolean;
};
/**
 * Text column constraints
 * Closed record - only these properties allowed
 */
export type TextConstraints = BaseConstraints & StringTransformConstraints & NumericRangeConstraints & StringValidationConstraints;
/**
 * Integer/numeric column constraints
 * Closed record - only these properties allowed
 */
export type IntegerConstraints = BaseConstraints & NumericRangeConstraints;
/**
 * JSON column constraints
 * Closed record - only these properties allowed
 */
export type JsonConstraints = BaseConstraints & {
    /** Maximum JSON stringified byte length */
    maxBytes?: number;
};
/**
 * Timestamp column constraints
 * Closed record - only these properties allowed
 */
export type TimestampConstraints = BaseConstraints & NumericRangeConstraints;
/**
 * State machine declared alongside an enum column's values.
 *
 * Type-level only, like every other constraint here — the analyzer reads it
 * out of the source. Hold the same object as a const and pass it to both
 * `c.enum` and `assertTransition`. See docs/state-machines.md.
 */
export type EnumStates<TState extends string = string> = {
    /** The value a row starts at */
    initial?: TState | undefined;
    /** Values from which no transition is legal */
    terminal?: readonly TState[] | undefined;
    /** Legal moves, `from` → `to[]`. Omitted states are unconstrained. */
    transitions?: Readonly<Partial<Record<TState, readonly TState[]>>> | undefined;
};
/** What c.enum takes beside its values: its state machine, and the flags every c helper takes (F9) */
export type EnumOptions<TState extends string = string> = EnumStates<TState> & Pick<BaseConstraints, "nomutate" | "private" | "tenant">;
export {};

/**
 * PostgREST-style filters and sort orders: parse and validate. DB-agnostic: no SQL, only
 * the AST and its checks. The grammar lives in filter-parse.ts, the checks in filter-validate.ts.
 */
import { FILTER_OPERATORS, parseFilter } from "./filter-parse.ts";
import { FILTER_LIMITS, OPERATORS_BY_TYPE, parseFilterNumber, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, validateFilter } from "./filter-validate.ts";
import type { SortField } from "./types.ts";
/**
 * Parse PostgREST-style `order` string into SortField array.
 *
 * Grammar:
 *   order  := entry ("," entry)*
 *   entry  := field ("." direction)? ("." nulls)?
 *   field  := any string not containing "." at the direction/nulls boundary
 *            (dotted relation paths OK: "author.name.asc")
 *   direction := "asc" | "desc"        (default "asc" when omitted)
 *   nulls     := "nullsfirst" | "nullslast"
 *
 * Examples:
 *   "name"                         → [{ field: "name", direction: "asc" }]
 *   "name.desc"                    → [{ field: "name", direction: "desc" }]
 *   "name.asc.nullslast"           → [{ field: "name", direction: "asc", nulls: "last" }]
 *   "col1.desc,col2.asc.nullsfirst"
 *     → [{field:"col1",direction:"desc"}, {field:"col2",direction:"asc",nulls:"first"}]
 *
 * Malformed entries are dropped silently — downstream schema validator
 * rejects unknown fields, which catches most mistakes.
 */
declare function parseOrder(input: string | null | undefined): SortField[];
export { FILTER_LIMITS, FILTER_OPERATORS, OPERATORS_BY_TYPE, parseFilter, parseFilterNumber, parseOrder, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, };
