/** The options the SQLite list builders take, and the query they build (build-list-query.ts). Declarations only. */
import type { SQL, SQLWrapper } from "drizzle-orm";
import type { SQLiteColumn, SQLiteTable } from "drizzle-orm/sqlite-core";
import type { ListPagination } from "../pagination.ts";
import type { ComputedFilterResolver, ComputedSortResolver, ListQueryInput } from "../types.ts";
import type { LikeSearchResolver } from "./search.ts";
import type { RelationConfig } from "./types.ts";
type BuildListQueryOpts<TTable extends SQLiteTable> = {
    /** Additional WHERE clause ANDed with filter/cursor — for joins, access filters, soft deletes */
    baseWhere?: SQL | ((sql: (strings: TemplateStringsArray, ...values: unknown[]) => SQL) => SQL) | undefined;
    computedFilters?: Record<string, ComputedFilterResolver> | undefined;
    computedSorts?: Record<string, ComputedSortResolver> | undefined;
    /** the list's scope, its route and tenant: a cursor minted in another opens in none here */
    cursorScope: string;
    /** the app's secret a cursor is sealed under, of at least 32 UTF-8 bytes */
    cursorSecret: string;
    /** "exact" when the driver reads ±Infinity back as Infinity, so a null cursor is a real NULL (CursorOptions) */
    nonFinite?: "exact" | undefined;
    /**
     * A list whose rows sit on several D1 shards: every shard reads it with this one query, and
     * paginateMerged merges what they read (K5 of r17). A page by number reads every row to its end
     * from each shard, past `maxRowsPerShard` (1001) a 400 `page_too_deep`
     */
    merged?: {
        maxRowsPerShard?: number | undefined;
    } | undefined;
    parsed: ListQueryInput;
    relations?: Record<string, RelationConfig> | undefined;
    /** LIKE-based search resolver — use `likeSearch(column)` factory */
    search?: LikeSearchResolver | undefined;
    table: TTable;
};
type BuildDerivedListQueryOpts<TSortField extends string = string> = {
    baseWhere?: SQL | ((sql: (strings: TemplateStringsArray, ...values: unknown[]) => SQL) => SQL) | undefined;
    /** the list's scope, its route and tenant: a cursor minted in another opens in none here */
    cursorScope: string;
    /** the app's secret a cursor is sealed under, of at least 32 UTF-8 bytes */
    cursorSecret: string;
    /**
     * the database the list reads, needed only where its id and keys hold no column: a column says it
     * wherever it sits, and a list of SQL expressions alone (a union, say) cannot (Q1 of r21)
     */
    dialect?: "postgres" | "sqlite" | undefined;
    filterWhere?: SQL | null | undefined;
    /**
     * Present when the rows come from several shards, which paginateMerged merges, as on
     * buildListQuery: each shard reads its window, an offset page its count. A merged derived list
     * carries no sort key by reference, and its rows carry each sort field under its name (Q12 of r18).
     */
    merged?: {
        maxRowsPerShard?: number | undefined;
    } | undefined;
    parsed: ListQueryInput<TSortField>;
    searchWhere?: SQL | null | undefined;
    /** each sort field's expression; one that yields epoch milliseconds is `{ sql, type: "date" }`, and its cursor value a Date */
    sortColumns: Record<TSortField, SQLWrapper | {
        sql: SQLWrapper;
        type: "date";
    }>;
} & ({
    /**
     * the one table the rows are, by its id column: a sort key too long for a cursor is
     * carried by reference and read back from it, as on a table's own list (C3)
     */
    anchor: {
        idColumn: SQLiteColumn;
        table: SQLiteTable;
    };
    idColumn?: never | undefined;
} | {
    anchor?: never | undefined;
    /** the id the order breaks ties on; with no table to read a key back from, a key too long for a cursor is a 500 */
    idColumn: SQLWrapper;
});
/**
 * The columns a built list's rows must carry beside the app's own, for `paginate` to read: select
 * them with the rest (`db.select({ ...cols, ...built.extras })`), and paginate strips them (Q2)
 */
type ListExtras = Partial<{
    /** 1 when the cursor's anchor row is gone where no bound is safe, which paginate answers invalid_cursor (U26) */
    _combAnchorGone: SQL.Aliased<number>;
    /** the value the list breaks ties on, which the next cursor resumes from; none when no cursor can */
    _combId: SQL.Aliased<unknown>;
    /** the class of each sort key and the tiebreak, comma-joined, where the list binds them as read or one is an undeclared SQLite expression (Q2 of r19, Q2 of r21) */
    _combKeyTypes: SQL.Aliased<string>;
    /** each sort key's stored time where the list binds it as read, comma-joined, empty for any other key: a Postgres time key's text, a SQLite REAL seconds (QL2 of r22b) */
    _combKeyTexts: SQL.Aliased<string>;
    /** an offset page's count of every row the filter matches; a cursor page selects none (keyset avoids it) */
    _total: SQL.Aliased<number>;
}>;
type ListQueryResult = {
    /** select these beside the row's own columns (ListExtras) */
    extras: ListExtras;
    limit: number;
    /** How `paginate` pages the rows this query reads (pagination.ts) */
    pagination: ListPagination & {
        merged: false;
    };
    meta: {
        limit: number;
        page: number;
        type: "cursor" | "offset";
    };
    offset: number;
    orderBy: SQL[];
    search: string | null;
    where: SQL | undefined;
};
/** A merged list's query, which every shard reads and paginateMerged alone pages (K8 of r17) */
type MergedListQueryResult = Omit<ListQueryResult, "pagination"> & {
    pagination: ListPagination & {
        merged: true;
    };
};
export type { BuildDerivedListQueryOpts, BuildListQueryOpts, ListExtras, ListQueryResult, MergedListQueryResult };

// sqlite/d1/sharding/router.d.ts
import type { D1DatabaseLike } from "../d1-types.ts";
import type { ShardBindingOf, ShardDbHandle, ShardEnv, ShardMapping, ShardMove, ShardPlacement, ShardResolverDeps, ShardRouterConfig, ShardVerifyDeps, WaitUntilCtx } from "./types.ts";
/**
 * Shard router — resolves org -> D1 database handle.
 *
 * Instantiate once per isolate (module-level) so the LRU cache persists.
 * 3-tier lookup: LRU -> KV -> Core D1; a new org goes to the shard `pick` names.
 */
declare class ShardRouter {
    private readonly cache;
    private readonly log;
    private readonly pick;
    private readonly totalShards;
    constructor(config: ShardRouterConfig);
    /**
     * Resolve existing mapping, or allocate a new shard if org is unknown.
     * This is the primary entry point for middleware.
     */
    resolveOrAllocate(orgId: string, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /**
     * Resolve shard mapping via LRU (SWR) -> KV -> Core D1.
     * Returns undefined if org has no mapping.
     * The core read goes to the nearest replica, and to the primary when the replica has no
     * mapping, which a fresh placement may not have reached yet; a miss there is remembered for
     * missTtlMs, so a run of requests for an unknown id cannot load the primary.
     */
    resolve(orgId: string, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping | undefined>;
    /**
     * Resolve as `resolve` does, then check with `deps.present` that the org's rows are on the shard
     * named: a mapping cached here can predate a `move` in another isolate, since invalidate
     * reaches only this isolate's LRU and KV. When they are not, the primary is read past every
     * cache. When it names the shard just probed, the org's rows may not be there yet: that mapping
     * is returned, and nothing is written or logged (Q13 of r17). When it names another, or none,
     * the org is invalidated, the primary read again under the tombstone, and that mapping, or
     * undefined when the org is gone, returned without a second probe; it is cached only while no
     * invalidate here, a move's among them, came after the tombstone (Q7 of r17).
     */
    resolveVerified(orgId: string, deps: ShardVerifyDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping | undefined>;
    /**
     * Allocate a shard for an org: the one `pick` names, told each shard's live count. A
     * soft-deleted org is revived on the shard it had, where its data is, without asking pick. The
     * count is read before the insert, so two isolates allocating at once can both take a shard's
     * last place: maxOrgsPerShard is a soft cap. Writes go to the primary.
     */
    allocate(orgId: string, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /**
     * Place an org on a shard the app chose. The first writer wins: an org that has a row keeps
     * its shard, a soft-deleted one revived there, and the mapping that stands is returned, so two
     * isolates placing one new org at once agree. To put a placed org elsewhere, `move` it.
     */
    place(placement: ShardPlacement, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /**
     * Move an org to `shardId` and invalidate it here. Moving its rows is the app's; its writes on
     * the core database for the move (`with`) batch with the mapping's, landing together or not at
     * all (R4). Another isolate routes to the old shard until its caches expire or it verifies.
     */
    move(placement: ShardMove, deps: ShardResolverDeps, executionCtx: WaitUntilCtx): Promise<ShardMapping>;
    /** Soft-delete an org's mapping and invalidate it here; a later allocate or place revives it on its shard */
    unplace(orgId: string, deps: ShardResolverDeps): Promise<void>;
    /**
     * Resolve a shard mapping to a concrete D1 database handle: the env's `SHARD_DB_{target}`
     * binding, in the type the env holds its shard bindings as (K2 of r22). An env typed by a string
     * index signature gives the D1 binding it holds, else D1DatabaseLike; one whose known keys hold no
     * `SHARD_DB_*` is refused.
     */
    getHandle<E extends object>(mapping: ShardMapping, env: E & ShardEnv<E>): ShardDbHandle<ShardBindingOf<E>>;
    /**
     * Resolve a shard mapping for an env TypeScript cannot read the keys of, such as a helper's type
     * parameter bounded by an env interface whose `SHARD_DB_0` is required: the binding type that key
     * holds, since shard 0 always exists, with every other `SHARD_DB_*` key it names a D1 binding
     * (D2-1 and D2-2 of the area D round-two review). An env bounded by no shard key is refused. So
     * is a bound whose `SHARD_DB_0` is optional, as `wrangler types` writes it when no shard is bound
     * at the top level: the helper widens its parameter to that env first, `const env: Env = given`,
     * and the first overload then gives shard 0's binding (D3-1 of the round-three review).
     */
    getHandle<E extends {
        readonly SHARD_DB_0: D1DatabaseLike;
    }>(mapping: ShardMapping, env: E & {
        readonly [K in keyof E & `SHARD_DB_${string}`]?: D1DatabaseLike | undefined;
    }): ShardDbHandle<E["SHARD_DB_0"]>;
    /**
     * Invalidate both cache tiers. Use after whale graduation or reassignment. It reaches this
     * isolate's LRU and the KV; another isolate's LRU keeps the mapping until it expires.
     */
    invalidate(orgId: string, kv: ShardResolverDeps["kv"]): Promise<void>;
    /** Places an org as `place` does, unchecked: an org being revived keeps the shard it had */
    private placeFirst;
    /**
     * Caches the row a write left standing, this isolate's miss for the org forgotten, and returns
     * its mapping; logs the placement only when the write placed it, so the events count rows (U40).
     * Cached only while no invalidate here came after `generation`, which was taken before the
     * write read the row: one that did, a move's among them, may have remapped it (Q6 of r18)
     */
    private settle;
}
export { ShardRouter };

/**
 * The binding getHandle hands back for an env of type `E` (K2 of r22): the type its `SHARD_DB_*`
 * keys hold, as `wrangler types` writes them, a key bound in only some environments among them
 * (AD1); an env with a string index signature gives the D1 binding its index holds (AD14), and
 * the structural D1DatabaseLike when it holds none
 */
type ShardBindingOf<E> = string extends keyof E ? Extract<NonNullable<E[string & keyof E]>, D1DatabaseLike> extends infer D extends D1DatabaseLike ? [D] extends [never] ? D1DatabaseLike : D : never : Extract<NonNullable<E[Extract<keyof E, `SHARD_DB_${string}`>]>, D1DatabaseLike>;
/**
 * What getHandle asks of an env of type `E` beyond `E` itself: each `SHARD_DB_*` key holds a D1
 * binding, and an env whose known keys hold no such key is refused, naming the key it lacks. A key
 * may be optional, as `wrangler types` writes one bound in only some environments (AD1): getHandle
 * throws on an absent one at run time. An env with a string index signature is taken when its
 * index may hold a D1 binding, and refused when it provably holds none (D2-5 of the area D
 * round-two review)
 */
type ShardEnv<E> = string extends keyof E ? unknown extends E[string & keyof E] ? unknown : [Extract<NonNullable<E[string & keyof E]>, D1DatabaseLike>] extends [never] ? {
    "SHARD_DB_<target>": D1DatabaseLike;
} : unknown : [Extract<keyof E, `SHARD_DB_${string}`>] extends [never] ? {
    "SHARD_DB_<target>": D1DatabaseLike;
} : {
    [K in Extract<keyof E, `SHARD_DB_${string}`>]?: D1DatabaseLike | undefined;
};

// sqlite/d1/d1-types.d.ts
/**
 * The D1 surface comb calls, as structural types of its own (K3 of r21), as the router's KV is: a
 * Worker typed by `wrangler types`, whose globals declare D1, and one typed by
 * `@cloudflare/workers-types` both fit them, so comb's D1 modules are typed in either and need
 * neither installed. Where comb hands a client on, it hands the caller's own type back.
 */
/** A prepared statement as comb uses one: bound, then batched */
type D1StatementLike = {
    bind(...values: unknown[]): D1StatementLike;
};
/** One statement's answer in a batch: the rows comb reads */
type D1AnswerLike<T> = {
    results: T[];
};
/** A D1 session: what comb prepares and batches on, and the bookmark it hands on */
type D1SessionLike = {
    batch<T = unknown>(statements: D1StatementLike[]): Promise<Array<D1AnswerLike<T>>>;
    getBookmark(): string | null;
    prepare(query: string): D1StatementLike;
};
/** A D1 database binding; one with no `withSession` is a runtime without sessions (an older Miniflare) */
type D1DatabaseLike = {
    batch<T = unknown>(statements: D1StatementLike[]): Promise<Array<D1AnswerLike<T>>>;
    prepare(query: string): D1StatementLike;
    withSession?: ((constraintOrBookmark?: string) => D1SessionLike) | undefined;
};
/** The session type `D`'s `withSession` opens: the caller's own, so drizzle() takes it as it takes `D`; a D1 session where `D` names none */
type SessionOf<D extends D1DatabaseLike> = D extends {
    withSession(...args: never[]): infer S;
} ? S : D1SessionLike;
export type { D1AnswerLike, D1DatabaseLike, D1SessionLike, D1StatementLike, SessionOf };

/**
 * One column a builder selects for paginate: the tiebreak, the gone-anchor flag, the count (Q3 of
 * r17), or the class of each sort key (Q2 of r19, Q2 of r21)
 */
type ListExtra = "_combAnchorGone" | "_combId" | "_combKeyTexts" | "_combKeyTypes" | "_total";
/**
 * A row as paginate and paginateMerged return it: the extras gone, from each member of a union
 * row type on its own, so narrowing on its discriminant still reaches each variant's fields (Q3 of r19)
 */
type ListRow<T> = T extends unknown ? Omit<T, ListExtra> : never;
/** The pagination meta paginate returns: the list response's own schema */
type PaginationResponse = z.output<typeof paginationResponseSchema>;
/** How a built list pages what it fetched */
type ListPagination = {
    /** the cursor the request resumed from; none on a page by number */
    cursor: string | undefined;
    /** false when no cursor can resume the list: its order holds a computed sort, or it has no tiebreak (Q6) */
    cursorable: boolean;
    cursorScope: string;
    /** the extras the builder selected, which every row it reads must carry (Q3) */
    expects: readonly ListExtra[];
    /** how a key too long for a cursor is carried: by reference to its row ("table"), or not at all ("derived") */
    keyset: KeysetKind;
    limit: number;
    /** read through buildListQuery's `merged`: paged by paginateMerged, never by paginate (K8 of r17) */
    merged: boolean;
    nonFinite: "exact" | undefined;
    page: number | undefined;
    parsedFields: ParsedFields | null;
    parsedSort: readonly SortField[];
    /** the most rows the built query reads: one past the page, so paginate can tell there is more */
    window: number;
    /** the cursor for the page after `row`, sealed under the builder's options */
    mint(row: object): string;
    /** `cursor` opened under the builder's options, for this list's sort; a 400 when it is no cursor of it */
    read(cursor: string): DecodedCursor;
};
/** What a builder knows of its list, beside the request it parsed */
type PaginationFacts = {
    /** the sort fields whose keys the list binds as read, with no row to read a bound from (QL1 of r22b); every one on a derived list */
    asRead?: ReadonlySet<string> | undefined;
    /** the sort fields whose keys read a reference back from the list's table (readsBack); every one when not given */
    byReference?: ReadonlySet<string> | undefined;
    cursorable: boolean;
    /** the extras the builder selected, which every row must carry */
    expects: readonly ListExtra[];
    keyset?: KeysetKind | undefined;
    query: ListQueryInput;
    window: number;
};
/** The descriptor a builder hands over: its facts, and the options it seals under closed over, never held */
declare function paginationOf<TMerged extends boolean = false>(options: CursorOptions, facts: PaginationFacts & {
    merged?: TMerged | undefined;
}): ListPagination & {
    merged: TMerged;
};
/** The count an offset page's row carries, in any encoding isCount takes, which refuseUnwired holds every such row to */
declare const totalOf: (row: object) => number;
/**
 * Rows that lack an extra their builder selected were read without the built query's extras, and
 * an absent flag or count would be read as none: a page_not_wired, whether or not the page needs
 * it yet (Q3 of r17)
 */
declare function refuseUnwired(items: readonly object[], pagination: ListPagination): void;
/** Read before anything else, _total included: a list past a gone anchor is no page of it (U26) */
declare function refuseGoneAnchor(items: readonly object[]): void;
/**
 * Last mutation of item shape. Query stays fat; the returned object is the
 * public subset. No-op (minus `_total`) when parsedFields is null.
 */
declare function applySelect<T extends Record<string, unknown>>(row: T, parsedFields: ParsedFields | null): T;
/**
 * `built` as paginate (or, `merged`, paginateMerged) takes it, or a 500 naming the call: a parsed
 * query in its place is the call before 0.15, and a merged list's pages are paginateMerged's alone
 */
declare function paginationIn<TMerged extends boolean>(built: {
    pagination: ListPagination & {
        merged: TMerged;
    };
} | undefined, merged: TMerged): ListPagination & {
    merged: TMerged;
};
/** More rows than the built query reads are rows of another query: one that applied no limit (C2) */
declare function refuseOverRead(count: number, pagination: ListPagination): void;
/**
 * The page of `items`, read with `built`, and its pagination: count, whether there is more, and
 * the next cursor, sealed under the builder's options; none for a list no cursor can resume
 */
declare function paginate<T extends object>(items: readonly T[], built: {
    pagination: ListPagination & {
        merged: false;

// query/filter.d.ts
/**
 * PostgREST-style filters and sort orders: parse and validate. DB-agnostic: no SQL, only
 * the AST and its checks. The grammar lives in filter-parse.ts, the checks in filter-validate.ts.
 */
import { FILTER_OPERATORS, parseFilter } from "./filter-parse.ts";
import { FILTER_LIMITS, OPERATORS_BY_TYPE, parseFilterNumber, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, type LimitsGiven, validateFilter } from "./filter-validate.ts";
import type { SortField } from "./types.ts";
/**
 * Parse PostgREST-style `order` string into SortField array.
 *
 * Grammar:
 *   order  := entry ("," entry)*
 *   entry  := field ("." direction)? ("." nulls)?
 *   field  := any string not containing "." at the direction/nulls boundary
 *            (dotted relation paths OK: "author.name.asc")
 *   direction := "asc" | "desc"        (default "asc" when omitted)
 *   nulls     := "nullsfirst" | "nullslast"
 *
 * Examples:
 *   "name"                         → [{ field: "name", direction: "asc" }]
 *   "name.desc"                    → [{ field: "name", direction: "desc" }]
 *   "name.asc.nullslast"           → [{ field: "name", direction: "asc", nulls: "last" }]
 *   "col1.desc,col2.asc.nullsfirst"
 *     → [{field:"col1",direction:"desc"}, {field:"col2",direction:"asc",nulls:"first"}]
 *
 * An entry that is empty or names no field (`asc`, `.desc`, a trailing comma) refuses the whole
 * order as null, as parseFilter refuses a filter, rather than drop it and widen the request to the
 * default order (QL7 of r22b).
 */
declare function parseOrder(input: string | null | undefined): SortField[] | null;
/** Each entry of `order` read as a sort field, or kept as its trimmed text when it is empty or names no field */
declare function orderEntries(input: string | null | undefined): Array<SortField | string>;
export { FILTER_LIMITS, FILTER_OPERATORS, OPERATORS_BY_TYPE, parseFilter, parseFilterNumber, orderEntries, parseOrder, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, type LimitsGiven, };

// query/filter-validate.d.ts
import type { FieldType, FilterAST, FilterOperator } from "./types.ts";
declare const OPERATORS_BY_TYPE: Record<FieldType, readonly FilterOperator[]>;
/**
 * A field as a runtime field map declares it: its type, the operators it takes when not its
 * type's (`OPERATORS_BY_TYPE`), and for a date, `format: "iso"` to take ISO 8601 text, as a
 * column that stores it holds, in place of epoch milliseconds
 */
type FilterFieldSpec = {
    format?: "iso" | undefined;
    operators?: readonly FilterOperator[] | undefined;
    type: FieldType;
};
/** A field of `validateFilter`'s map: a bare type, or a spec */
type FilterField = FieldType | FilterFieldSpec;
type FilterValidationResult = {
    ast: FilterAST;
    valid: true;
} | {
    errors: string[];
    valid: false;
};
/** Bounds on one filter expression. Each condition is a predicate SQLite evaluates per row. */
type FilterLimits = {
    maxConditions: number;
    /** Nested or()/and() groups below the top level */
    maxDepth: number;
    maxInValues: number;
    /** Bytes of a like or ilike pattern, a value with an inner `*`, once escaped: D1 caps a pattern at 50 */
    maxPatternBytes: number;
    /** Values the whole filter binds, every in/nin member counted: D1 binds at most 100 per statement */
    maxValues: number;
};
declare const FILTER_LIMITS: FilterLimits;
/** A limits bag as a caller gives it: any key left out or set to undefined keeps its default (KT1 of r22b) */
type LimitsGiven<T> = {
    [K in keyof T]?: number | undefined;
};
/**
 * `given` over `defaults`, a key set to undefined left out as every other option bag reads it, each
 * other a whole number from 0, a TypeError naming it and `owner` otherwise (KT1 of r22b): a cap of
 * NaN or undefined compared false, which turned it off. No cap is a large whole number
 */
declare function limitsOver<T extends Record<string, number>>(owner: string, defaults: T, given: LimitsGiven<T> | undefined): T;
/**
 * A number field's value: finite (1e999 reads as Infinity, which no column holds), 0 or normal
 * (1e-400 reads as 0, which would match the rows holding 0), and within
 * ±(2^53 − 1), past which a double binds a neighbouring value. The value is read, not its
 * spelling: `9007199254740993.0` and `9.007199254740993e15` are the same unsafe integer.
 */
declare function parseFilterNumber(raw: string): {
    value: number;
} | {
    error: string;
};
/**
 * Parse `filterString` and check it against `allowedFields`: known fields, operators their type
 * or spec allows, values of their type, within `limits`. Null when there is no filter. A spec
 * that lists an operator its type cannot take, or a format on no date, throws a TypeError.
 */
declare function validateFilter(filterString: string | null | undefined, allowedFields: Record<string, FilterField>, limits?: LimitsGiven<FilterLimits> | undefined): FilterValidationResult | null;
export { FILTER_LIMITS, limitsOver, OPERATORS_BY_TYPE, parseFilterNumber, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type LimitsGiven, type FilterValidationResult, };

// query/filter-parse.d.ts
/**
 * The PostgREST-style filter grammar: a filter string to its AST. Values stay as the request
 * wrote them; `validateFilter` checks them against the declared field types.
 *
 * - `field.op.value`: one condition; `,` joins conditions with AND
 * - `or(...)`, `and(...)`: groups, nestable
 * - `in.(a,b)`, `nin.(a,b)`: lists
 * - `"..."`: a quoted value may hold `,` `(` `)` and `.`; `\"` and `\\` escape inside it
 * - `@name`: a computed field
 */
import type { FilterAST, FilterOperator } from "./types.ts";
declare const FILTER_OPERATORS: readonly [FilterOperator, ...FilterOperator[]];
/**
 * The AST of a filter string, or null when there is none or any part of it cannot be read: a
 * filter the parser does not fully understand is refused, never reduced to "no filter", which
 * would widen the result to the whole table on a typo.
 */
declare function parseFilter(input: string | null | undefined): FilterAST | null;
export { FILTER_OPERATORS, parseFilter };
