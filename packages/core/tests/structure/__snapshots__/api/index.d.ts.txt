// error.d.ts
import type { StatusKey } from "./types.ts";
/**
 * A database or query failure in your domain's terms: an error key, an HTTP status, and the table and
 * column it concerns. Its message is `errorKey`; `toJSON()` gives the body a client may see, and the
 * original error stays on `source`, never serialized.
 */
export declare class CombError extends Error {
    /** A message safe to show the client. The original error is on `source`. */
    readonly cause?: string | undefined;
    /** The SQL name of the column the error concerns; unset when it is private or unknown */
    readonly column?: string | undefined;
    /** What went wrong, as a stable key, one of `combErrorKeys` or a constraint map's own */
    readonly errorKey: string;
    /** The HTTP status code for `statusKey`, such as 409 for `conflict` */
    readonly status: number;
    /** The status the error answers with, by name */
    readonly statusKey: StatusKey;
    /** The SQL name of the table the error concerns, when known */
    readonly table?: string | undefined;
    /** The error this one was made from, for logs. Never serialized: it can hold SQL and bound values. */
    readonly source: unknown;
    /** An error with `opts.errorKey` as its message; `status` is a status key, and the code is derived from it */
    constructor(opts: {
        /** A message safe to show the client. The original error goes on `source`. */
        cause?: string | undefined;
        /** The SQL name of the column the error concerns */
        column?: string | undefined;
        /** What went wrong, as a stable key */
        errorKey: string;
        /** The original error, kept for logs and never serialized. */
        source?: unknown;
        /** The status the error answers with, by name: `conflict`, not 409 */
        status: StatusKey;
        /** The SQL name of the table the error concerns */
        table?: string | undefined;
    });
    /** The body a client gets: the documented fields alone, however the engine enumerates them */
    toJSON(): {
        /** The column, left out when unset */
        column?: string | undefined;
        /** What went wrong, as the stable key a client branches on */
        errorKey: string;
        /** The HTTP status code */
        status: number;
        /** The status, by name */
        statusKey: StatusKey;
        /** The table, left out when unset */
        table?: string | undefined;
    };
}

// password-text.d.ts
/**
 * The plaintext a password column's hash is made from (C1, C4 of r21), after NIST SP 800-63B rev 4
 * §3.1.1.2: any storable text, normalized to NFC and then counted by code point, never trimmed,
 * with no composition rule. `passwordText()` sets a password, at least 15 characters (a password
 * used only with a second factor may take `min: 8`); `passwordText.typed()` reads one typed to
 * sign in, holding only the max, so a password set before a minimum still signs in. Under
 * `hash: "bcrypt"` both refuse past the 72 UTF-8 bytes bcrypt reads, and `set` refuses a `min` past
 * them. The column itself holds the hash, which codegen holds to its declared bounds alone.
 */
import * as z from "zod";
/** The scheme a password is hashed with, where it limits what the hash reads */
export type PasswordHash = "bcrypt";
/** What a typed password is held to: the most characters, and the scheme it is hashed with */
export type TypedPasswordOptions = {
    /** The scheme the password is hashed with; `bcrypt` also refuses a password past the 72 UTF-8 bytes bcrypt reads */
    hash?: PasswordHash | undefined;
    /** The most characters, counted by code point after NFC normalization; a whole number from 1, 128 by default */
    max?: number | undefined;
};
/** What a password set is held to: a typed one's, and the fewest characters */
export type PasswordTextOptions = TypedPasswordOptions & {
    /** The fewest characters, counted as `max` is; a whole number from 0 up to `max`, 15 by default, and at most 72 under `bcrypt` */
    min?: number | undefined;
};
/** A password typed to sign in: storable, NFC, at most `max` characters (128 by default), and bcrypt's bytes under it */
declare function typed({ hash, max }?: TypedPasswordOptions): z.ZodString;
/** A password set: a typed one's rules and at least `min` characters (15 by default) */
declare function set({ hash, max, min }?: PasswordTextOptions): z.ZodString;
/** A password set, and `typed` for one typed to sign in */
export declare const passwordText: typeof set & {
    /** A password typed to sign in: storable, NFC, at most `max` characters (128 by default), and bcrypt's bytes under it; it holds no minimum, so a password set before one still signs in */
    typed: typeof typed;
};
export {};

// storable-text.d.ts
/**
 * Text every engine stores as sent. No engine does with U+0000 or a lone UTF-16 surrogate: jsonb
 * refuses both, Postgres text refuses U+0000 and takes a lone surrogate as U+FFFD, libsql cuts at
 * U+0000, and bun:sqlite reads a lone surrogate back as other text (F11). Generated write DTOs hold
 * every text to it through `isStorableText`, so the rule has one source; `storableText()` gives an
 * app's own schema the same rule (G1). A refine, never a pattern: published as one, Go's and Rust's
 * engines refuse \u and a surrogate, and a field's own pattern would move into an allOf (F56).
 */
import * as z from "zod";
/** The `x-comb-text` value a storable text schema publishes in JSON Schema, which names the rule a pattern cannot express */
export declare const STORABLE_TEXT_RULE = "no U+0000 or unpaired surrogate";
/** Whether every engine stores `value` as sent: it holds no U+0000 and no lone surrogate */
export declare function isStorableText(value: string): boolean;
/** A string schema that refuses what `isStorableText` refuses, publishing the rule as `x-comb-text` */
export declare const storableText: () => z.ZodString;

// id.d.ts
/**
 * ULID + prefixed ID generation.
 * Zero dependencies — uses Web Crypto API (CF Workers, Node, Bun, Deno).
 *
 * Format: 26 chars Crockford's Base32
 *   - First 10: 48-bit timestamp (ms since epoch)
 *   - Last 16: 80-bit randomness
 *
 * `generateUlid` / `generateId` are ordered by millisecond only: two ids from the same
 * millisecond compare at random. `createUlidFactory` keeps its last id and issues strictly
 * increasing ones, through a clock that steps back too. The state lives in the factory you
 * hold, not in this module, so it is never shared across requests unless you share it.
 */
/** The clock and the randomness a ULID factory reads, each replaceable for tests or a runtime without them */
type UlidFactoryOptions = {
    /** The clock, ms since epoch, a fraction read at its millisecond; `Date.now` by default */
    now?: (() => number) | undefined;
    /** Fills the bytes with randomness; `crypto.getRandomValues` by default */
    random?: ((bytes: Uint8Array<ArrayBuffer>) => void) | undefined;
};
/** Throw for a prefix generateId refuses: c.id checks it too, so a tables module fails where the app would (F55) */
declare function checkPrefix(prefix: string): void;
/**
 * A ULID source whose ids strictly increase: within one millisecond the random part counts up
 * by one, a clock that steps back is held at the last time issued, and a random part that cannot
 * count up any further moves on to the next millisecond. A clock reading outside ULID's milliseconds,
 * 0 to 2^48 - 1, throws a RangeError, as does a move past the last of them.
 */
declare function createUlidFactory(options?: UlidFactoryOptions): {
    /**
     * A prefixed id, `{prefix}_{ulid}`, from this factory's ULIDs; throws for a prefix that is not 2 to
     * 10 lowercase letters and digits starting with a letter
     */
    id: (prefix: string) => string;
    /** The next ULID, 26 base32 characters, greater than every one this factory issued; throws `RangeError` for a clock reading outside a ULID's time */
    ulid: () => string;
};
/** Generate a ULID (26-char, base32, ordered by millisecond) */
declare function generateUlid(): string;
/** Generate a prefixed ID: `{prefix}_{ulid}` */
declare function generateId(prefix: string): string;
export { checkPrefix, createUlidFactory, generateId, generateUlid, type UlidFactoryOptions };

// lru.d.ts
/** An entry as `getWithMeta` returns it, with whether it is past its soft TTL */
type LruMeta<V> = {
    /** True between the entry's soft and hard expiry: serve `value`, and refresh it */
    stale: boolean;
    /** The value the entry was set with */
    value: V;
};
/**
 * A least-recently-used cache with a bounded size, an optional soft TTL and an optional hard TTL,
 * for stale-while-revalidate. Past its soft TTL an entry is stale: `get` and `has` miss it, and
 * `getWithMeta` still returns it, marked stale, until its hard TTL evicts it. With a hard TTL, an
 * entry's soft TTL is at most the hard one, and an entry with none is fresh until the hard expiry.
 * A `set` on a full cache evicts the least recently read or set entry. Times are read from
 * `Date.now()`.
 */
declare class LruCache<K, V> {
    private cache;
    private readonly hardTtlMs;
    private readonly maxSize;
    private readonly ttlMs;
    /**
     * An empty cache. Throws `RangeError` for a `maxSize` that is not a positive integer, a TTL that
     * is not a positive finite number, or a `hardTtlMs` shorter than `ttlMs`.
     */
    constructor(options: {
        /** Milliseconds after a set when the entry is evicted, whatever reads it; none when omitted */
        hardTtlMs?: number | undefined;
        /** The most entries the cache holds, a positive integer */
        maxSize: number;
        /** Milliseconds after a set when the entry turns stale, unless `set` is given its own; never when omitted and no `hardTtlMs` is set */
        ttlMs?: number | undefined;
    });
    /**
     * Get value from cache.
     * Returns undefined if not found or past soft expiry.
     * Updates access order (LRU).
     */
    get(key: K): V | undefined;
    /**
     * Get value with staleness metadata for SWR pattern.
     * - Before softExpiry: { value, stale: false }
     * - Between soft and hardExpiry: { value, stale: true }
     * - After hardExpiry or not found: undefined
     *
     * When hardTtlMs not configured, behaves like get() wrapped in meta.
     */
    getWithMeta(key: K): LruMeta<V> | undefined;
    /**
     * Set value in cache.
     * Evicts oldest entry if at max size.
     * @param ttlMs - Optional per-key soft TTL override
     */
    set(key: K, value: V, ttlMs?: number): void;
    /** Delete key from cache */
    delete(key: K): boolean;
    /** Clear all entries */
    clear(): void;
    /** Get current cache size */
    get size(): number;
    /** Check if key exists and is not past soft expiry */
    has(key: K): boolean;
}
export { LruCache, type LruMeta };

// constraint-map.d.ts
/**
 * Looking a constraint violation up in the app's ConstraintMaps, shared by the SQLite and the
 * Postgres handlers: each dialect finds the table, column or constraint name its own way, and
 * both turn what they found into the same CombError.
 */
import { CombError } from "./error.ts";
import type { CombDatabaseErrorKey, ConstraintMap, ConstraintMapEntry, StatusKey } from "./types.ts";
export type TableMap = NonNullable<ConstraintMap[string]>;
/** The generic error for a kind of violation, used when no map entry matches. */
export type Fallback = {
    /** the message safe to show the client, which the CombError carries as its `cause` */
    cause: string;
    /** the key a client branches on */
    errorKey: CombDatabaseErrorKey;
    /** the HTTP status the error answers with, by key: `conflict` is 409, `service_unavailable` 503 */
    status: StatusKey;
};
/** What a handler found: the map entry, if any, the generic error otherwise, and where. */
export type Found = {
    column?: string | undefined;
    entry: ConstraintMapEntry | undefined;
    fallback: Fallback;
    table?: string | undefined;
};
/** `record[key]` when `key` is the record's own (never an inherited name such as `constructor`). */
export declare const own: <T>(record: Record<string, T> | undefined, key: string | undefined) => T | undefined;
/**
 * The entry `map` holds for `table`: its own key's, else that of the longest prefix pattern it
 * starts with, a key ending in `*` (`"ut_*"`), which keys the tables an app makes at run time and
 * no generated map can name (R5 of plan 17). A key with a `*` anywhere else matches nothing
 */
export declare function entryOf(map: ConstraintMap, table: string | undefined): TableMap | undefined;
/**
 * The maps an app passes its error handlers, every key checked where it builds them, at module load:
 * a TypeError naming a key with a `*` anywhere but its end, never a lookup at the first violation.
 * The array comes back frozen, and no handler checks it again.
 */
export declare function defineConstraintMaps(...maps: ConstraintMap[]): readonly ConstraintMap[];
/**
 * `maps`, a handler's first look at each map reporting its malformed keys to the operator, through the
 * warn sink, each key once per process however many arrays or maps carry it: the violation is still classified with the well-formed
 * keys, as the client's answer must not change, nor say anything of the app's configuration (Q4 of r19)
 */
export declare function checkedOnce(maps: readonly ConstraintMap[]): readonly ConstraintMap[];
/**
 * The columns an error may name of those a violation reports: all of them, or none when a map lists
 * one as the table's private column, since naming the rest would point at it (F67), or when the
 * maps given key no such table (fail closed)
 */
export declare function shown(maps: readonly ConstraintMap[], table: string | undefined, columns: readonly string[]): readonly string[];
/** `column`, or undefined when a map lists it among `table`'s private columns (F67), or shown hides the table's */
export declare const shownColumn: (maps: readonly ConstraintMap[], table: string | undefined, column: string | undefined) => string | undefined;
/** The first entry `pick` finds for `table` across `maps`. */
export declare function lookup(maps: readonly ConstraintMap[], table: string | undefined, pick: (t: TableMap) => ConstraintMapEntry | undefined): ConstraintMapEntry | undefined;
/** The `kind` entry named `name` (a CHECK, or a unique index) in any table, with that table. */
export declare function findNamed(maps: readonly ConstraintMap[], kind: "check" | "unique", name: string | undefined): {
    entry: ConstraintMapEntry;
    table: string;
} | undefined;
/** The CombError for what a handler found; `source` is the original error, kept for logs. */
export declare function toCombError({ column, entry, fallback, table }: Found, source: unknown): CombError;

/** Constraint map entry for mapping DB constraint violations to domain errors */

    cause: string;
    /** The `CombError.errorKey` the violation answers with, your domain's key */
    errorKey: string;
    /** The status the violation answers with; `bad_request` for a violation comb reports as one, such as a CHECK, and `conflict` otherwise */
    statusKey?: "bad_request" | "conflict" | "forbidden" | undefined;
};
/** Per-table constraint violation mappings */
export type ConstraintMap = {
    /**
     * A table's constraints, keyed by its SQL name; a key ending in `*` is a prefix pattern for the
     * tables an app makes at run time (`"ut_*"`). A table's own key wins over a pattern, and the
     * longest pattern over a shorter one.
     */
    [table: string]: {
        /** CHECK constraints, by constraint name */
        check?: Record<string, ConstraintMapEntry> | undefined;
        /** Postgres EXCLUDE constraints, by name */
        exclusion?: Record<string, ConstraintMapEntry> | undefined;
        /**
         * Foreign keys, by the SQL name of each one's first column. SQLite and D1 name no foreign key
         * in their message, so there an entry is blamed only when it is the table's one listed key
         * and the statement sets that column; list every foreign key of the table, as codegen does.
         */
        foreignKey?: Record<string, ConstraintMapEntry> | undefined;
        /**
         * The partitions of this Postgres table, keyed as the map keys a table: Postgres reports a
         * violation on the partition, which is read as this table's, under its keys and its `private`
         * list. Codegen writes it from comb.config.ts's `partitions`; a table no entry lists or keys
         * names no column.
         */
        partitions?: readonly string[] | undefined;
        /** The primary key's violation, a duplicate id */
        primaryKey?: ConstraintMapEntry | undefined;
        /**
         * The columns of this table the client never sees, by SQL name: an error names none of them,
         * in its `column` or its cause. Codegen writes it; a map without it names every column.
         */
        private?: readonly string[] | undefined;
        /**
         * The columns of this table some foreign key points at, by SQL name: an UPDATE, or an
         * upsert's DO UPDATE, that assigns one may have failed on a row referencing the old value,
         * so SQLite's handler answers it with the neutral cause. Codegen writes it.
         */
        referenced?: readonly string[] | undefined;
        /** Unique constraints, by their columns' SQL names comma-joined (`org_id,slug`); a unique index on an expression by its index name */
        unique?: Record<string, ConstraintMapEntry> | undefined;
    };
};
/** Function signature for database error handlers */
export type DatabaseErrorHandler = (error: unknown, constraintMaps: readonly ConstraintMap[]) => void;

/**
 * The comb meta contract — facts about tables and queries, stamped onto Zod
 * schemas so they survive the trip to JSON Schema and into an OpenAPI document.
 *
 * comb publishes facts in comb's vocabulary. Mapping them to a consumer's tag
 * names is the consumer's job. See docs/meta-contract.md.
 */
import type { PartialUniqueIndexMeta, UniqueIndexMeta } from "../codegen/analyzer-types.ts";
/** Reserved key. One collision surface, one version field. */
declare const COMB_META_KEY = "x-comb";
/**
 * Contract version. Bumped ONLY on a breaking change — a field removed, or a
 * field whose meaning changed. Adding a field never bumps it, which is what
 * lets comb and its consumers ship independently.
 */
declare const COMB_META_VERSION = 1;
/** Which payload a stamp carries: facts about a table (`entity`) or about a list query (`query`) */
type CombMetaKind = "entity" | "query";
/** Filter grammar comb's parser implements — `field.op.value`. */
declare const COMB_FILTER_GRAMMAR = "postgrest";
/** The filter grammar a query meta names, `postgrest`: `field.op.value` */
type CombFilterGrammar = typeof COMB_FILTER_GRAMMAR;
/** Facts about an entity, derived from its table definition. */
type CombEntityMeta = {
    /** The contract version the payload was stamped with, `COMB_META_VERSION` */
    v: number;
    /** Always `entity` */
    kind: "entity";
    /** Entity name — the SQL table name */
    name: string;
    /** Primary key / stable identifier column */
    identity: string;
    /** Server-assigned: omitted from create bodies, expected in responses */
    generated: string[];
    /** Must reject or ignore a mutation */
    immutable: string[];
    /** Tombstone column, or null when the entity is hard-deleted */
    softDelete: string | null;
    /** Column scoping rows to a tenant, or null when comb cannot know (see docs §6.3) */
    tenantColumn: string | null;
    /**
     * Lifecycle of the entity's state column, or null when none is declared.
     *
     * `transitions` is deliberately absent: it is a write-side rule, and
     * publishing the graph would invite a consumer to drive state by walking it.
     * See docs/state-machines.md §5.
     */
    states: CombStatesMeta | null;
    /**
     * The uniqueness that holds over every row: a `uniqueIndex()`, a `unique().on()`
     * constraint and a column's `.unique()`, in the order the analyzer extracted
     * them. Empty when none. Never inferred from PKs or column names, and never a
     * partial index, which `partialUniqueIndexes` holds.
     */
    uniqueIndexes: UniqueIndexMeta[];
    /**
     * Partial (`.where()`) unique indexes: their columns are unique only among the
     * rows `where` matches. Apart from `uniqueIndexes`, so a reader that knows no
     * predicate claims no uniqueness (docs §6.8). Added within v1, so a producer
     * may leave it out; readCombEntityMeta reads its absence as [].
     */
    partialUniqueIndexes?: PartialUniqueIndexMeta[] | undefined;
};
/** The published subset of a column state machine — describes a lifecycle, not how to drive it. */
type CombStatesMeta = {
    /** The column the machine governs */
    column: string;
    /** Every legal value, in declaration order */
    values: string[];
    /** The value a row starts at, or null when the machine declares none */
    initial: string | null;
    /** The values no transition may leave; empty when none are declared */
    terminal: string[];
};
/** One filterable field with the operators its type accepts. */
type CombFilterField = {
    /** The field's name, as a filter names it */
    field: string;
    /** The field's filter type: `string`, `number`, `boolean`, `date` or `enum` */
    type: string;
    /** The operators a filter may apply to the field, such as `eq` or `in` */
    ops: string[];
};
/**
 * Facts about a list query, derived from the config that parses the request.
 * Fields marked optional were added after v1 shipped; a v1 reader built before
 * them ignores them, which is why adding them did not bump `v`.
 */
type CombQueryMeta = {
    /** The contract version the payload was stamped with, `COMB_META_VERSION` */
    v: number;
    /** Always `query` */
    kind: "query";
    /** The fields a `filter` may name */
    filterable: string[];
    /** `filterable` with types and operators */
    filterFields?: CombFilterField[] | undefined;
    /** The fields an `order` may name */
    sortable: string[];
    /** Null placements every sortable field accepts */
    nulls?: Array<"first" | "last"> | undefined;
    /** Conditions one `filter` may hold */
    maxFilterConditions?: number | undefined;
    /** Nested `or()` and `and()` groups one `filter` may hold below its top level */
    maxFilterDepth?: number | undefined;
    /** Values one filter binds in all, every in/nin member counted */
    maxFilterValues?: number | undefined;
    /** Values one `in` or `nin` list may hold */
    maxInValues?: number | undefined;
    /**
     * UTF-8 bytes of the escaped pattern of a `like` or `ilike` value with a `*` inside it, as SQLite
     * reads it: a `[` in a `like` is the three bytes `[[]`, a `%` or `_` in an `ilike` two. A value whose
     * only `*` are at its ends is no pattern and takes no cap. Not the raw value's characters.
     */
    maxPatternBytes?: number | undefined;
    /** Bytes one `filter` may take however a client encodes it, beside its character cap */
    maxFilterBytes?: number | undefined;
    /** Bytes the whole query string comb accepts takes at most, however a client encodes it: 16 KB less a 2 KB path budget */
    maxQueryBytes?: number | undefined;
    /** Whitespace-separated words one `q` may hold */
    maxSearchTerms?: number | undefined;
    /** Bytes one `q` may take however a client encodes it */
    maxSearchBytes?: number | undefined;
    /** Keys one `order` may hold */
    maxSortKeys?: number | undefined;
    /** null = not knowable at this layer, which is not the same as "none" (docs §6.1) */
    searchable: string[] | null;
    /** The fields a `select` may name */
    selectable: string[];
    /** The largest `limit` a request may ask for */
    maxLimit: number;
    /** The last page offset pagination serves; past it, a client pages by cursor */
    maxPage?: number | undefined;
    /** e.g. "created_at.desc" — the order applied when the request names none */
    defaultOrder: string;
    /** Tiebreak column for keyset pagination */
    stableTiebreak: string;
    /** The filter grammar `filter` is parsed with */
    grammar: CombFilterGrammar;
};
/** A comb meta payload: an entity's or a list query's, told apart by `kind` */
type CombMeta = CombEntityMeta | CombQueryMeta;
/** Payload accepted by combMeta() — the version is stamped for you. */
type CombEntityMetaInput = Omit<CombEntityMeta, "v">;
/** A query payload as combMeta() takes it, without `v` */
type CombQueryMetaInput = Omit<CombQueryMeta, "v">;
/** A payload as combMeta() takes it, entity or query, without `v` */
type CombMetaInput = CombEntityMetaInput | CombQueryMetaInput;
/** The object combMeta() produces, ready for `.meta()`. */
type CombMetaStamp<T extends CombMeta = CombMeta> = {
    [COMB_META_KEY]: T;
};
export { COMB_FILTER_GRAMMAR, COMB_META_KEY, COMB_META_VERSION, type CombEntityMeta, type CombEntityMetaInput, type CombFilterField, type CombFilterGrammar, type CombMeta, type CombMetaInput, type CombMetaKind, type CombMetaStamp, type CombQueryMeta, type CombQueryMetaInput, type CombStatesMeta, type PartialUniqueIndexMeta, type UniqueIndexMeta, };

/**
 * Reading comb's facts off a JSON Schema. Returns null, never throws: a refused or malformed
 * payload degrades to "no facts", never to a confident wrong answer. See docs/meta-contract.md §4.
 */
import { isRecord } from "./read-indexes.ts";
import { type CombEntityMeta, type CombMeta, type CombQueryMeta } from "./contract.ts";
/** How readCombMeta, readCombEntityMeta and readCombQueryMeta read a payload */
type ReadOptions = {
    /**
     * Highest contract version this reader understands. A payload newer than
     * this is refused rather than partially read — see docs §4.2.
     * Defaults to the version of the comb build doing the reading.
     */
    maxVersion?: number | undefined;
    /** Called instead of throwing when a payload is refused or malformed. */
    onDiagnostic?: ((message: string) => void) | undefined;
};
/**
 * Read comb's facts off the ROOT of a JSON Schema.
 *
 * Does not unwrap `items` for array schemas — the consumer that already
 * resolves which schema to read owns that step, so the two cannot disagree.
 *
 * Returns null, never throws: a refused or malformed payload must degrade to
 * "no facts" (which downstream handles as a visible coverage gap), not to a
 * confident wrong answer.
 */
declare function readCombMeta(jsonSchema: unknown, options?: ReadOptions): CombMeta | null;
/** Narrowing convenience — null when absent, refused, or the other kind. */
declare function readCombEntityMeta(jsonSchema: unknown, options?: ReadOptions): CombEntityMeta | null;
/** Narrowing convenience — null when absent, refused, or the other kind. */
declare function readCombQueryMeta(jsonSchema: unknown, options?: ReadOptions): CombQueryMeta | null;
export { isRecord, type ReadOptions, readCombEntityMeta, readCombMeta, readCombQueryMeta };
