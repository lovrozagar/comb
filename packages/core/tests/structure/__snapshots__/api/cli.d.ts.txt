// cli/output.d.ts
import type { Run } from "../migrate/core/process.ts";
import type { LiveRead } from "../migrate/core/pg-live-types.ts";
/** Where the CLI reads its working directory, writes its output, asks, and starts processes. */
export type CliIo = {
    cwd: string;
    out(line: string): void;
    err(line: string): void;
    run?: Run | undefined;
    /** How a Postgres baseline reads the database's catalog of what Atlas OSS does not compare, postgres.js unless a test fakes it (5.4b) */
    readPgObjects?: LiveRead | undefined;
    /** How a diff, baseline or adoption gives the dev database a non-public target schema, comb's own unless a test fakes it (P12) */
    readyPgDev?: ((target: {
        devUrl: string;
        schema: string;
    }) => Promise<void>) | undefined;
    /** Ask the person running the CLI a yes/no question; absent when nobody can answer (no TTY) */
    confirm?: ((question: string) => Promise<boolean>) | undefined;
};
/** What `comb migrate check` reports once every pending migration is approved */
/** What a migrate apply, check or status found: the files it applied, or found applied, and those pending */
export type MigrateOutcome = {
    applied: string[];
    pending: string[];
};
export declare const checkPassed: (names: readonly string[]) => string;
/** A failure the CLI reports as its message and exit code instead of a stack trace. */
export declare class CliError extends Error {
    readonly exitCode: number;
    constructor(message: string, exitCode?: number);
}
/** A failure a command already reported as its --json object: printed on stderr, never as a second object (K10 of r18) */
export declare class Reported extends CliError {
    constructor(error: unknown);
}
/** A request for help: the CLI prints `text` on out and exits 0, wherever in argv -h or --help came */
export declare class ShowHelp extends Error {
    readonly text: string;
    constructor(text: string);
}
type Terminal = {
    input: NodeJS.ReadableStream & {
        isTTY?: boolean | undefined;
    };
    output: NodeJS.WritableStream;
};
/**
 * The IO the comb binary runs with: the console, and questions on the terminal when there is one,
 * asked on stderr: a question is a human line, as every other --json sends there, so stdout stays
 * what the command prints (P4 of the 0.15.0 review)
 */
export declare function processIo({ input, output }?: Partial<Terminal>): CliIo;
export {};

/** The help, from the declarations the commands parse with */
export declare const help: () => string;

// migrate/core/process.d.ts
export type RunResult = {
    code: number;
    stdout: string;
    stderr: string;
};
export type RunOptions = {
    cwd?: string | undefined;
    /** Variables set over the parent's environment for the child; an undefined value removes one. */
    env?: Readonly<Record<string, string | undefined>> | undefined;
    /** The child's whole environment, none of the parent's, in place of `env` */
    exactEnv?: Readonly<Record<string, string>> | undefined;
    input?: string | undefined;
    /** Bytes of stdout or stderr kept before the process is stopped; default 256 MiB. */
    maxBuffer?: number | undefined;
    /** Milliseconds before the process is stopped, which answers 124 as timeout(1) does; default none */
    timeoutMs?: number | undefined;
};
export type Run = (cmd: string, args: readonly string[], options?: RunOptions) => RunResult;
/** Runs `cmd` synchronously and reports how it ended; it never throws for a failed command. */
export declare const run: Run;

// migrate/core/pg-live-types.d.ts
/** The shape of a Postgres baseline's catalog read (pg-live-objects.ts), apart so the CLI's io names it without its code. */
/**
 * What a database holds of what Atlas OSS will not compare: each routine by name, its kind and each
 * overload's signature (its input argument types); each table's triggers by name, with when and for
 * what they fire (tgtype) and whether they do (P13); its views, materialized views, sequences and
 * policies by name
 */
export type LiveObjects = {
    /** each extension, by name, but plpgsql (P10 of r18) */
    extensions: Set<string>;
    /** each domain, composite and range type and collation, by name, with what it is (P10 of r18) */
    kinds: Map<string, "collation" | "composite type" | "domain" | "range type">;
    /** each table's policies, by name, with their command, roles and kind: `for all, to public, permissive` (P9, P10) */
    policies: Map<string, Map<string, string>>;
    /** each view, materialized view and sequence no column owns, by name, with what it is (P9) */
    relations: Map<string, "materialized view" | "sequence" | "view">;
    /** each table's row level security where it is on: `enabled, not forced` (P10 of r18) */
    rls: Map<string, string>;
    routines: Map<string, {
        kind: "function" | "procedure";
        signatures: Set<string>;
    }>;
    /** each table's rules, by name, but a view's own (P10 of r18) */
    rules: Map<string, Set<string>>;
    /** each sequence's settings: `start 1, increment 1, min 1, max …, no cycle` (P10 of r18) */
    sequences: Map<string, string>;
    triggers: Map<string, Map<string, {
        enabled: boolean;
        type: number;
    }>>;
    /** the role the read connects as: the one the migrations ran as, which CURRENT_USER names in a policy (R2 of plan 17) */
    user: string;
};
/** The database a read reaches, and the schema it reads */
export type LiveTarget = {
    schema: string;
    url: string;
};
/** How a baseline reads them: postgres.js, unless a test fakes it */
export type LiveRead = (target: LiveTarget) => Promise<LiveObjects>;
