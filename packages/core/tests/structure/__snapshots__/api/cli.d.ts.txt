/** Where the CLI reads its working directory, writes its output, asks, and starts processes. */
export type CliIo = {
    cwd: string;
    out(line: string): void;
    err(line: string): void;
    run?: Run | undefined;
    /** How a Postgres baseline reads the database's catalog of what Atlas OSS does not compare, postgres.js unless a test fakes it (5.4b) */
    readPgObjects?: LiveRead | undefined;
    /** How a Postgres baseline reads the history replayed on its dev database, comb's own (pg-replay.ts) unless a test fakes it (P5 of r19) */
    replayPgHistory?: typeof replayCatalog | undefined;
    /** How a diff, baseline or adoption gives the dev database a non-public target schema, comb's own unless a test fakes it (P12) */
    readyPgDev?: ((target: {
        devUrl: string;
        schema: string;
    }) => Promise<void>) | undefined;
    /** Ask the person running the CLI a yes/no question; absent when nobody can answer (no TTY) */
    confirm?: ((question: string) => Promise<boolean>) | undefined;
};
/** What `comb migrate check` reports once every pending migration is approved */

export type MigrateOutcome = {

    pending: string[];
};
export declare const checkPassed: (names: readonly string[]) => string;

    readonly exitCode: number;

}
/** A failure a command already reported as its --json object: printed on stderr, never as a second object (K10 of r18) */
export declare class Reported extends CliError {

}
/** A request for help: the CLI prints `text` on out and exits 0, wherever in argv -h or --help came */
export declare class ShowHelp extends Error {
    readonly text: string;
    constructor(text: string);
}
type Terminal = {
    input: NodeJS.ReadableStream & {

    };
    output: NodeJS.WritableStream;

/**
 * The IO the comb binary runs with: the console, and questions on the terminal when there is one,

 * what the command prints (P4 of the 0.15.0 review)
 */

export type RunResult = {
    code: number;
    stdout: string;
    stderr: string;
};
export type RunOptions = {
    cwd?: string | undefined;
    /** Variables set over the parent's environment for the child; an undefined value removes one. */
    env?: Readonly<Record<string, string | undefined>> | undefined;

    exactEnv?: Readonly<Record<string, string>> | undefined;
    input?: string | undefined;

    maxBuffer?: number | undefined;

    timeoutMs?: number | undefined;
};

// migrate/core/pg-live-types.d.ts
/** The shape of a Postgres baseline's catalog read (pg-live-objects.ts), apart so the CLI's io names it without its code. */
/**
 * What a database holds of what Atlas OSS will not compare: each routine by name, its kind and each
 * overload's signature (its input argument types); each table's triggers by name, with when and for
 * what they fire (tgtype) and whether they do (P13); its views, materialized views, sequences and
 * policies by name
 */
export type LiveObjects = {
    /** each extension, by name, but plpgsql (P10 of r18) */
    extensions: Set<string>;
    /** each domain, composite and range type and collation, by name, with what it is (P10 of r18) */
    kinds: Map<string, "collation" | "composite type" | "domain" | "range type">;
    /** each table's policies, by name, with their command, roles and kind: `for all, to public, permissive` (P9, P10) */
    policies: Map<string, Map<string, string>>;
    /** each view, materialized view and sequence no column owns, by name, with what it is (P9) */
    relations: Map<string, "materialized view" | "sequence" | "view">;
    /** each table's row level security where it is on: `enabled, not forced` (P10 of r18) */
    rls: Map<string, string>;
    routines: Map<string, {
        kind: "function" | "procedure";
        signatures: Set<string>;
    }>;
    /** each table's rules, by name, but a view's own (P10 of r18) */
    rules: Map<string, Set<string>>;
    /** each sequence's settings: `start 1, increment 1, min 1, max …, no cycle` (P10 of r18) */
    sequences: Map<string, string>;
    triggers: Map<string, Map<string, {
        enabled: boolean;
        type: number;
    }>>;
    /** the role the read connects as: the one the migrations ran as, which CURRENT_USER names in a policy (R2 of plan 17) */
    user: string;
};
/** The database a read reaches, and the schema it reads */
export type LiveTarget = {
    schema: string;
    url: string;
};
/** How a baseline reads them: postgres.js, unless a test fakes it */
export type LiveRead = (target: LiveTarget) => Promise<LiveObjects>;

// migrate/core/pg-replay.d.ts
import type { PgTarget } from "./atlas-target.ts";
import { type Connect } from "./pg-live-objects.ts";
import type { LiveObjects } from "./pg-live-types.ts";
type Node = Record<string, unknown>;
/** A history file as baseline reads it */
type HistoryFile = {
    name: string;
    sql: string;
};
/**
 * `sql` without its one CONCURRENTLY keyword: the code tokens are read by comb's scanner, so the word
 * in a string, a comment or a quoted identifier is never touched, and the reparse proves only the flag changed
 */
declare function cutConcurrently(sql: string, tree: unknown): string;
/** The statements the replay sends for `files`, in order, each with the file it is from */
declare function replayStatements(files: readonly HistoryFile[]): Array<{
    file: string;
    sql: string;
}>;
/**
 * The catalog the history `files` leave in `target`'s schema, replayed on its dev database in one
 * transaction that is rolled back, read by the target's own query (readCatalog). The role the replay
 * ran as stands for the target's role wherever the read names it (`targetUser`), as CURRENT_USER
 * names the role each database was built as
 */
declare function replayCatalog(target: PgTarget, { files, targetUser }: {
    files: readonly HistoryFile[];
    targetUser: string;
}, load?: () => Promise<Connect>): Promise<LiveObjects>;
/** Whether a statement's node makes what a catalog arm reads */
type Makes = (node: Node) => boolean;
/**
 * Each arm of the catalog read (pg-live-catalog.ts), by its `what`, with the statement nodes that make
 * what it reads, held to the read's arms by a structure test so an arm cannot be added without its
 * gate (R1 of plan 18's stack 2). `user` is the role the read connects as, which nothing makes. A
 * sequence a column owns (serial, identity) is the table's, left out of the read, so no arm needs it
 */
declare const ARM_MAKERS: Readonly<Record<string, Readonly<Record<string, Makes>>>>;
/**
 * The statements whose effect the tree cannot see, which may make anything (R1 of stack 2): a DO
 * block, a procedure CALL, a prepared statement's EXECUTE
 */
declare const UNSEEN: ReadonlySet<string>;
/**
 * Whether `files` make anything the catalog read compares, which a docker:// dev database, Atlas's
 * own container, keeps out of comb's reach: by parse-tree node, read once. A statement the tree cannot
 * see the effect of counts; a function the history defines, which a later call may run DDL through, is
 * itself compared, so its CREATE already counts
 */
declare function makesCompared(files: readonly HistoryFile[]): boolean;
export { ARM_MAKERS, cutConcurrently, type HistoryFile, makesCompared, replayCatalog, replayStatements, UNSEEN };

// migrate/core/atlas-target.d.ts
/** A Postgres database as Atlas reaches it: target, dev database, schema and lock wait */
type PgTarget = {
    devUrl: string;
    lockTimeout: string;
    schema: string;
    url: string;
};
/** What an Atlas call reads from its config: the target, the dev database, the migrations dir, and whose history */
type AtlasNeeds = {
    dev?: boolean | undefined;
    dir?: string | undefined;
    /** the target unscoped, for the one call that lists the database's schemas (M78) */
    allSchemas?: boolean | undefined;
    /** a history other than the target's own, by its schema: the shared one 0.12 kept every schema's in (M77), or another schema's own (M78) */
    revisions?: string | undefined;
    url?: boolean | undefined;
};
/** The flags and environment an Atlas call runs with, its scratch directory, and the scrubber for its output */
type AtlasEnv = {
    args: readonly string[];
    dir: string;
    env: Readonly<Record<string, string>>;
    /** free text: the call's secrets, and any other URL's password or secret query value, masked */
    scrub(text: string): string;
};
/** What is wrong with the URLs `needs` names before Atlas runs: errors refuse it, warnings are printed */
declare function pgUrlProblems(target: PgTarget, needs?: AtlasNeeds): {
    errors: string[];
    warnings: string[];
};
/** The schema Atlas keeps public's history in, as it always has */
declare const SHARED_REVISIONS = "atlas_schema_revisions";
/**
 * The schema Atlas keeps `target`'s history in: the shared one for public, and one of its own for
 * any other scope, so two schemas of one database never read each other's history. Postgres cuts a
 * name at 63 bytes, so a scope too long, or not a plain lower-case name, is named by its hash.
 */
declare function revisionsSchema(target: PgTarget): string;
/** Both URLs as Atlas gets them */
declare const atlasUrls: (target: PgTarget) => {
    devUrl: string;
    url: string;
};
/** The atlas.hcl for `needs`: getenv for each URL, and the history pinned to `revisions` */
declare function atlasConfig(needs: AtlasNeeds, revisions: string): string;
/**
 * Run `body` with an Atlas config for `target` and `needs` in a fresh 0700 directory, removed
 * afterwards however `body` ends. A target with errors is refused before anything is written.
 */
declare function withAtlasEnv<T>(target: PgTarget, needs: AtlasNeeds, body: (atlas: AtlasEnv) => T): T;
/**
 * What Atlas said on stderr as it succeeded, a line each: comb passes each on as a warning, as Atlas
 * OSS may say it left part of a schema out of what it compared (plan 14's 5.4)
 */
declare const atlasNotices: (stderr: string) => string[];
export { atlasConfig, type AtlasEnv, type AtlasNeeds, atlasNotices, atlasUrls, pgUrlProblems, type PgTarget, revisionsSchema, SHARED_REVISIONS, withAtlasEnv, };

// migrate/core/pg-live-objects.d.ts
import type { LiveObjects, LiveTarget } from "./pg-live-types.ts";
/** The postgres.js client, as far as comb uses it */
type Sql = {
    end(): Promise<void>;
    unsafe(query: string, params: unknown[]): Promise<Array<Record<string, unknown>>>;
};
type Connect = (options: Record<string, unknown>) => Sql;
/**
 * What a postgres.js failure says: its message, or, for the AggregateError it throws with none when
 * every address refuses, each address's answer joined (P15 of the 0.15.0 review)
 */
declare function failureOf(error: unknown, field?: "devUrl" | "url"): string;
/** postgres.js, or the refusal naming it and `why` comb needs it here */
declare function loadPostgres(why: string): Promise<Connect>;
/** What a baseline compares itself, as its refusals name it */
declare const COMPARED = "functions, triggers, views, sequences, policies, types, rules, collations, extensions and row level security";
/**
 * What `schema` holds of what Atlas OSS does not compare, read through `sql` by the one catalog
 * query: the target's read and the dev replay's are this one function (P5 of r19), so the two sides
 * cannot be read two ways
 */
declare function readCatalog(sql: Pick<Sql, "unsafe">, schema: string): Promise<LiveObjects>;
/** What the target schema of `target`'s database holds of what Atlas OSS does not compare; it writes nothing */
declare function readLiveObjects(target: LiveTarget, load?: () => Promise<Connect>): Promise<LiveObjects>;
/**
 * What the database and the history disagree on, both read by readCatalog (P5 of r19), the history's
 * from its replay on dev: each object only one of them has, by name and signature or how a trigger
 * fires, and how many of each kind they share, whose bodies neither read
 */
declare function liveDrift(history: LiveObjects, live: LiveObjects): {
    matched: {
        routines: number;
        triggers: number;
        views: number;
    };
    only: string[];
};
export { COMPARED, type Connect, failureOf, liveDrift, loadPostgres, readCatalog, readLiveObjects };
