/** Where the CLI reads its working directory, writes its output, asks, and starts processes. */
export type CliIo = {
    /** The folder the CLI finds comb.config.ts, comb.migrate.config.ts and their files in, and resolves relative paths against */
    cwd: string;
    /** Writes one line of a command's output, with no trailing newline; with --json, the command's one JSON object */
    out(line: string): void;
    /** Writes one line of diagnostics, a refusal or a question's context, with no trailing newline; never part of --json output */
    err(line: string): void;
    /** How the CLI starts a process (atlas, wrangler, drizzle-kit): comb's own, which spawns it and resolves with its exit code and output, unless the embedder supplies one */
    run?: Run | undefined;
    /** Ask the person running the CLI a yes/no question; absent when nobody can answer (no TTY) */
    confirm?: ((question: string) => Promise<boolean>) | undefined;
};
/**
 * What a command reads besides its io, comb's own unless its tests fake one (KT7 of r22b): typed by
 * comb's internals, so they stay off the public CliIo and reach the CLI through runCliWith alone.
 */
export type CliDeps = {
    /** How a Postgres baseline reads the database's catalog of what Atlas OSS does not compare, postgres.js unless a test fakes it (5.4b) */
    readPgObjects?: LiveRead | undefined;
    /** How a Postgres baseline reads the history replayed on its dev database, comb's own (pg-replay.ts) unless a test fakes it (P5 of r19) */
    replayPgHistory?: typeof replayCatalog | undefined;
    /** How a diff, baseline or adoption gives the dev database a non-public target schema, comb's own unless a test fakes it (P12) */
    readyPgDev?: ((target: {
        devUrl: string;
        schema: string;
    }) => Promise<void>) | undefined;
    /** How a migrate command finds an optional peer's entry before its first I/O, as comb and `bun` find it unless a test fakes it (K1 of r21) */
    locatePeer?: ((peer: string) => string) | undefined;
    /** How a migrate command reads Atlas's version before its first I/O, `atlas version` through `run` unless a test fakes it (PA5 of r22) */
    readAtlasVersion?: (() => Promise<RunResult>) | undefined;
};
/** The io a command runs on, with the seams runCliWith was given */
export type CliContext = CliIo & CliDeps;

/** An optional peer comb spawns rather than imports */
export type BinPeer = "drizzle-kit" | "wrangler";
/**
 * A bin peer as the project holds it: its package.json, the bin file it names for the peer, and the
 * runtime that runs it, `node` or the executable running comb (process.execPath)
 */
export type ProjectBin = {
    bin: string;
    manifest: string;
    peer: BinPeer;
    runtime: string;
};
/** How a process ended, as `Run` resolves it */
export type RunResult = {
    /** The exit code: 0 on success, 128 plus the signal's number when a signal ended it, 124 past `timeoutMs`, 1 past `maxBuffer`, 127 when the command is not found, 126 when it could not start */
    code: number;
    /** Everything the process wrote to stdout, as UTF-8 */
    stdout: string;
    /** Everything the process wrote to stderr, as UTF-8; on a failure comb made or saw, such as a timeout or a signal, it says what happened */
    stderr: string;
};
/**
 * The Atlas releases comb runs on, which comb reads from `atlas version` before a command's first
 * I/O and refuses outside it (PA5 of r22): the floor is the oldest release comb's integration suite
 * passes on, and the next major may change what comb reads from Atlas
 */
export declare const ATLAS_RANGE = ">=1.2.0 <2";
/** Where Atlas, which no package ships, is installed from */
export declare const ATLAS_INSTALL = "https://atlasgo.io/getting-started";
/** How `Run` starts a process; every field is optional */
export type RunOptions = {
    /** The folder the process starts in; the parent's when omitted */
    cwd?: string | undefined;
    /** Variables set over the parent's environment for the child; an undefined value removes one. */
    env?: Readonly<Record<string, string | undefined>> | undefined;
    /** The child's whole environment, none of the parent's, in place of `env` */
    exactEnv?: Readonly<Record<string, string>> | undefined;
    /** Bytes of stdout or stderr kept before the process is stopped; default 256 MiB. */
    maxBuffer?: number | undefined;
    /** The bin peer the process runs, whose runtime a failure to start names */
    peer?: BinPeer | undefined;
    /** Milliseconds before the process is stopped, which answers 124 as timeout(1) does; default none */
    timeoutMs?: number | undefined;
    /** Milliseconds a process comb stopped gets to end before SIGKILL, as timeout(1)'s -k gives it; default 10 s */
    killAfterMs?: number | undefined;
} & ({
    /** Text written to the child's stdin, which is then closed; closed with nothing written when omitted */
    input?: string | undefined;
    /** False or omitted: the child's stdin is closed once `input` is written */
    lifeline?: false | undefined;
} | {
    /** No `input` with a lifeline: the child's stdin is never written */
    input?: undefined;
    /**
     * The child's stdin left open and never written while comb runs, so its end tells the child
     * comb is gone, however comb ended, a SIGKILL included
     */
    lifeline: true;
});
/**
 * How comb starts a process (atlas, wrangler, drizzle-kit): `cmd` with `args`, no shell. It resolves
 * with the process's end, a failed or unstartable command among them, and never rejects for one.
 */
export type Run = (cmd: string, args: readonly string[], options?: RunOptions) => Promise<RunResult>;
/** Runs `cmd` and reports how it ended; it never rejects for a failed command. Started while a signal stops comb, it waits with the rest. */
export declare const run: Run;
/**
 * What runs `bin`'s peer through `exec`: its runtime on the bin file of the package the project
 * holds, the one comb checked (K1 of r22), never a name looked up on PATH
 */
export declare const runPeer: (exec: Run, bin: ProjectBin) => (args: readonly string[], options?: RunOptions) => Promise<RunResult>;

/** The shape of a Postgres baseline's catalog read (pg-live-objects.ts), apart so the CLI's io names it without its code. */
/**
 * What a database holds of what Atlas OSS will not compare: each routine by name, its kind and each
 * overload's signature (its input argument types); each table's triggers by name, with when and for
 * what they fire (tgtype) and whether they do (P13); its views, materialized views, sequences and
 * policies by name
 */
export type LiveObjects = {
    /** each extension, by name, but plpgsql (P10 of r18) */
    extensions: Set<string>;
    /** each domain, composite and range type and collation, by name, with what it is (P10 of r18) */
    kinds: Map<string, "collation" | "composite type" | "domain" | "range type">;
    /**
     * what the read names by one label each, in the target schema (P12 of r19): `aggregate f(int4)`,
     * `function f(int4) security definer`, `operator @@(none, int4)`, `cast (pair as int4)`, `view v
     * (security_invoker=true)`, `index i on m`, `replica identity on t (full)`
     */
    others: Set<string>;
    /** what the read names database-wide, which a platform may hold of its own: `event trigger e`, `publication p` (P12 of r19) */
    platform: Set<string>;
    /** each table's policies, by name, with their command, roles and kind: `for all, to public, permissive` (P9, P10) */
    policies: Map<string, Map<string, string>>;
    /** each view, materialized view and sequence no column owns, by name, with what it is (P9) */
    relations: Map<string, "materialized view" | "sequence" | "view">;
    /** each table's row level security where it is on: `enabled, not forced` (P10 of r18) */
    rls: Map<string, string>;
    routines: Map<string, {
        kind: "function" | "procedure";
        signatures: Set<string>;
    }>;
    /** each table's rules, by name, but a view's own (P10 of r18) */
    rules: Map<string, Set<string>>;
    /** each sequence's settings: `start 1, increment 1, min 1, max …, no cycle` (P10 of r18) */
    sequences: Map<string, string>;
    triggers: Map<string, Map<string, {
        enabled: boolean;
        type: number;
    }>>;
    /** the role the read connects as: the one the migrations ran as, which CURRENT_USER names in a policy (R2 of plan 17) */
    user: string;
};
/** The database a read reaches, and the schema it reads */
export type LiveTarget = {
    schema: string;
    url: string;
};
/** How a baseline reads them: postgres.js, unless a test fakes it */
export type LiveRead = (target: LiveTarget) => Promise<LiveObjects>;

/** How the replay words what it does, and where it says what it left out */
type Replay = {
    /** what the replay was doing, as a refusal names it */
    doing: string;
    /** a statement left out, from `file`, and what it is */
    leftOut(file: string, sent: string, what: string): void;
    /** a CREATE ROLE or ALTER ROLE `sent` from `file` the replay ran without some of what it says */
    madeBare?: ((file: string, sent: string) => void) | undefined;
    /** a statement `sent` from `file` the replay ran in pg_default, not in the tablespaces `names` it left out */
    movedTo?: ((file: string, sent: string, names: readonly string[]) => void) | undefined;
    /** a role `file` added inside a statement the replay cannot leave out, adopted as one it made */
    adopted?: ((file: string, role: string) => void) | undefined;
    /** what `file`'s transaction held before its `statement`th statement, a VALIDATE, read from pg_locks (pg-replay-held.ts) */
    held?: {
        at(file: string, statement: number, locks: ReadonlyMap<string, boolean>): void;
        schema: string;
    } | undefined;
    /** what a refusal stops */
    stops: string;
};
/** A refusal: what failed, with devUrl's error, and what that stops */
type Refuse = (what: string, error: unknown) => ReplayFailed;
/** A replay's run of each file: the refusal it words, and the extensions waiting on one a file makes (PR1 of r22b) */
type Replaying = Replay & {
    refused: Refuse;
    waits: (sql: Sql) => Promise<void>;
};
/** A statement of `name`: its type and node, as nodeOf reads its parse tree, and what Atlas sends */
type Sent = {
    name: string;
    node: Record<string, unknown>;
    sent: string;
    type: string;
};

// migrate/core/pg-live-catalog.d.ts
/**
 * The one catalog read a Postgres baseline makes (pg-live-objects.ts), in one round trip, of what
 * Atlas OSS leaves out of its comparison. Each row says `what` it is; `how` spells its settings as
 * the history's are spelled (P10 of r18), so the two compare as text.
 *
 * - The target schema's functions and procedures, but an extension's (a member of it through
 *   pg_depend, deptype e) and one another object makes (deptype i: a range type's constructors),
 *   each with its input argument types as the parser spells them (the catalog's type name, an array
 *   as its element's and [], the schema left out for pg_catalog and the target's).
 * - Its tables' triggers, but the internal ones a constraint makes, with tgtype and whether they
 *   fire: O and A do, D and R (replica only) do not, as the history reads ENABLE and DISABLE (P13);
 *   a partition's clone of its parent's trigger is Postgres's, so it is left out (tgparentid).
 * - Its views, materialized views and sequences, but an extension's and a sequence a column owns (a
 *   serial's or an identity's, deptype a or i), a sequence with its settings; each table's policies,
 *   with their command, roles and whether they are permissive (P9 of the 0.15.0 review).
 * - Its domains, composite and range types and collations, but an extension's; each table's rules but
 *   a view's own; each table's row level security where it is on; every extension but plpgsql.
 * - The role the read connects as, which a policy's CURRENT_USER names (R2 of plan 17).
 * - P12 of r19, as `other` labels: its aggregates; each function or procedure that is SECURITY DEFINER;
 *   its operators; each cast over one of its types or functions; each view's and materialized view's

/**
 * One replay's state: its connection, the server as committed (`admin`), its database, the backend
 * that owns it, its role (`role`, which CURRENT_USER and its kin name in a file), whether that role
 * is a superuser, which reads pg_authid, whether it has CREATEROLE (`createrole`), and the roles it made; the
 * open transaction's own writes, comb's CREATE ROLEs', with the rows they left; the shared catalogs

 * added (`strays`), which the cleanup drops
 */

    admin: Sql;
    created: string[];

    database: string;
    ours: {
        rows: Rows;
        writes: Writes;
    };
    owner: string;
    role: string;
    seen: Snapshot;
    sql: Sql;
    strays: Strays;
    /** the tablespaces it left out the CREATE of, which a statement then runs in pg_default instead (PR5 of r22b) */
    tablespaces: Set<string>;
    superuser: boolean;
};
/** The role `sql` runs as, which CURRENT_USER and its kin name in a file, whether it is a superuser, and whether it has CREATEROLE */
declare function devRoleOf(sql: Sql): Promise<{
    createrole: boolean;
    role: string;
    superuser: boolean;
}>;
/** `act` run, its writes to the shared catalogs counted as comb's own, with the rows of them that are `mine` */
declare function own(run: Run, act: () => Promise<void>, mine: (subject: string) => boolean): Promise<void>;
/**
 * The CREATE ROLE a replay runs for a history's `node`: the role with BARE's attributes and the
 * memberships among roles in `made`; `whole` when it leaves out nothing the history said

declare function bareRole(node: Record<string, unknown>, made: readonly string[]): {
    sql: string;
    whole: boolean;
    within: string[];
};
/**
 * The CREATE ROLE `sent` of `role`, run in the replay's `database`: an existing role is used, any
 * other made by `sent` and tagged with its `owner`, and either held by a grant on `database`;
 * whether the role is a replay's, which the replay then drops, so whichever of two replays using
 * one role ends last drops it. It runs in a transaction, as every statement a transaction allows
 * does (P1 of r21), and each try is a savepoint, so a role another replay made or dropped since the
 * read is read again

declare function createRole(sql: Sql, { database, owner, role, sent }: {
    database: string;
    owner: string;
    role: string;
    sent: string;

/**

 */
declare function ownRole(run: Run, node: Record<string, unknown>): Promise<boolean>;
/**
 * `roles` of the replay's, changed in its open transaction, made bare again before it commits: no
 * login, no password, and none of the attributes BARE denies, each said only where the role has it,

declare function bareAgain(run: Run, roles: readonly string[]): Promise<void>;

declare function adopt(run: Run, roles: readonly string[]): Promise<void>;
/** Which of `roles` the server holds, a replay's own left out: another replay's role is gone once it ends */
declare function heldRoles(sql: Sql, roles: readonly string[]): Promise<Set<string>>;

// migrate/core/pg-server-writes.d.ts
/**
 * What a replayed transaction wrote past its database (P1 of r21), read from Postgres, never from the
 * file's text: a DO block, a function, EXECUTE and CALL hide a role's or a database's change from the
 * parse tree. `pg_stat_xact_sys_tables` counts this backend's own writes to each shared catalog in the
 * open transaction, for any role and before COMMIT, so another session's work never trips it;
 * pg_shdepend is left out, which a GRANT on a table or an owner moves. The catalogs devUrl's role can
 * read then name the change: the open transaction sees it, a second connection sees the server as it
 * was. A role cannot read pg_authid unless it is a superuser, so a role is read from pg_roles, and a
 * write it does not show, such as a password, is named as one; a superuser reads pg_authid, a
 * password as its hash, so a password's change is named by its role (PR6 of r22b). Probed on Postgres 17 as a superuser
 * and as a CREATEDB CREATEROLE role (.tmp/grunt/r21-premise/p1-xact-stats.log, p1-snapshot.log).
 */
import type { Sql } from "./pg-replay-database.ts";
/** Writes by shared catalog: how many rows the open transaction inserted, updated or deleted in each */
type Writes = Map<string, number>;
/**
 * Whether the server counts this backend's writes, which with `track_counts` off read as none, and
 * the open transaction's writes to each shared catalog, pg_shdepend aside
 */
declare function counts(sql: Sql): Promise<{
    on: boolean;
    writes: Writes;
}>;
/** The writes in `after` that `before` does not account for, by catalog */
declare function writesSince(after: Writes, before: Writes): Writes;
/** `more` added into `into` */
declare function addWrites(into: Writes, more: Writes): void;
declare const CATALOGS: Readonly<Record<string, Catalog>>;
/** How a catalog's rows read: the names each is about, what a row is, and its query */
type Catalog = {
    names: string;
    noun: string;
    query: string;
};
/** The shared catalogs as devUrl's role reads them: pg_authid itself when it is a `superuser` */
declare const catalogsOf: (superuser: boolean) => Readonly<Record<string, Catalog>>;
/** The fields of a row that differ, in order */
declare const changedFields: (before: Record<string, unknown>, after: Record<string, unknown>) => string[];
/** Rows by subject, each as JSON: what comb's own CREATE ROLE left them */
type Rows = Map<string, string>;
/** One row's change: its kind, its subject, the names it is about, the fields that changed, none when added or removed, and how it reads */
type Change = {
    fields: string[];
    kind: "added" | "changed" | "removed";
    names: ReadonlyArray<string | null>;
    said: string;
    subject: string;
};
/** Whether a change is the replay's own */
type Own = (change: Change) => boolean;
/** The rows of the catalogs `written` that `mine` names, as the open transaction on `sql` reads them through `catalogs` */
declare function subjectRows({ catalogs, sql }: {
    catalogs: Readonly<Record<string, Catalog>>;
    sql: Sql;
}, written: Writes, mine: (subject: string) => boolean): Promise<Rows>;
/**
 * What the `counted` writes changed, by catalog in order, as the open transaction on `sql` sees the
 * server against the committed server `was` reads, through `catalogs`, one change per row; a row
 * still as comb's own CREATE ROLE left it (`ours`) is comb's, and no change. The committed server is
 * read before the transaction's read and after it, and a change is one only against both: what
 * another session commits meanwhile, on a server others share, is in one of them
 */
declare function rowChanges({ catalogs, sql, was }: {
    catalogs: Readonly<Record<string, Catalog>>;
    sql: Sql;
    was: Sql;
}, counted: Writes, ours: Rows): Promise<Map<string, Change[]>>;
/**
 * The changes of `byCatalog` the replay must refuse: each that is not its `own`, and a counted
 * catalog with no row to name, a write devUrl's role cannot see; a catalog whose every change is
 * the replay's own counted no other
 */
declare function refusedChanges(byCatalog: ReadonlyMap<string, Change[]>, own: Own): string[];
/** Refuses a devUrl whose server keeps no counts for its role: the replay would read every file's writes as none */
declare function refuseUncounted(sql: Sql, stops: string): Promise<void>;
export { addWrites, CATALOGS, type Catalog, catalogsOf, type Change, changedFields, counts, refusedChanges, refuseUncounted, type Own, rowChanges, type Rows, subjectRows, type Writes, writesSince, };

// migrate/core/pg-server-snapshot.d.ts
/**
 * The server's shared catalogs read as committed, in one query (P1 and P2 of r22): what a bare
 * statement changed, which no transaction counts, and what a second connection committed, which this
 * backend's counts never see (pg-server-writes.ts). The read is the server's, another session's work
 * in it too, so each row carries the names of the objects it is about, and a change is a file's only
 * when one is a name the file spells (namesIn in sql-code.ts). A replay's own rows are left out, but
 * for the fields of its own database's row that keep the cleanup from dropping it, which only that
 * replay changes. Beside the catalogs, the server's replication slots (PR2 of r22b), which no catalog
 * holds.
 */
import type { Sql } from "./pg-replay-database.ts";
/**
 * The server's shared catalogs as committed, a replay's own rows left out: each row's JSON by its
 * subject, with the names of the roles, databases or other objects the row is about, as stored
 */
type Snapshot = Map<string, {
    names: readonly string[];
    row: Record<string, unknown>;
}>;
/**
 * The read prepared on `admin` for snapshot, as its role reads it, with the replay's own
 * `database`'s row: planning it is half its time, and a replay reads it around every file and bare
 * statement
 */
declare function prepareSnapshot(admin: Sql, { database, superuser }: {
    database: string;
    superuser: boolean;
}): Promise<void>;
/** The shared catalogs as `admin`, a connection with no transaction open that prepareSnapshot prepared, reads them committed */
declare function snapshot(admin: Sql): Promise<Snapshot>;
/**
 * What changed from `before` to `after`, one change per subject, in order: of a subject about a
 * name in `names`, the names a file spells (P2 of r22), when given
 */
declare function snapshotChanges(before: Snapshot, after: Snapshot, names?: ReadonlySet<string>): string[];
/** What the cleanup drops, by the noun its subject starts with */
type Strays = {
    databases: string[];
    roles: string[];
    slots: string[];
};
/** The roles, databases and replication slots named in `names` that `after` holds and `before` does not */
declare function added(before: Snapshot, after: Snapshot, names: ReadonlySet<string>): Strays;
export { added, prepareSnapshot, type Snapshot, snapshot, snapshotChanges, type Strays };
