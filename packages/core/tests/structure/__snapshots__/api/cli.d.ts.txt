/** Where the CLI reads its working directory, writes its output, asks, and starts processes. */
export type CliIo = {
    cwd: string;
    out(line: string): void;
    err(line: string): void;
    /** How the CLI starts a process (atlas, wrangler, drizzle-kit), spawnSync unless the embedder supplies one */
    run?: Run | undefined;
    /** Ask the person running the CLI a yes/no question; absent when nobody can answer (no TTY) */
    confirm?: ((question: string) => Promise<boolean>) | undefined;
};
/**
 * What a command reads besides its io, comb's own unless its tests fake one (KT7 of r22b): typed by
 * comb's internals, so they stay off the public CliIo and reach the CLI through runCliWith alone.
 */
export type CliDeps = {
    /** How a Postgres baseline reads the database's catalog of what Atlas OSS does not compare, postgres.js unless a test fakes it (5.4b) */
    readPgObjects?: LiveRead | undefined;
    /** How a Postgres baseline reads the history replayed on its dev database, comb's own (pg-replay.ts) unless a test fakes it (P5 of r19) */
    replayPgHistory?: typeof replayCatalog | undefined;
    /** How a diff, baseline or adoption gives the dev database a non-public target schema, comb's own unless a test fakes it (P12) */
    readyPgDev?: ((target: {
        devUrl: string;
        schema: string;
    }) => Promise<void>) | undefined;
    /** How a migrate command finds an optional peer's entry before its first I/O, as comb and `bun` find it unless a test fakes it (K1 of r21) */
    locatePeer?: ((peer: string) => string) | undefined;
    /** How a migrate command reads Atlas's version before its first I/O, `atlas version` through `run` unless a test fakes it (PA5 of r22) */
    readAtlasVersion?: (() => RunResult) | undefined;
};
/** The io a command runs on, with the seams runCliWith was given */
export type CliContext = CliIo & CliDeps;
/** What `comb migrate check` reports once every pending migration is approved */

export type MigrateOutcome = {
    applied: string[];
    pending: string[];
};
export declare const checkPassed: (names: readonly string[]) => string;
/** A failure the CLI reports as its message and exit code instead of a stack trace. */

    readonly exitCode: number;
    constructor(message: string, exitCode?: number, options?: ErrorOptions);
}
/** A failure a command already reported as its --json object: printed on stderr, never as a second object (K10 of r18) */
export declare class Reported extends CliError {
    constructor(error: unknown);
}
/** A request for help: the CLI prints `text` on out and exits 0, wherever in argv -h or --help came */
export declare class ShowHelp extends Error {
    readonly text: string;
    constructor(text: string);
}

    };

// migrate/core/pg-live-types.d.ts
/** The shape of a Postgres baseline's catalog read (pg-live-objects.ts), apart so the CLI's io names it without its code. */
/**
 * What a database holds of what Atlas OSS will not compare: each routine by name, its kind and each
 * overload's signature (its input argument types); each table's triggers by name, with when and for
 * what they fire (tgtype) and whether they do (P13); its views, materialized views, sequences and
 * policies by name
 */
export type LiveObjects = {
    /** each extension, by name, but plpgsql (P10 of r18) */
    extensions: Set<string>;
    /** each domain, composite and range type and collation, by name, with what it is (P10 of r18) */
    kinds: Map<string, "collation" | "composite type" | "domain" | "range type">;
    /**
     * what the read names by one label each, in the target schema (P12 of r19): `aggregate f(int4)`,
     * `function f(int4) security definer`, `operator @@(none, int4)`, `cast (pair as int4)`, `view v
     * (security_invoker=true)`, `index i on m`, `replica identity on t (full)`
     */
    others: Set<string>;
    /** what the read names database-wide, which a platform may hold of its own: `event trigger e`, `publication p` (P12 of r19) */
    platform: Set<string>;
    /** each table's policies, by name, with their command, roles and kind: `for all, to public, permissive` (P9, P10) */
    policies: Map<string, Map<string, string>>;
    /** each view, materialized view and sequence no column owns, by name, with what it is (P9) */
    relations: Map<string, "materialized view" | "sequence" | "view">;
    /** each table's row level security where it is on: `enabled, not forced` (P10 of r18) */
    rls: Map<string, string>;
    routines: Map<string, {
        kind: "function" | "procedure";
        signatures: Set<string>;
    }>;
    /** each table's rules, by name, but a view's own (P10 of r18) */
    rules: Map<string, Set<string>>;
    /** each sequence's settings: `start 1, increment 1, min 1, max …, no cycle` (P10 of r18) */
    sequences: Map<string, string>;
    triggers: Map<string, Map<string, {
        enabled: boolean;
        type: number;
    }>>;
    /** the role the read connects as: the one the migrations ran as, which CURRENT_USER names in a policy (R2 of plan 17) */
    user: string;
};
/** The database a read reaches, and the schema it reads */
export type LiveTarget = {
    schema: string;
    url: string;
};
/** How a baseline reads them: postgres.js, unless a test fakes it */
export type LiveRead = (target: LiveTarget) => Promise<LiveObjects>;

// migrate/core/pg-replay.d.ts
import { type PgTarget } from "./atlas-target.ts";
import { type Connect } from "./pg-live-objects.ts";
import type { LiveObjects } from "./pg-live-types.ts";
import { type Sql } from "./pg-replay-database.ts";
/** A history file as baseline reads it */
type HistoryFile = {
    name: string;
    sql: string;
};
/** How the replay words what it does, and where it says what it left out */
type Replay = {
    /** what the replay was doing, as a refusal names it */
    doing: string;
    /** a statement left out, from `file`, and what it is */
    leftOut(file: string, sent: string, what: string): void;
    /** a CREATE ROLE `sent` from `file` the replay ran without some of what it says */
    madeBare?: ((file: string, sent: string) => void) | undefined;
    /** a role `file` added inside a statement the replay cannot leave out, adopted as one it made */
    adopted?: ((file: string, role: string) => void) | undefined;
    /** what a refusal stops */
    stops: string;
};
/**
 * `files` replayed on `target`'s dev server in a database of their own, the schema made and searched
 * first; then `after` on its connection. A statement dev refuses is named, with dev's error, as what
 * was `doing`, and then what that `stops`
 */
declare function onDev<T>(target: PgTarget, files: readonly HistoryFile[], { after, load, ...replay }: Omit<Replay, "leftOut"> & {
    after: (sql: Sql) => Promise<T>;
    leftOut?: Replay["leftOut"] | undefined;
    load: () => Promise<Connect>;
}): Promise<T>;
/**
 * The catalog the history `files` leave in `target`'s schema, replayed on its dev server as Atlas
 * applied them (onDev), read by the target's own query (readCatalog). The role the replay
 * ran as stands for the target's role wherever the read names it (`targetUser`), as CURRENT_USER
 * names the role each database was built as
 */
declare function replayCatalog(target: PgTarget, { files, targetUser }: {
    files: readonly HistoryFile[];
    targetUser: string;
}, load?: () => Promise<Connect>): Promise<LiveObjects>;
export { type HistoryFile, onDev, replayCatalog };

// migrate/core/atlas-target.d.ts
/** A Postgres database as Atlas reaches it: target, dev database, schema and lock wait */
type PgTarget = {
    devUrl: string;
    lockTimeout: string;
    schema: string;
    url: string;
};
/** What an Atlas call reads from its config: the target, the dev database, the migrations dir, and whose history */
type AtlasNeeds = {
    dev?: boolean | undefined;
    dir?: string | undefined;
    /** the target unscoped, for the one call that lists the database's schemas (M78) */
    allSchemas?: boolean | undefined;
    /** a history other than the target's own, by its schema: the shared one 0.12 kept every schema's in (M77), or another schema's own (M78) */
    revisions?: string | undefined;
    url?: boolean | undefined;
};
/** The flags and environment an Atlas call runs with, its scratch directory, and the scrubber for its output */
type AtlasEnv = {
    args: readonly string[];
    dir: string;
    env: Readonly<Record<string, string>>;
    /** free text: the call's secrets, and any other URL's password or secret query value, masked */
    scrub(text: string): string;
};
/** What is wrong with the URLs `needs` names before Atlas runs: errors refuse it, warnings are printed */
declare function pgUrlProblems(target: PgTarget, needs?: AtlasNeeds): {
    errors: string[];
    warnings: string[];
};
/** The schema Atlas keeps public's history in, as it always has */
declare const SHARED_REVISIONS = "atlas_schema_revisions";
/**
 * The schema Atlas keeps `target`'s history in: the shared one for public, and one of its own for
 * any other scope, so two schemas of one database never read each other's history. Postgres cuts a
 * name at 63 bytes, so a scope too long, or not a plain lower-case name, is named by its hash.
 */
declare function revisionsSchema(target: PgTarget): string;
/** Both URLs as Atlas gets them */
declare const atlasUrls: (target: PgTarget) => {
    devUrl: string;
    url: string;
};
/** The atlas.hcl for `needs`: getenv for each URL, and the history pinned to `revisions` */
declare function atlasConfig(needs: AtlasNeeds, revisions: string): string;
/** Refuse `target` as pgUrlProblems does for `needs`, before anything reaches a database (P2 of r19) */
declare function refuseUrlProblems(target: PgTarget, needs: AtlasNeeds): void;
/**
 * Run `body` with an Atlas config for `target` and `needs` in a fresh 0700 directory, removed
 * afterwards however `body` ends. A target with errors is refused before anything is written.
 */
declare function withAtlasEnv<T>(target: PgTarget, needs: AtlasNeeds, body: (atlas: AtlasEnv) => T): T;
/**
 * What Atlas said on stderr as it succeeded, a line each: comb passes each on as a warning, as Atlas
 * OSS may say it left part of a schema out of what it compared (plan 14's 5.4)
 */
declare const atlasNotices: (stderr: string) => string[];
export { atlasConfig, type AtlasEnv, type AtlasNeeds, atlasNotices, atlasUrls, pgUrlProblems, type PgTarget, refuseUrlProblems, revisionsSchema, SHARED_REVISIONS, withAtlasEnv, };

/** The postgres.js client, as far as comb uses it */
type Sql = {

        timeout?: number | undefined;
    }): Promise<void>;
    unsafe(query: string, params: unknown[]): Promise<Array<Record<string, unknown>>>;
};
type Connect = (options: Record<string, unknown>) => Sql;
/** Where a postgres.js failure came from: the field whose URL comb connected to, and every URL whose secrets it may hold */
type FailedAt = {
    field: "devUrl" | "url";
    url: string;
    urls: readonly string[];
};
/**
 * What a postgres.js failure on `at.field` says: its message, or, for the AggregateError it throws
 * with none when every address refuses, each address's answer joined (P15 of the 0.15.0 review), each
 * secret of `at.urls` masked; for a server sslrootcert=system cannot verify, where comb's CAs came from
 * (R1 of 5.3). comb's own words follow the mask, so a password that is one of them stays out of them
 * (P5 of r20)
 */
declare function failureOf(error: unknown, at: FailedAt): string;
/** postgres.js, or the refusal naming it, its range and `why` comb needs it here */
declare function loadPostgres(why: string): Promise<Connect>;
/**
 * What `schema` holds of what Atlas OSS does not compare, read through `sql` by the one catalog
 * query: the target's read and the dev replay's are this one function (P5 of r19), so the two sides
 * cannot be read two ways
 */
declare function readCatalog(sql: Pick<Sql, "unsafe">, schema: string): Promise<LiveObjects>;
/** What the target schema of `target`'s database holds of what Atlas OSS does not compare; it writes nothing */
declare function readLiveObjects(target: LiveTarget, load?: () => Promise<Connect>): Promise<LiveObjects>;
/**
 * What the database and the history disagree on, both read by readCatalog (P5 of r19), the history's
 * from its replay on dev: each object only one of them has, by name and signature or how a trigger
 * fires, and how many of each kind they share, whose bodies neither read
 */
declare function liveDrift(history: LiveObjects, live: LiveObjects): {
    matched: {
        routines: number;
        triggers: number;
        views: number;
    };
    only: string[];
};
export { COMPARED, type Connect, failureOf, liveDrift, loadPostgres, readCatalog, readLiveObjects };

import type { PgTarget } from "./atlas-target.ts";

type Sql = ReturnType<Connect>;
declare const TAG = "comb_replay";
declare const STALE_MS: number;
/** A Postgres error's SQLSTATE */
declare const codeOf: (error: unknown) => unknown;
declare const literal: (text: string) => string;
/** `step` run, its failure left to the next run's sweep: cleaning up never replaces the replay's verdict */
declare function settle(step: () => Promise<unknown>): Promise<void>;
/** Drops the replication slot `slot`, if it is still there */
declare function dropSlot(admin: Sql, slot: string): Promise<void>;
/**
 * Drop `role`, unless something still depends on it or it is not this role's to drop: a later sweep
 * takes it. Two sessions dropping one role race in Postgres's catalog, and the loser fails without
 * a SQLSTATE of its own (XX000, "could not find tuple for role"), so a role gone after any failure
 * is dropped
 */
declare function dropRole(admin: Sql, role: string): Promise<void>;
/**
 * The backend `admin` runs on, as a replay's database name and role tag carry it, `<pid>_<start>`,
 * and the settings a replay connection takes (P4 of r22): that owner as its application_name, by
 * which the sweep finds a killed replay's backends, and on Postgres 14 and later a check every
 * second that its client is there, by which Postgres ends their statement itself; `admin` takes them
 * too
 */
declare function claim(admin: Sql): Promise<{
    owner: string;
    settings: Record<string, string>;
}>;
/**
 * Drop what replays over at `now` left, each drop settled: every replay database this role may
 * drop, whichever role made it (PR8 of r22b), once the backends its replay left in it are ended (P4
 * of r22), and the tagged roles
 */
declare function sweep(admin: Sql, now: number): Promise<void>;
/**
 * The sweep, run on `target`'s dev server before an apply (P2 of r21): a check killed past its
 * cleanup leaves a role there, which apply's CREATE ROLE of it stops at where dev and target share
 * a server. Only that server matters, so apply asks the target's own server first, which it reaches
 * anyway, and connects to devUrl only when the target's holds what a sweep would take, waiting
 * SWEEP_TIMEOUT for it (PR7 of r22b). Housekeeping never stops an apply, so without the postgres
 * package, on Atlas's docker:// container or with nothing left it sweeps nothing, and a sweep that
 * fails is `print`ed and passed
 */
declare function sweepDev(target: PgTarget, print: (line: string) => void, load?: () => Promise<Connect>): Promise<void>;
/** A new replay database on `admin`'s server, named for its `owner`; a role without CREATEDB is `refused` */
declare function makeDatabase(admin: Sql, owner: string, refused: () => Error): Promise<string>;
/** The extensions the history `texts` create, which the replay leaves them to make */
declare const extensionsOf: (texts: readonly string[]) => Set<string>;
/**
 * Every extension `from` holds made in `to`, in its schema at its version, but those the history
 * `makes`; one `to` cannot make is `refused`
 */
declare function copyExtensions(from: Sql, to: Sql, { makes, refused }: {
    makes: ReadonlySet<string>;
    refused: (name: string, error: unknown) => Error;
}): Promise<void>;
export { codeOf, copyExtensions, dropRole, dropSlot, extensionsOf, literal, makeDatabase, claim, settle, STALE_MS, type Sql, sweep, sweepDev, TAG, };

// migrate/core/pg-live-catalog.d.ts
/**
 * The one catalog read a Postgres baseline makes (pg-live-objects.ts), in one round trip, of what
 * Atlas OSS leaves out of its comparison. Each row says `what` it is; `how` spells its settings as
 * the history's are spelled (P10 of r18), so the two compare as text.
 *
 * - The target schema's functions and procedures, but an extension's (a member of it through
 *   pg_depend, deptype e) and one another object makes (deptype i: a range type's constructors),
 *   each with its input argument types as the parser spells them (the catalog's type name, an array
 *   as its element's and [], the schema left out for pg_catalog and the target's).
 * - Its tables' triggers, but the internal ones a constraint makes, with tgtype and whether they
 *   fire: O and A do, D and R (replica only) do not, as the history reads ENABLE and DISABLE (P13);
 *   a partition's clone of its parent's trigger is Postgres's, so it is left out (tgparentid).
 * - Its views, materialized views and sequences, but an extension's and a sequence a column owns (a
 *   serial's or an identity's, deptype a or i), a sequence with its settings; each table's policies,
 *   with their command, roles and whether they are permissive (P9 of the 0.15.0 review).
 * - Its domains, composite and range types and collations, but an extension's; each table's rules but
 *   a view's own; each table's row level security where it is on; every extension but plpgsql.
 * - The role the read connects as, which a policy's CURRENT_USER names (R2 of plan 17).
 * - P12 of r19, as `other` labels: its aggregates; each function or procedure that is SECURITY DEFINER;
 *   its operators; each cast over one of its types or functions; each view's and materialized view's
 *   options (security_invoker, security_barrier, check_option); each materialized view's indexes; each
 *   table's replica identity where it is not the default. As `platform` labels, database-wide, which a
 *   platform may hold of its own: event triggers and publications. Each but an extension's.
 */
import type { LiveObjects } from "./pg-live-types.ts";
export declare const CATALOG: string;
/**
 * What each arm of CATALOG compares, in the arms' order, as a refusal names it: the one list COMPARED
 * is written from, held to CATALOG's arms by its test, so an arm cannot go unnamed
 */
export declare const ARM_KINDS: ReadonlyArray<readonly [what: string, kinds: readonly string[]]>;
/** What a baseline compares itself, as its refusals name it */
export declare const COMPARED: string;
/** `live` read on dev, with dev's role named as the target's wherever a policy names it (the reviewer's addition A) */
export declare function asTarget(live: LiveObjects, targetUser: string): LiveObjects;
