// migrate/migration-types.d.ts
/**
 * The types of a migration file as comb reads it: the schema its history leaves, and what its
 * statements risk. Declarations only.
 */
/** A foreign key as a migration declares it; an empty `parentColumns` is the parent's primary key */
export type ForeignKey = {
    /** the table that holds the key */
    table: string;
    /** its columns, in order */
    columns: string[];
    /** the table it references */
    parent: string;
    /** the parent's columns it references, in order */
    parentColumns: string[];
    /** what a delete of a parent row does; `NO ACTION` when the migration says nothing */
    onDelete: "CASCADE" | "NO ACTION" | "RESTRICT" | "SET DEFAULT" | "SET NULL";
};
/** What a table rebuild must know of a table to save and restore its rows */
export type TableInfo = {
    /** its primary key's columns, in order; empty when it declares none */
    primaryKey: string[];
    /** a WITHOUT ROWID table, whose rows are saved by primary key */
    withoutRowid: boolean;
    /** its primary key is the rowid: a lone INTEGER column, so writing the key writes the rowid */
    rowidAlias: boolean;
    /** columns no INSERT may write */
    generatedColumns: string[];
    /** SQLite keeps its next id in sqlite_sequence */
    autoincrement: boolean;
    /** each column's type affinity by SQLite's rules: INTEGER, TEXT, BLOB, REAL or NUMERIC */
    affinity: Record<string, string>;
    /** each ON CONFLICT resolution other than ABORT the table declares, as `<column or key>: <resolution>` */
    conflicts: string[];
    /** a STRICT table, which refuses a value of another type */
    strict: boolean;
};
/** A write that fires a trigger */
export type TriggerEvent = "delete" | "insert" | "update";
/**
 * The schema a migration history leaves, followed statement by statement. A name is bare when its
 * object is in the schema comb migrates, else `schema.name`
 */
export type ExtractedSchema = {
    /** each table left, by name, with its column names */
    tables: Map<string, string[]>;
    /** each index created, by name */
    indexes: string[];
    /** each table a DROP TABLE dropped, in order */
    droppedTables: string[];
    /** each foreign key of the tables left */
    foreignKeys: ForeignKey[];
    /**
     * each trigger left, by its table, with the writes that fire it; `ftsSync` when its body only
     * writes into an external-content FTS5 table (the sync triggers comb writes)
     */
    triggers: Array<{
        /** the writes to `table` that fire it */
        events: TriggerEvent[];
        /** its body only writes into an external-content FTS5 table */
        ftsSync: boolean;
        /** the trigger's name */
        name: string;
        /** the table it is on */
        table: string;
    }>;
    /** the FTS5 tables whose text lives elsewhere (`content='src'`, or contentless `content=''`): a drop of one loses nothing */
    externalFts: string[];
    /**
     * the statements of the history, in order: on Postgres, what its parse tree shows a write runs
     * (parentage, defaults, rules, operators, casts) is read from them
     */
    statements: string[];
    /** what a table rebuild must know of each table left, by name */
    tableInfo: Map<string, TableInfo>;
};
/** How much a migration risks: `unsafe` loses data or breaks deployed code, `cautious` drops or changes something no row holds or is not one transaction, `safe` neither */
export type MigrationSafety = "safe" | "cautious" | "unsafe";
/** What a migration's statements risk, and which do */
export type MigrationClassification = {
    /** `unsafe` when any statement is, else `cautious` when any statement is, else `safe` */
    safety: MigrationSafety;
    /** statements that lose data or break deployed code by name */
    unsafeStatements: string[];
    /** statements that drop or change something no row holds, or a file that is not one transaction */
    cautiousStatements: string[];
    /** Postgres statements that block a table while they run, with what to do instead */
    lockWarnings: string[];
    /**
     * the SQLite writes among `unsafeStatements` the classifier could not clear on its own, pending
     * the proof run's read of what each writes; the proof's findings replace them
     */
    pendingStatements: string[];
};
/** A migration file, read and classified */
export type ParsedMigration = {
    /** the file's SQL, as written */
    sql: string;
    /** the sha256 of the file's normalized SQL, which the history records for it */
    hash: string;
    /** the file's name in the migrations folder */
    filename: string;
    /** the file's statements, in order */
    statements: string[];
    /** what the file drops, as `DROP TABLE users` or `DROP COLUMN post.body` */
    drops: string[];
    /** the schema the file alone builds, on an empty database */
    expected: ExtractedSchema;
    /** what the file's statements risk */
    classification: MigrationClassification;
};

/** The rows one statement answered */
export type QueryResult = {
    /** the column names, in order; on bun:sqlite and D1 they come from the first row, so a result with no rows has none */
    columns: string[];
    /** each row's values, in `columns`' order */
    rows: unknown[][];
};
/** An open connection to one database, which a migrate command runs its statements through */
export type Connection = {
    /** the SQL dialect the database speaks */
    dialect: Dialect;
    /**
     * The rows `sql` answers, with each `?` in it bound to the next of `params`. On D1 the params are
     * written into the SQL as literals, since wrangler binds none
     */
    execute(sql: string, params?: unknown[]): Promise<QueryResult>;
    /**
     * `statements` in order, as one transaction on SQLite, libsql and bun:sqlite. On D1 they go in order
     * through wrangler, in no transaction and in as many `--command` calls as one argument's length
     * needs, so a caller sends only statements that are safe to run again.
     */
    batch(statements: Array<{
        /** one statement, its `?` placeholders bound to `params` */
        sql: string;
        /** the values of `sql`'s placeholders, in order */
        params?: unknown[] | undefined;
    }>): Promise<void>;
    /** Closes the connection; on D1, which keeps none open, it does nothing */
    close(): Promise<void>;
    /** the prefix of the tables the engine keeps for itself (D1: `_cf_`), which comb never counts or copies */
    ownedPrefix?: string | undefined;
};

/**
 * The destructive-change gate. Each statement's code (codeOf in sql-lexer.ts) is read per action,
 * so a drop after an `ADD` in one `ALTER TABLE` counts, and a name may be quoted or schema-qualified.
 *
 * unsafe: loses data or breaks deployed code by name (table, column, schema, type, sequence,
 * role drops; CASCADE; retypes; SET NOT NULL; a NOT NULL column without a default; renames;
 * TRUNCATE; a statement comb does not recognize, a trigger or routine made or dropped among them).
 * cautious: drops nothing a row holds (indexes, constraints, views, policies, defaults), a grant, or
 * a file Atlas runs outside a transaction. A lock warning names a
 * statement that blocks a Postgres table while it runs. What a statement writes is read from the
 * engine: on SQLite by the proof run (proof-engine.ts), on Postgres from its parse tree
 * (pg-tree.ts), which also rates the code a statement runs: DO blocks, routines and function
 * calls. A statement Postgres refuses inside a transaction is refused outright unless its file
 * runs outside one, and one it refuses outside a block is refused in a file that does, but within
 * the file's own BEGIN.
 */

import type { Dialect } from "../types.ts";

    kind?: TargetKind | undefined;
    /** how the file is read; default SQLite, as the CLI's */

    /** the Postgres schema the URLs' search_path names, where an unqualified name resolves; default `public` */
    schema?: string | undefined;
};
/**
 * The file as its dialect reads it; with none named, as SQLite reads it, the CLI's default. On
 * Postgres `read` is the file's one parse, its caller's when it holds one, and what check's
 * replay on devUrl measured of it, when it `replayed` the file

/**
 * SQL split into statements, normalized and hashed, read through the quote-aware lexer
 * (sql-lexer.ts), so `;`, `--` and `/*` inside a string or identifier are always data.
 */
import type { Dialect } from "../types.ts";
/**
 * The statements of `sql`, without their terminating `;`. On SQLite they end where SQLite ends
 * them: at a `;` in code, never inside a literal or a comment, nor inside a trigger's body. On
 * Postgres they are the statements Postgres's own parser splits it into, so `await
 * loadPostgresParser()` first: until then it throws. A stretch that holds only comments is not a statement.
 */
declare function splitSqlStatements(sql: string, dialect?: Dialect): string[];
/**
 * `sql` with comments dropped and SQLite's whitespace collapsed outside strings and identifiers, those
 * verbatim: two migrations that differ only in layout or comments normalize alike, two that differ
 * in data or names never. A migration's hash is taken over this form, so it stays as recorded: a
 * word written against a quoted name stays against it. `foldAll` is the form comb 0.5.0 to 0.20
 * hashed, which folded every Unicode space too, so `a b` and `a\u00a0b` (a column of type b, and a
 * column named a b) normalized alike.
 */
declare function hashedSql(sql: string, foldAll?: boolean): string;
/**
 * `sql` as hashedSql collapses it, read as SQLite tokenizes it: a quoted name is a token of its own,
 * so a word written against one is set off by a space, as `ALTER TABLE "users"DROP COLUMN email`
 * drops the column `ALTER TABLE "users" DROP COLUMN email` does; `"main".t` and `"a"(` stay as written
 */
declare function normalizeSql(sql: string): string;
export { hashedSql, normalizeSql, splitSqlStatements };

/**
 * The code of a statement as the destructive-change gate reads it: identifiers kept, strings and
 * comments blanked, a `DO` body read as the code it runs. Names come back unquoted, and a list or
 * an action list splits only at its top-level commas.
 */

declare const QNAME: string;
