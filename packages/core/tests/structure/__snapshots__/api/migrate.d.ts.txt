// migrate/core/classify.d.ts
/**
 * The destructive-change gate. Each statement's code (sql-code.ts) is read per action, so a drop
 * after an `ADD` in one `ALTER TABLE` counts, and a name may be quoted or schema-qualified.
 *
 * unsafe: loses data or breaks deployed code by name (table, column, schema, type, sequence,
 * role drops; CASCADE; retypes; SET NOT NULL; a NOT NULL column without a default; renames;
 * TRUNCATE; a statement comb does not recognize, a trigger or routine made or dropped among them).
 * cautious: drops nothing a row holds (indexes, constraints, views, policies, defaults), a grant, or
 * a file Atlas runs outside a transaction. A lock warning names a
 * statement that blocks a Postgres table while it runs. What a statement writes is read from the
 * engine: on SQLite by the proof run (proof-engine.ts), on Postgres from its parse tree
 * (pg-tree.ts), which also rates the code a statement runs: DO blocks, routines and function
 * calls. A statement Postgres refuses inside a transaction is refused outright unless its file
 * runs outside one.
 */
import type { Dialect, ExtractedSchema, MigrationClassification } from "../types.ts";
type ClassifyOptions = {
    /** the schema the earlier migrations leave, which tells a lossless SQLite table rebuild from a drop */
    before?: ExtractedSchema | undefined;
    /** how the file is read; unnamed, as SQLite, the CLI's default */
    dialect?: Dialect | undefined;
    /** the Postgres schema the URLs' search_path names, where an unqualified name resolves; public unnamed (M72) */
    schema?: string | undefined;
};
/** The table, column and index drops a migration makes, as `DROP TABLE t`, `DROP COLUMN t.c`, `DROP INDEX i` */
declare function extractDropStatements(sql: string, options?: ClassifyOptions): string[];
/** How destructive a migration is, which statements make it so, and which block a Postgres table */
declare function classifyMigration(sql: string, options?: ClassifyOptions): MigrationClassification;
export { classifyMigration, type ClassifyOptions, extractDropStatements };

// migrate/core/sql-scan.d.ts
/**
 * A quote-aware reading of SQL text: code, strings, identifiers and comments. Every migration
 * step that splits, hashes, terminates or classifies SQL reads it through here, so `;`, `--` and
 * `/*` inside a string or identifier are always data. The dialect decides the lexer: SQLite quotes
 * identifiers with `"`, backticks or brackets and has flat comments; Postgres quotes them with `"`
 * only, nests comments, escapes with backslashes in `E'…'`, and dollar-quotes `$tag$…$tag$`.
 */
import type { Dialect } from "../types.ts";
/**
 * `body` is a dollar-quoted `DO` body: it runs during the migration, so it is code to classify,
 * yet one statement to split, while a function body (`literal`) is only defined.
 */
type Segment = {
    kind: "body" | "code" | "comment" | "identifier" | "literal";
    text: string;
};
/** `sql` as consecutive segments that concatenate back to it, read as `dialect` reads it */
declare function segments(sql: string, dialect?: Dialect): Segment[];
/**
 * The statements of `sql`, without their terminating `;`. On SQLite a `;` ends a statement only
 * in code: never inside a literal or a comment, nor inside a trigger's BEGIN … END. On Postgres
 * they are the statements Postgres runs, the one Postgres split (pg-split.ts, M75), read after
 * loadPostgresParser(). A stretch that holds only comments is not a statement.
 */
declare function splitSqlStatements(sql: string, dialect?: Dialect): string[];
/**
 * `sql` with comments dropped and whitespace collapsed outside strings and identifiers, those
 * verbatim: two migrations that differ only in layout or comments normalize alike, two that differ
 * in data or names never. A migration's hash is taken over this form.
 */
declare function normalizeSql(sql: string): string;
export { normalizeSql, segments, splitSqlStatements, type Segment };

    fts: string[];
    plain: string[];
    state: FtsState;
};

 * `sql`'s statements split into those Atlas may see and the FTS state they leave, continuing

declare function ftsDelta({ from, plainDiff, to, toStatements }: FtsDelta): string;
