// query/sqlite/condition.d.ts
/**
 * One filter condition as SQL on one column: the single lowering filterToSQL (drizzle's columns)
 * and buildFilterSQL (a table alias's columns by name) share, so no operator or value conversion
 * exists in one and not the other (Q5 of r17). A column is described by what it stores (its kind)
 * and the SQL its values compare against.
 */
import { type SQL, type SQLWrapper } from "drizzle-orm";
import type { SQLiteColumn } from "drizzle-orm/sqlite-core";
import type { FilterCondition } from "../types.ts";
/**
 * What a column stores, which decides how a validated value binds: text; a number (`numeric`, text
 * SQLite compares by number affinity); a date as epoch milliseconds or whole seconds; or `as-is`, a
 * value bound as validated (a boolean, JSON, or a raw column whose kind was not given)
 */
type ColumnKind = "as-is" | "date-ms" | "date-s" | "number" | "numeric" | "text";

    kind: ColumnKind;

};

/** One condition on one drizzle column; null when the operator or value is not one this column can take */

// query/sqlite/keyset.d.ts
/**
 * Ordering and keyset predicates — one implementation for every SQLite builder.
 *
 * ORDER BY and the cursor predicate must agree on direction AND null placement
 * for every key, or pagination skips and repeats rows. Both are derived here
 * from the same `SortKey[]`, so they cannot disagree.
 */
import { type SQL, type SQLWrapper } from "drizzle-orm";
import type { SortDirection, SortNulls } from "../types.ts";
type SortKey = {
    column: SQLWrapper;
    direction: SortDirection;
    nulls?: SortNulls | undefined;
};
type Tiebreak = {
    column: SQLWrapper;
    direction: SortDirection;
};
/** `col ASC NULLS LAST` — explicit placement so SQLite's default never leaks in. */
declare function orderSQL(key: SortKey): SQL;
/** ORDER BY for the keys followed by the tiebreak, which makes the order total. */
declare function orderByKeys(keys: readonly SortKey[], tiebreak: Tiebreak): SQL[];
/** A count as a SQL literal, where a bound parameter would cost the D1 budget one */
declare const countLiteral: (value: number) => SQL;
/** An integer, a JS number's or a bigint's, as a SQL literal, where a bound parameter would cost the D1 budget one (Q15 of r17) */
declare const integerLiteral: (value: bigint | number) => SQL;
/**
 * A bound the keyset reads from the cursor's own row (cursor-ref.ts) rather than binds: a long text
 * value carried by reference, or a seconds-mode timestamp, whose fraction no Date holds (`row`); or
 * a SQLite number the cursor carried as null, which may be a ±Infinity D1 hands over as null. Each is SQL that binds, so the keyset reads it twice at most;
 * `ends` when a NULL bound means no bound is safe, and the list ends there.
 */
declare class AnchorBound {
    readonly ends: boolean;
    readonly kind: "infinite" | "row";
    readonly value: SQL;
    constructor(kind: "infinite" | "row", value: SQL, ends?: boolean);
}
/** Where a cursor points: the sort keys and the row's values for them, then the tiebreak and its id */
type KeysetPosition = {
    id: unknown;
    keys: readonly SortKey[];
    tiebreak: Tiebreak;
    values: readonly unknown[];
};
/**
 * Lexicographic "after this row" over every key plus the tiebreak, expanded rather than a row-value
 * comparison because each key carries its own direction and null placement. A bound that `ends`
 * is NULL once its anchor is gone: every row passes then, each carrying anchorGone's flag, so the
 * page answers invalid_cursor rather than end the list unseen (U26).
 */
declare function keysetSQL(position: KeysetPosition): SQL;
/**
 * 1 where a bound with no safe fallback is NULL, its anchor row gone or renamed, else 0; undefined
 * when no bound can be. It reads the anchor row alone, so Postgres plans it once a query (U26).
 */
declare function anchorGone(values: readonly unknown[]): SQL | undefined;
export { AnchorBound, anchorGone, countLiteral, integerLiteral, type KeysetPosition, keysetSQL, orderByKeys, orderSQL, type SortKey, type Tiebreak, };

// query/sqlite/pattern.d.ts
/**
 * The one lowering of `like` / `ilike` for SQLite. Every filter path calls this,
 * so the two operators cannot drift apart between builders.
 *
 * A value whose only `*` are at its ends is a fixed text to find, not a pattern (U21): it lowers
 * to `instr`, `substr` or `=`, which no length cap limits, where D1 refuses a LIKE or GLOB
 * pattern over 50 bytes. `like` stays case-sensitive, as GLOB is; `ilike` compares both sides
 * through `lower()`, ASCII as LIKE folds. A NULL column matches neither, as it matches no pattern.
 */
import { type SQL, type SQLWrapper } from "drizzle-orm";
/** `like` → GLOB (case-sensitive). `ilike` → LIKE (ASCII case-insensitive). A fixed text → a search for it. */
declare function patternToSQL(column: SQLWrapper, operator: "like" | "ilike", value: unknown): SQL;
export { patternToSQL };

// query/sqlite/search.d.ts
/**
 * FTS5 + spellfix search helpers for SQLite. User text is bound, never spliced: each term becomes
 * a quoted prefix phrase (`"term"*`), in which FTS5 reads no syntax, and identifiers go through quoteIdent.
 */
import { type SQL } from "drizzle-orm";
import type { SQLiteColumn, SQLiteTable } from "drizzle-orm/sqlite-core";
/**
 * A search term as its quoted phrase searches it: every word and character kept, as FTS5 reads
 * none of them as syntax inside the quotes, control characters removed. "" when no letter or digit
 * is left, as no token matches it. Quotes stay: the phrase builder escapes them where it quotes the term.
 */
declare function sanitizeFtsTerm(term: string): string;
/**
 * `<fts table> MATCH ?`, every term required; null for an empty query, no row for one only FTS
 * syntax spells. `folded` when the FTS table indexes a searchText column: each term is folded so
 */
declare function buildFtsMatch(ftsTableName: string, query: string | null, { folded }?: {
    folded?: boolean | undefined;
}): SQL | null;
/**
 * A table and the FTS5 table that indexes it by rowid (external content, `content_rowid` its
 * rowid or its integer key)
 */
export type FtsSource = {
    /**
     * The FTS table indexes a searchText column (`_search`): each term is folded with searchText as
     * the column was, so ß, æ, a ligature or a Greek accent typed as written finds it (U16)
     */
    folded?: boolean | undefined;
    ftsTable: string;
    table: SQLiteTable;
};
/**
 * The rows of `source.table` whose FTS entry matches every term; null for an empty query, no row
 * for one only FTS syntax spells
 */
declare function buildFtsWhere(source: FtsSource, query: string | null): SQL | null;
/** `highlight()` of one FTS column, its marker tags bound */
declare function buildFtsHighlight(ftsTableName: string, columnIndex: number, tags: {
    end: string;
    start: string;
}): SQL;
export type SpellfixRow = {
    word: string;
};
type SpellfixDb = {
    all: (query: SQL) => Promise<Array<SpellfixRow>>;
    /** the client drizzle's wrapper holds (d1, libsql, bun:sqlite), which keys the spellfix1 probe */
    $client?: unknown;
};
type SpellfixSearch = FtsSource & {
    /** editdist3 bound below which a word counts as a correction (default 300) */
    maxEditDistance?: number | undefined;
    /** corrections taken per term (default 3) */
    maxCorrections?: number | undefined;
    /** false where spellfix1 is never loaded: the terms are searched as typed and nothing is probed (Q4 of r20); a D1 client is known without it (Q3 of r21) */
    spellfix?: boolean | undefined;
    spellfixTable: string;
};
/**
 * Like buildFtsWhere, with each term of three or more characters widened by its spellfix1
 * corrections: every term is required, each matched by itself or any of its corrections.
 */
declare function buildFtsWhereWithSpellfix(db: SpellfixDb, search: SpellfixSearch, query: string | null): Promise<SQL | null>;
/**
 * Resolver that converts a raw search string into a LIKE WHERE clause.
 * Returns null when input is empty — callers can skip the clause entirely.
 */
type LikeSearchResolver = (q: string) => SQL | null;
/**
 * Factory for search on a `_search` column that searchText wrote, or any other SQL expression. Each
 * whitespace-separated term is folded by searchText and trimmed, and kept only when it holds a letter
 * or digit, as FTS keeps its terms: a mark alone folds to nothing or a space, and would match every
 * row. Each term is found with `instr` in the expression's `lower()`, so no LIKE pattern is built and
 * none meets D1's 50-byte pattern cap (U21), and any expression is matched by ASCII case as LIKE
 * matched it (QE2 of r22b); only a searchText column matches its accents and other folds. Every
 * term is required. A query whose terms all fold away matches no row, as FTS answers it.
 */
declare function likeSearch(column: SQLiteColumn | SQL): LikeSearchResolver;
export { buildFtsHighlight, buildFtsMatch, buildFtsWhere, buildFtsWhereWithSpellfix, likeSearch, type LikeSearchResolver, sanitizeFtsTerm, type SpellfixDb, type SpellfixSearch, };

// query/sqlite/search-text.d.ts
/**
 * The text a `_search` column stores for `parts`, and what likeSearch folds each term to: the
 * present parts joined by a space, compatibility forms folded (ＦＩＬＥ, ﬁ), lowercased, Greek final
 * sigma ς as σ (so a letter folds the same wherever it stands), combining marks removed but those
 * that make a Cyrillic letter (й, ї, ё), the Latin letters with no decomposition spelled in ASCII
 * (ß ss, æ ae, œ oe, ø o, ł l, đ and ð d, þ th, ı i, ĸ k, ŋ n, ŧ t, ħ h), then composed again (NFC),

 * are one function.
 */

// query/sqlite/refold.d.ts
/**
 * A `_search` column refolded with searchText (C15 of the 0.14.1 review). searchText folds in JS,
 * which no SQL can express, so a backfill reads each row and writes its fold back: `select` reads
 * every row's rowid, its sources and the text it stores, and `updates` turns those rows into one
 * UPDATE per row whose fold differs from what it stores, so a second run writes nothing and a D1
 * database pays a row write only where one is due. Both are plain SQL, run on any driver, in
 * batches of the caller's size. The table needs its rowid, which every table but a WITHOUT ROWID
 * one has.
 */
import { type SQL } from "drizzle-orm";
import type { SQLiteColumn, SQLiteTable } from "drizzle-orm/sqlite-core";
/** The statements of one refold: the read, and the writes its rows call for */
type Refold = {
    select: SQL;
    updates(rows: ReadonlyArray<Record<string, unknown>>): SQL[];
};
/** The refold of `column` from `sources`, in that order, on `table` */
declare function refoldStatements(table: SQLiteTable, column: SQLiteColumn, sources: readonly SQLiteColumn[]): Refold;
export { type Refold, refoldStatements };

/** The options the SQLite list builders take, and the query they build (build-list-query.ts). Declarations only. */

import type { SQLiteColumn, SQLiteTable } from "drizzle-orm/sqlite-core";

import type { ComputedFilterResolver, ComputedSortResolver, ListQueryInput } from "../types.ts";
import type { LikeSearchResolver } from "./search.ts";
import type { RelationConfig } from "./types.ts";
type BuildListQueryOpts<TTable extends SQLiteTable> = {
    /** Additional WHERE clause ANDed with filter/cursor — for joins, access filters, soft deletes */
    baseWhere?: SQL | ((sql: (strings: TemplateStringsArray, ...values: unknown[]) => SQL) => SQL) | undefined;
    computedFilters?: Record<string, ComputedFilterResolver> | undefined;
    computedSorts?: Record<string, ComputedSortResolver> | undefined;
    /** the list's scope, its route and tenant: a cursor minted in another opens in none here */
    cursorScope: string;
    /** the app's secret a cursor is sealed under, of at least 32 UTF-8 bytes */
    cursorSecret: string;
    /** "exact" when the driver reads ±Infinity back as Infinity, so a null cursor is a real NULL (CursorOptions) */
    nonFinite?: "exact" | undefined;
    /**
     * A list whose rows sit on several D1 shards: every shard reads it with this one query, and
     * paginateMerged merges what they read (K5 of r17). A page by number reads every row to its end
     * from each shard, past `maxRowsPerShard` (1001) a 400 `page_too_deep`
     */
    merged?: {
        maxRowsPerShard?: number | undefined;
    } | undefined;
    parsed: ListQueryInput;
    relations?: Record<string, RelationConfig> | undefined;
    /** LIKE-based search resolver — use `likeSearch(column)` factory */
    search?: LikeSearchResolver | undefined;
    table: TTable;
};
type BuildDerivedListQueryOpts<TSortField extends string = string> = {
    baseWhere?: SQL | ((sql: (strings: TemplateStringsArray, ...values: unknown[]) => SQL) => SQL) | undefined;
    /** the list's scope, its route and tenant: a cursor minted in another opens in none here */
    cursorScope: string;
    /** the app's secret a cursor is sealed under, of at least 32 UTF-8 bytes */
    cursorSecret: string;
    /**
     * the database the list reads, needed only where its id and keys hold no column: a column says it
     * wherever it sits, and a list of SQL expressions alone (a union, say) cannot (Q1 of r21)
     */
    dialect?: "postgres" | "sqlite" | undefined;
    filterWhere?: SQL | null | undefined;
    /**
     * Present when the rows come from several shards, which paginateMerged merges, as on
     * buildListQuery: each shard reads its window, an offset page its count. A merged derived list
     * carries no sort key by reference, and its rows carry each sort field under its name (Q12 of r18).
     */
    merged?: {
        maxRowsPerShard?: number | undefined;
    } | undefined;
    parsed: ListQueryInput<TSortField>;
    searchWhere?: SQL | null | undefined;
    /** each sort field's expression; one that yields epoch milliseconds is `{ sql, type: "date" }`, and its cursor value a Date */
    sortColumns: Record<TSortField, SQLWrapper | {
        sql: SQLWrapper;
        type: "date";
    }>;
} & ({
    /**
     * the one table the rows are, by its id column: a sort key too long for a cursor is
     * carried by reference and read back from it, as on a table's own list (C3)
     */
    anchor: {
        idColumn: SQLiteColumn;
        table: SQLiteTable;
    };
    idColumn?: never | undefined;
} | {
    anchor?: never | undefined;
    /** the id the order breaks ties on; with no table to read a key back from, a key too long for a cursor is a 500 */
    idColumn: SQLWrapper;
});
/**
 * The columns a built list's rows must carry beside the app's own, for `paginate` to read: select
 * them with the rest (`db.select({ ...cols, ...built.extras })`), and paginate strips them (Q2)
 */
type ListExtras = Partial<{
    /** 1 when the cursor's anchor row is gone where no bound is safe, which paginate answers invalid_cursor (U26) */
    _combAnchorGone: SQL.Aliased<number>;
    /** the value the list breaks ties on, which the next cursor resumes from; none when no cursor can */
    _combId: SQL.Aliased<unknown>;
    /** the class of each sort key and the tiebreak, comma-joined, where the list binds them as read or one is an undeclared SQLite expression (Q2 of r19, Q2 of r21) */
    _combKeyTypes: SQL.Aliased<string>;
    /** each sort key's stored time where the list binds it as read, comma-joined, empty for any other key: a Postgres time key's text, a SQLite REAL seconds (QL2 of r22b) */
    _combKeyTexts: SQL.Aliased<string>;
    /** each SQLite timestamp key's TEXT where the list reads it back and the row holds TEXT, a JSON array with null for any other key, never comma-joined (B2-3 of area B) */

     * this object included (C7 of r19). A parsed object built by hand leaves it out for the default
     * maxPage, or states the bound it means (query/trusted.ts, U29)
     */
    pageBound?: number | undefined;
    /** The schema's own maxLimit, which holds the limit as `pageBound` holds the page (Q16 of r18, C7 of r19) */
    limitBound?: number | undefined;
    /** The column the schema says the list breaks ties on, `id` by default, which a derived list's idColumn names */
    tiebreak?: string | undefined;
    /** Always true: the filter and the order were checked against the endpoint's fields (query/trusted.ts) */
    validated: true;
};
type RetrieveQueryOutput = {
    lang: string | undefined;
    parsedFields: ParsedFields | null;
};

// query/filter.d.ts
/**
 * PostgREST-style filters and sort orders: parse and validate. DB-agnostic: no SQL, only
 * the AST and its checks. The grammar lives in filter-parse.ts, the checks in filter-validate.ts.
 */
import { FILTER_OPERATORS, parseFilter } from "./filter-parse.ts";
import { parseFilterNumber } from "./filter-typed.ts";
import { FILTER_LIMITS, OPERATORS_BY_TYPE, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, type LimitsGiven, validateFilter } from "./filter-validate.ts";
import type { SortField } from "./types.ts";
/**
 * Parse PostgREST-style `order` string into SortField array.
 *
 * Grammar:
 *   order  := entry ("," entry)*
 *   entry  := field ("." direction)? ("." nulls)?
 *   field  := any string not containing "." at the direction/nulls boundary
 *            (dotted relation paths OK: "author.name.asc")
 *   direction := "asc" | "desc"        (default "asc" when omitted)
 *   nulls     := "nullsfirst" | "nullslast"
 *
 * Examples:
 *   "name"                         → [{ field: "name", direction: "asc" }]
 *   "name.desc"                    → [{ field: "name", direction: "desc" }]
 *   "name.asc.nullslast"           → [{ field: "name", direction: "asc", nulls: "last" }]
 *   "col1.desc,col2.asc.nullsfirst"
 *     → [{field:"col1",direction:"desc"}, {field:"col2",direction:"asc",nulls:"first"}]
 *
 * An entry that is empty, names no field (`asc`, `.desc`, a trailing comma) or has whitespace around
 * it, and a field named twice, refuse the whole order as null, as parseFilter refuses a filter, rather
 * than drop it and widen the request to the default order (QL7 of r22b).
 */
declare function parseOrder(input: string | null | undefined): SortField[] | null;
/** Each field `fields` names more than once, once each */
declare function namedTwice(fields: readonly SortField[]): string[];
/**
 * Each entry of `order` read as a sort field, or kept as its text when it is empty, names no field or
 * has whitespace around it, which the published pattern refuses too
 */
declare function orderEntries(input: string | null | undefined): Array<SortField | string>;
export { FILTER_LIMITS, FILTER_OPERATORS, OPERATORS_BY_TYPE, parseFilter, parseFilterNumber, namedTwice, orderEntries, parseOrder, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, type LimitsGiven, };

// query/filter-validate.d.ts
import type { FieldType, FilterAST, FilterOperator } from "./types.ts";
declare const OPERATORS_BY_TYPE: Record<FieldType, readonly FilterOperator[]>;
/**
 * A field as a runtime field map declares it: its type, the operators it takes when not its
 * type's (`OPERATORS_BY_TYPE`), and for a date, `format: "iso"` to take ISO 8601 text, as a
 * column that stores it holds, in place of epoch milliseconds
 */
type FilterFieldSpec = {
    format?: "iso" | undefined;
    operators?: readonly FilterOperator[] | undefined;
    type: FieldType;
};
/** A field of `validateFilter`'s map: a bare type, or a spec */
type FilterField = FieldType | FilterFieldSpec;
type FilterValidationResult = {
    ast: FilterAST;
    valid: true;
} | {
    errors: string[];
    valid: false;
};
/** Bounds on one filter expression. Each condition is a predicate SQLite evaluates per row. */
type FilterLimits = {
    maxConditions: number;
    /** Nested or()/and() groups below the top level */
    maxDepth: number;
    maxInValues: number;
    /** Bytes of a like or ilike pattern, a value with an inner `*`, once escaped: D1 caps a pattern at 50 */
    maxPatternBytes: number;
    /** Values the whole filter binds, every in/nin member counted: D1 binds at most 100 per statement */
    maxValues: number;
};
declare const FILTER_LIMITS: FilterLimits;
/** A limits bag as a caller gives it: any key left out or set to undefined keeps its default (KT1 of r22b) */
type LimitsGiven<T> = {
    [K in keyof T]?: number | undefined;
};
/**
 * `given` over `defaults`, a key set to undefined left out as every other option bag reads it, each
 * other a whole number from 0, a TypeError naming it and `owner` otherwise (KT1 of r22b): a cap of
 * NaN or undefined compared false, which turned it off. No cap is a large whole number
 */
declare function limitsOver<T extends Record<string, number>>(owner: string, defaults: T, given: LimitsGiven<T> | undefined): T;
/**
 * Parse `filterString` and check it against `allowedFields`: known fields, operators their type
 * or spec allows, values of their type, within `limits`. Null when there is no filter. A spec
 * that lists an operator its type cannot take, or a format on no date, throws a TypeError.
 */
declare function validateFilter(filterString: string | null | undefined, allowedFields: Record<string, FilterField>, limits?: LimitsGiven<FilterLimits> | undefined): FilterValidationResult | null;
export { FILTER_LIMITS, limitsOver, OPERATORS_BY_TYPE, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type LimitsGiven, type FilterValidationResult, };

// query/filter-parse.d.ts
/**
 * The PostgREST-style filter grammar: a filter string to its AST. Values stay as the request
 * wrote them; `validateFilter` checks them against the declared field types.
 *
 * - `field.op.value`: one condition; `,` joins conditions with AND
 * - `or(...)`, `and(...)`: groups, nestable
 * - `in.(a,b)`, `nin.(a,b)`: lists
 * - `"..."`: a quoted value may hold `,` `(` `)` and `.`; `\"` and `\\` escape inside it
 * - `@name`: a computed field
 */
import type { FilterAST, FilterOperator } from "./types.ts";
declare const FILTER_OPERATORS: readonly [FilterOperator, ...FilterOperator[]];
/**
 * The AST of a filter string, or null when there is none or any part of it cannot be read: a
 * filter the parser does not fully understand is refused, never reduced to "no filter", which
 * would widen the result to the whole table on a typo.
 */
declare function parseFilter(input: string | null | undefined): FilterAST | null;
export { FILTER_OPERATORS, parseFilter };
