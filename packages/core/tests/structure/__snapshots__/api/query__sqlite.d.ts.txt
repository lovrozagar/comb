// query/sqlite/condition.d.ts
/**

 * and buildFilterSQL (a table alias's columns by name) share, so no operator or value conversion

// query/sqlite/keyset.d.ts
/**
 * Ordering and keyset predicates — one implementation for every SQLite builder.
 *
 * ORDER BY and the cursor predicate must agree on direction AND null placement
 * for every key, or pagination skips and repeats rows. Both are derived here
 * from the same `SortKey[]`, so they cannot disagree.
 */
import { type SQL, type SQLWrapper } from "drizzle-orm";
import type { SortDirection, SortNulls } from "../types.ts";
/** One key of a list's order */
type SortKey = {
    /** the column or expression sorted on */
    column: SQLWrapper;
    /** its direction */
    direction: SortDirection;
    /** where its nulls sort; left out, last ascending and first descending */
    nulls?: SortNulls | undefined;
};
/** The unique column that breaks ties after every sort key, so the order is total */
type Tiebreak = {
    /** the column; unique and never null, or a page repeats or skips rows */
    column: SQLWrapper;
    /** its direction */
    direction: SortDirection;
};
/** `col ASC NULLS LAST` — explicit placement so SQLite's default never leaks in. */
declare function orderSQL(key: SortKey): SQL;
/** ORDER BY for the keys followed by the tiebreak, which makes the order total. */
declare function orderByKeys(keys: readonly SortKey[], tiebreak: Tiebreak): SQL[];
/** A count as a SQL literal, where a bound parameter would cost the D1 budget one */
declare const countLiteral: (value: number) => SQL;
/** An integer, a JS number's or a bigint's, as a SQL literal, where a bound parameter would cost the D1 budget one (Q15 of r17) */
declare const integerLiteral: (value: bigint | number) => SQL;
/**
 * A bound the keyset reads from the cursor's own row (cursor-ref.ts) rather than binds: a long text
 * value carried by reference, or a seconds-mode timestamp, whose fraction no Date holds (`row`); or
 * a SQLite number the cursor carried as null, which may be a ±Infinity D1 hands over as null. Each is SQL that binds, so the keyset reads it twice at most;
 * `ends` when a NULL bound means no bound is safe, and the list ends there.
 */
declare class AnchorBound {
    readonly ends: boolean;
    readonly kind: "infinite" | "row";
    readonly value: SQL;
    constructor(kind: "infinite" | "row", value: SQL, ends?: boolean);
}
/** Where a cursor points: the sort keys and the row's values for them, then the tiebreak and its id */
type KeysetPosition = {
    /** the row's tiebreak value */
    id: unknown;
    /** the list's sort keys, in order */
    keys: readonly SortKey[];
    /** the list's tiebreak */
    tiebreak: Tiebreak;
    /** the row's value for each key, in the keys' order; null for a NULL */
    values: readonly unknown[];
};
/**
 * The WHERE clause of the rows after `position` in the list's order: every key, then the tiebreak,
 * each in its own direction and null placement, so it agrees with `orderSQL` and the page after
 * repeats and skips no row
 */
declare function keysetSQL(position: KeysetPosition): SQL;
/**
 * 1 where a bound with no safe fallback is NULL, its anchor row gone or renamed, else 0; undefined
 * when no bound can be. It reads the anchor row alone, so Postgres plans it once a query (U26).
 */
declare function anchorGone(values: readonly unknown[]): SQL | undefined;
export { AnchorBound, anchorGone, countLiteral, integerLiteral, type KeysetPosition, keysetSQL, orderByKeys, orderSQL, type SortKey, type Tiebreak, };

// query/sqlite/pattern.d.ts
/**
 * The one lowering of `like` / `ilike` for SQLite. Every filter path calls this,
 * so the two operators cannot drift apart between builders.
 *
 * A value whose only `*` are at its ends is a fixed text to find, not a pattern (U21): it lowers
 * to `instr`, `substr` or `=`, which no length cap limits, where D1 refuses a LIKE or GLOB
 * pattern over 50 bytes. `like` stays case-sensitive, as GLOB is; `ilike` compares both sides
 * through `lower()`, ASCII as LIKE folds. A NULL column matches neither, as it matches no pattern.
 */
import { type SQL, type SQLWrapper } from "drizzle-orm";
/** `like` → GLOB (case-sensitive). `ilike` → LIKE (ASCII case-insensitive). A fixed text → a search for it. */
declare function patternToSQL(column: SQLWrapper, operator: "like" | "ilike", value: unknown): SQL;
export { patternToSQL };

 * a quoted prefix phrase (`"term"*`), in which FTS5 reads no syntax, and identifiers go through quoteIdent.
 */

import type { SQLiteColumn, SQLiteTable } from "drizzle-orm/sqlite-core";
/**
 * A search term as its quoted phrase searches it: every word and character kept, as FTS5 reads
 * none of them as syntax inside the quotes, control characters removed. "" when no letter or digit
 * is left, as no token matches it. Quotes stay: the phrase builder escapes them where it quotes the term.
 */
declare function sanitizeFtsTerm(term: string): string;

    /** true when the FTS table indexes a searchText column: each term is folded as the column was; default false */
    folded?: boolean | undefined;
}): SQL | null;

 * rowid or its integer key)
 */
export type FtsSource = {
    /**
     * The FTS table indexes a searchText column (`_search`): each term is folded with searchText as
     * the column was, so ß, æ, a ligature or a Greek accent typed as written finds it
     */
    folded?: boolean | undefined;
    /** the FTS5 table's name */
    ftsTable: string;
    /** the table it indexes */
    table: SQLiteTable;
};
/**
 * The rows of `source.table` whose FTS entry matches every term; null for an empty query, no row
 * for one only FTS syntax spells
 */
declare function buildFtsWhere(source: FtsSource, query: string | null): SQL | null;
/** `highlight()` of one FTS column, its marker tags bound */
declare function buildFtsHighlight(ftsTableName: string, columnIndex: number, tags: {
    /** the text after each match */
    end: string;
    /** the text before each match */
    start: string;
}): SQL;
/** A row of a spellfix1 vocabulary query */
export type SpellfixRow = {
    /** a word the vocabulary offers */
    word: string;
};
/** The database `buildFtsWhereWithSpellfix` asks for corrections: a drizzle SQLite database */
type SpellfixDb = {
    /** runs a query and resolves to its rows */
    all: (query: SQL) => Promise<Array<SpellfixRow>>;
    /** the client drizzle's wrapper holds (d1, libsql, bun:sqlite), which keys the spellfix1 probe */
    $client?: unknown;
};
/** An FTS search whose terms are widened by a spellfix1 vocabulary's corrections */
type SpellfixSearch = FtsSource & {
    /** editdist3 bound below which a word counts as a correction (default 300) */
    maxEditDistance?: number | undefined;
    /** corrections taken per term (default 3) */
    maxCorrections?: number | undefined;
    /** false where spellfix1 is never loaded: the terms are searched as typed and nothing is probed; a D1 client is known without it */
    spellfix?: boolean | undefined;
    /** the spellfix1 vocabulary table's name */
    spellfixTable: string;
};

 * corrections: every term is required, each matched by itself or any of its corrections.
 */

/** A search: the `q` text in, its WHERE clause out; null when it searches nothing */
type LikeSearchResolver = (q: string) => SQL | null;
/**
 * The search of a `_search` column that searchText wrote, or any other SQL expression. Each
 * whitespace-separated term is folded by searchText and trimmed, and kept only when it holds a letter
 * or digit, as FTS keeps its terms: a mark alone folds to nothing or a space, and would match every
 * row. Each term is found with `instr` in the expression's `lower()`, so no LIKE pattern is built and
 * none meets D1's 50-byte pattern cap, and any expression is matched ignoring ASCII case; only a

 * all fold away matches no row, as FTS answers it.
 */
declare function likeSearch(column: SQLiteColumn | SQL): LikeSearchResolver;

// query/sqlite/refold.d.ts
/**
 * A `_search` column refolded with searchText (C15 of the 0.14.1 review). searchText folds in JS,
 * which no SQL can express, so a backfill reads each row and writes its fold back: `select` reads
 * every row's rowid, its sources and the text it stores, and `updates` turns those rows into one
 * UPDATE per row whose fold differs from what it stores, so a second run writes nothing and a D1
 * database pays a row write only where one is due. Both are plain SQL, run on any driver, in
 * batches of the caller's size. The table needs its rowid, which every table but a WITHOUT ROWID
 * one has.
 */
import { type SQL } from "drizzle-orm";
import type { SQLiteColumn, SQLiteTable } from "drizzle-orm/sqlite-core";
/** The statements of one refold: the read, and the writes its rows call for */
type Refold = {
    /** reads every row's rowid, stored fold and sources */
    select: SQL;
    /** one UPDATE per row `select` read whose stored fold differs from its sources' fold; none for a row already folded */
    updates(rows: ReadonlyArray<Record<string, unknown>>): SQL[];
};
/** The refold of `column` from `sources`, in that order, on `table` */
declare function refoldStatements(table: SQLiteTable, column: SQLiteColumn, sources: readonly SQLiteColumn[]): Refold;
export { type Refold, refoldStatements };

// query/types.d.ts
/**
 * Query helper types — DB-agnostic.
 * Shared by parsers, validators, and DB-specific SQL generators.
 */
import type { SQL } from "drizzle-orm";
/** The fields one level of a `select` names, as `parseSelect` reads it */
type FieldSelection = {
    /** each relation embedded as `name(...)`, by name, with what it selects; null selects every field of it */
    relations: Record<string, FieldSelection | null>;
    /** the field names at this level, each once, in the order written; `*` stands for every field, a leading `@` for a computed one */
    scalars: string[];
};
/** A parsed `select`; `filterBySelect` projects a row by it */
type ParsedFields = {
    /** what the top level selects */
    root: FieldSelection;
};
/**
 * A filter condition's operator, as written between the field and the value: `ne` and `neq` both
 * mean not equal, `in` and `nin` take a parenthesized list, `like` is case-sensitive and `ilike`
 * ignores ASCII case (both with `*` as the wildcard), and `is` takes `null`, `notnull`, `true` or `false`
 */
type FilterOperator = "eq" | "ne" | "neq" | "gt" | "gte" | "lt" | "lte" | "in" | "nin" | "like" | "ilike" | "is"
/** Parsed for callers that validate and run it themselves (JSON-array membership); comb's own validator and SQL take none */
 | "contains";
/** A filterable field's type, which decides the operators it takes and how its value is read: a `date` as epoch milliseconds unless its spec says `iso` */
type FieldType = "string" | "number" | "boolean" | "date" | "enum";
/** One `field.operator.value` term of a filter */
type FilterCondition = {
    /** the field as written: a column, `relation.column`, or a `@`-prefixed computed field */
    field: string;
    /** the operator as written; `ne` and `neq` are one */
    operator: FilterOperator;
    /** the value as parsed: a string, null for `is.null`, or an array of strings for a list; `validateFilter` types it as its field */
    value: unknown;
};
/** Conditions and nested groups joined by one logic */
type FilterGroup = {
    /** the group's own terms, in the order written */
    conditions: FilterCondition[];
    /** how the group's conditions and subgroups combine; the top level is `and` */
    logic: "and" | "or";
    /** each nested `and(...)` or `or(...)`, in the order written */
    subgroups: FilterGroup[];
};
/** A parsed filter, as `parseFilter` and `validateFilter` return it */
type FilterAST = {
    /** the top-level group, joined by its own `logic`; `parseFilter` and `validateFilter` make it an `and` */
    root: FilterGroup;
};
/** A sort key's direction */
type SortDirection = "asc" | "desc";
/** Where a sort key puts nulls; left out, they sort last ascending and first descending, as on Postgres */
type SortNulls = "first" | "last";
/** One key of an `order`, as `parseOrder` returns it */
type SortField<T extends string = string> = {
    /** the direction as written; `asc` when the key names none */
    direction: SortDirection;
    /** the field sorted on: a column, or a `@`-prefixed computed field */
    field: T;
    /** null placement as written; left out, the direction decides it (see SortNulls) */
    nulls?: SortNulls | undefined;
};
/**
 * What every list builder takes (buildListQuery, buildDerivedListQuery, queryHelpers.listOptions):
 * the list schema's output, or a query the app builds itself. The builders never change its arrays
 */
type ListQueryInput<TSortField extends string = string> = {
    /** the previous page's next cursor, to read the page after it; null or absent reads the first page */
    cursor?: string | null | undefined;
    /** the parsed filter; null or absent filters nothing */
    filterAst?: FilterAST | null | undefined;
    /** the most rows the page holds, a whole number from 1 */
    limit: number;
    /** the 1-based page to read by number; null or absent is page 1. A cursor, when given, wins */
    page?: number | null | undefined;
    /** the deepest page this query may read; absent, the default maxPage. A deeper page is a 400 */
    pageBound?: number | undefined;
    /** the largest limit this query may take; absent, the default maxLimit. A larger limit is a 400 */
    limitBound?: number | undefined;
    /** the column the list schema breaks ties on; a derived list's idColumn must name it */
    tiebreak?: string | undefined;
    /** the parsed `select`; null or absent selects every field */
    parsedFields?: ParsedFields | null | undefined;
    /** the sort keys in order; empty sorts by the tiebreak column alone */
    parsedSort: readonly SortField<TSortField>[];
    /** the search text, trimmed; null, absent or blank searches nothing */
    q?: string | null | undefined;
    /**
     * The list schema sets it. A builder throws a 500 on a filter or an order without it, as neither
     * was checked against the fields the endpoint declares; set it only on ones your code checked
     */
    validated?: true | undefined;
};
/** Lowers a computed filter field's condition to SQL; null when it does not take the operator, which the builder answers with a 500 */
type ComputedFilterResolver = (operator: FilterOperator, value: unknown) => SQL | null;
/** The ORDER BY term of a computed sort field in `direction`; a list sorted on one pages by number, and a cursor on it is a 400 */
type ComputedSortResolver = (direction: SortDirection) => SQL;
export type { ComputedFilterResolver, ComputedSortResolver, FieldSelection, FieldType, FilterAST, FilterCondition, FilterGroup, FilterOperator, ListQueryInput, ParsedFields, SortDirection, SortField, SortNulls, };

// query/sqlite/apply-list-query.d.ts
/** The one way a built list query reaches a drizzle select: its extras, WHERE, ORDER BY, LIMIT and OFFSET. */
import type { SQL } from "drizzle-orm";
import type { ListExtras, ListQueryResult } from "./build-list-query-types.ts";
/** A builder whose methods return its own type, as drizzle's select does once made dynamic */
type ListQueryAppliable<TQuery> = {
    /** LIMIT: the rows to read */
    limit: (limit: number) => TQuery;
    /** OFFSET: the rows to skip */

    /** ORDER BY the terms, in order */

 * minted from it compares as one, a string binding as text, which SQLite orders after every
 * integer; a REAL ±Infinity D1's transport read as null as itself (Q3 of r20); a time key its
 * stored value `_combKeyTexts` carries (QL2 of r22b), a SQLite timestamp holding TEXT the TEXT
 * `_combTextKeys` carries (B2-3 of area B), and a SQLite boolean holding a value but 0 or 1 the value
 * it carries (AB3-2 of area B). A text key that looks like a number stays
 * text. The merge compares rows through it (Q2 of r20).
 */
declare function typedKeys(row: Record<string, unknown>, sort: readonly SortField[]): Record<string, unknown>;
/**
 * The sort fields whose value in `typedKeys(row, sort)` is the stored text `_combKeyTexts` or `_combTextKeys` carried in
 * place of the value as read (comb-9d's AB2 ruling), which a cursor binds as itself, past the
 * column's mapper; every other key binds through it, a text key that looks like a number included
 */
declare function carriedTexts(row: Record<string, unknown>, sort: readonly SortField[]): Set<string>;
export { carriedTexts, COMB_ID, INTEGER_TEXT, type KeyClass, malformed, refuseUnheld, storedClasses, typedKeys, unencoded, };

/**
 * PostgREST-style filters and sort orders: parse and validate. DB-agnostic: no SQL, only
 * the AST and its checks. The grammar lives in filter-parse.ts, the checks in filter-validate.ts.
 */
import { FILTER_OPERATORS, parseFilter } from "./filter-parse.ts";
import { parseFilterNumber } from "./filter-typed.ts";

/**
 * Parse PostgREST-style `order` string into SortField array.
 *
 * Grammar:
 *   order  := entry ("," entry)*
 *   entry  := field ("." direction)? ("." nulls)?
 *   field  := any string not containing "." at the direction/nulls boundary
 *            (dotted relation paths OK: "author.name.asc")
 *   direction := "asc" | "desc"        (default "asc" when omitted)
 *   nulls     := "nullsfirst" | "nullslast"
 *
 * Examples:
 *   "name"                         → [{ field: "name", direction: "asc" }]
 *   "name.desc"                    → [{ field: "name", direction: "desc" }]
 *   "name.asc.nullslast"           → [{ field: "name", direction: "asc", nulls: "last" }]
 *   "col1.desc,col2.asc.nullsfirst"
 *     → [{field:"col1",direction:"desc"}, {field:"col2",direction:"asc",nulls:"first"}]
 *
 * An entry that is empty, names no field (`asc`, `.desc`, a trailing comma) or has whitespace around
 * it, and a field named twice, refuse the whole order as null, as parseFilter refuses a filter, rather
 * than drop it and widen the request to the default order.
 */
declare function parseOrder(input: string | null | undefined): SortField[] | null;
/** Each field `fields` names more than once, once each */
declare function namedTwice(fields: readonly SortField[]): string[];
/**
 * Each entry of `order` read as a sort field, or kept as its text when it is empty, names no field or
 * has whitespace around it, which the published pattern refuses too
 */
declare function orderEntries(input: string | null | undefined): Array<SortField | string>;
export { FILTER_LIMITS, FILTER_OPERATORS, OPERATORS_BY_TYPE, parseFilter, parseFilterNumber, namedTwice, orderEntries, parseOrder, validateFilter, type FilterField, type FilterFieldSpec, type FilterLimits, type FilterValidationResult, type LimitsGiven, };

 * (1e-400 reads as 0, which would match the rows holding 0), and within
 * ±(2^53 − 1), past which a double binds a neighbouring value. The value is read, not its
 * spelling: `9007199254740993.0` and `9.007199254740993e15` are the same unsafe integer.
 */
declare function parseFilterNumber(raw: string): {
    /** the number `raw` names */
    value: number;
} | {
    /** why `raw` is no number a filter takes */
    error: string;
};
declare function typedValue(field: CheckedField, { operator, value }: FilterCondition): Typed;
/** A field of the map with its operators resolved */
type CheckedField = {
    format: "iso" | undefined;
    operators: readonly FilterOperator[];
    type: FieldType;
};
export { parseFilterNumber, typedValue, type CheckedField };
