// query/sqlite/build-list-query.d.ts
import { type SQL, type SQLWrapper } from "drizzle-orm";
import type { SQLiteTable } from "drizzle-orm/sqlite-core";
import type { ComputedFilterResolver, ComputedSortResolver, FilterAST, SortField } from "../types.ts";
import type { LikeSearchResolver } from "./search.ts";
import type { RelationConfig } from "./types.ts";
/**
 * Parsed list query input — matches createListQuerySchema output.
 * Intentionally loose to work with any codegen or manual construction.
 */
type ListQueryParsed = {
    cursor?: string | null | undefined;
    filterAst?: FilterAST | null | unknown;
    limit: number;
    page?: number | null | undefined;
    parsedSort: SortField[];
    q?: string | null | undefined;
    /** The list schema sets it; a filter or an order without it is refused (query/trusted.ts) */
    validated?: true | undefined;
    [key: string]: unknown;
};
type BuildListQueryOpts<TTable extends SQLiteTable> = {
    /** Additional WHERE clause ANDed with filter/cursor — for joins, access filters, soft deletes */
    baseWhere?: SQL | ((sql: (strings: TemplateStringsArray, ...values: unknown[]) => SQL) => SQL);
    computedFilters?: Record<string, ComputedFilterResolver>;
    computedSorts?: Record<string, ComputedSortResolver>;
    /** the list's scope, its route and tenant: a cursor minted in another opens in none here */
    cursorScope: string;
    /** the app's secret a cursor is sealed under, of at least 32 UTF-8 bytes */
    cursorSecret: string;
    parsed: ListQueryParsed;
    relations?: Record<string, RelationConfig>;
    /** LIKE-based search resolver — use `likeSearch(column)` factory */
    search?: LikeSearchResolver;
    table: TTable;
};
type BuildDerivedListQueryOpts<TSortField extends string = string> = {
    baseWhere?: SQL | ((sql: (strings: TemplateStringsArray, ...values: unknown[]) => SQL) => SQL);
    /** the list's scope, its route and tenant: a cursor minted in another opens in none here */
    cursorScope: string;
    /** the app's secret a cursor is sealed under, of at least 32 UTF-8 bytes */
    cursorSecret: string;
    filterWhere?: SQL | null;
    idColumn: SQLWrapper;
    parsed: ListQueryParsed & {
        parsedSort: SortField<TSortField>[];
    };
    searchWhere?: SQL | null;
    /** each sort field's expression; one that yields epoch milliseconds is `{ sql, type: "date" }`, and its cursor value a Date */
    sortColumns: Record<TSortField, SQLWrapper | {
        sql: SQLWrapper;
        type: "date";
    }>;
};
type ListQueryResult = {
    limit: number;
    meta: {
        limit: number;
        page: number;
        type: "cursor" | "offset";
    };
    offset: number;
    orderBy: SQL[];
    search: string | null;
    where: SQL | undefined;
};
type ListQueryAppliable<TQuery> = {
    limit: (limit: number) => TQuery;
    offset: (offset: number) => TQuery;
    orderBy: (...orderBy: SQL[]) => TQuery;
    where: (where: SQL) => TQuery;
};
declare function buildListQuery<TTable extends SQLiteTable>(opts: BuildListQueryOpts<TTable>): ListQueryResult;
declare function buildDerivedListQuery<TSortField extends string>(opts: BuildDerivedListQueryOpts<TSortField>): ListQueryResult;
declare function applyListQuery<TQuery extends ListQueryAppliable<TQuery>>(query: TQuery, q: ListQueryResult): TQuery;
export { applyListQuery, buildDerivedListQuery, buildListQuery, type BuildDerivedListQueryOpts, type BuildListQueryOpts, type ListQueryParsed, type ListQueryResult, };

/**
 * SQLite query executor — Drizzle-native filter/sort/pagination orchestrator.
 */
import { type SQL } from "drizzle-orm";
import { type SQLiteColumn, type SQLiteTable } from "drizzle-orm/sqlite-core";
import type { ComputedFilterResolver, FilterAST, FilterCondition } from "../types.ts";
import type { RelationConfig } from "./types.ts";
/** The operators a date field takes with one number, which a seconds column rounds */
type SecondsOperator = "eq" | "gt" | "gte" | "lt" | "lte" | "ne" | "neq";
declare const isSecondsOperator: (operator: string) => operator is SecondsOperator;
/**
 * How a seconds column compares with epoch milliseconds `ms` under `operator`: the comparison on
 * the whole second that keeps it exact, or, for eq and ne on a value between two seconds, no row
 * ("none") or every row that has a value ("present"). The raw builder binds the seconds as they are.
 */
declare function secondsBound(operator: SecondsOperator, ms: number): "none" | "present" | {
    operator: "eq" | "gt" | "gte" | "lt" | "lte" | "ne";
    seconds: number;
};
/** One condition on one column; null when the operator or value is not one this column can take */
declare function conditionToSQL(condition: FilterCondition, column: SQLiteColumn): SQL | null;
/**
 * What a validated filter lowers against. Values arrive typed by `validateFilter` (or the list
 * query schema), so they bind as the field declares.
 */
type FilterToSQLConfig = {
    computedFilters?: Record<string, ComputedFilterResolver> | undefined;
    mainTable: SQLiteTable;
    relations?: Record<string, RelationConfig> | undefined;
};
declare function filterToSQL(ast: FilterAST | null, config: FilterToSQLConfig): SQL | null;
export { conditionToSQL, filterToSQL, type FilterToSQLConfig, isSecondsOperator, secondsBound };

/**
 * Raw SQL filter builders — table-alias-based SQL generation from FilterAST.
 * Used when building raw SQL queries (not Drizzle relational).
 */
import { type SQL } from "drizzle-orm";
import type { FilterCondition, FilterGroup } from "../types.ts";
/** A mapped column: its name, and for a date field the unit it stores (milliseconds unless `"s"`) */
type MappedColumn = string | {
    column: string;
    unit: "ms" | "s";
};
type FilterSQLConfig = {
    /** Column map: API field name -> SQL column name, or `{ column, unit: "s" }` for a date stored in seconds */
    columnMap: Record<string, MappedColumn>;
    /** Table alias used in query (e.g. "pc" for productCollection) */
    tableAlias: string;
};
/**
 * A filter group as SQL, its conditions and subgroups joined by its logic; an empty group is the
 * identity of its logic: or() matches nothing, and() everything
 */
declare function buildFilterSQL(group: FilterGroup, config: FilterSQLConfig): SQL;
/** One condition as SQL on its mapped column; null when the field is unmapped or the value unusable */
declare function buildConditionSQL(condition: FilterCondition, config: FilterSQLConfig): SQL | null;
export { buildConditionSQL, buildFilterSQL, type FilterSQLConfig, type MappedColumn };

// query/sqlite/keyset.d.ts
/**
 * Ordering and keyset predicates — one implementation for every SQLite builder.
 *
 * ORDER BY and the cursor predicate must agree on direction AND null placement
 * for every key, or pagination skips and repeats rows. Both are derived here
 * from the same `SortKey[]`, so they cannot disagree.
 */
import { type SQL, type SQLWrapper } from "drizzle-orm";
import type { SortDirection, SortNulls } from "../types.ts";
type SortKey = {
    column: SQLWrapper;
    direction: SortDirection;
    nulls?: SortNulls | undefined;
};
type Tiebreak = {
    column: SQLWrapper;
    direction: SortDirection;
};
/** `col ASC NULLS LAST` — explicit placement so SQLite's default never leaks in. */
declare function orderSQL(key: SortKey): SQL;
/** ORDER BY for the keys followed by the tiebreak, which makes the order total. */
declare function orderByKeys(keys: readonly SortKey[], tiebreak: Tiebreak): SQL[];
/** A count as a SQL literal, where a bound parameter would cost the D1 budget one */
declare const countLiteral: (value: number) => SQL;
/**
 * A bound the keyset reads from the cursor's own row (cursor-ref.ts) rather than binds: a long text
 * value carried by reference, or a seconds-mode timestamp, whose fraction no Date holds (`row`); or
 * a SQLite number the cursor carried as null, which may be a ±Infinity D1 hands over as null. Each is SQL that binds, so the keyset reads it twice at most;
 * `ends` when a NULL bound means no bound is safe, and the list ends there.
 */
declare class AnchorBound {
    readonly ends: boolean;
    readonly kind: "infinite" | "row";
    readonly value: SQL;
    constructor(kind: "infinite" | "row", value: SQL, ends?: boolean);
}
/** Where a cursor points: the sort keys and the row's values for them, then the tiebreak and its id */
type KeysetPosition = {
    id: unknown;
    keys: readonly SortKey[];
    tiebreak: Tiebreak;
    values: readonly unknown[];
};
/**
 * Lexicographic "after this row" over every key plus the tiebreak, expanded rather than a row-value
 * comparison because each key carries its own direction and null placement. A bound that `ends`
 * must not be NULL, or no row is after it.
 */
declare function keysetSQL(position: KeysetPosition): SQL;
export { AnchorBound, countLiteral, type KeysetPosition, keysetSQL, orderByKeys, orderSQL, type SortKey, type Tiebreak };

// query/sqlite/search.d.ts
/**
 * FTS5 + spellfix search helpers for SQLite. User text is bound, never spliced: each term becomes
 * a quoted prefix phrase (`"term"*`), in which FTS5 reads no syntax, and identifiers go through quoteIdent.
 */
import { type SQL } from "drizzle-orm";
import type { SQLiteColumn, SQLiteTable } from "drizzle-orm/sqlite-core";
/**
 * A search term as its quoted phrase searches it: every word and character kept, as FTS5 reads
 * none of them as syntax inside the quotes, control characters removed. "" when no letter or digit
 * is left, as no token matches it. Quotes stay: the phrase builder escapes them where it quotes the term.
 */
declare function sanitizeFtsTerm(term: string): string;
/**
 * `<fts table> MATCH ?`, every term required; null for an empty query, no row for one only FTS
 * syntax spells. `folded` when the FTS table indexes a searchText column: each term is folded so
 */
declare function buildFtsMatch(ftsTableName: string, query: string | null, { folded }?: {
    folded?: boolean | undefined;
}): SQL | null;
/**
 * A table and the FTS5 table that indexes it by rowid (external content, `content_rowid` its
 * rowid or its integer key)
 */
type FtsSource = {
    /**
     * The FTS table indexes a searchText column (`_search`): each term is folded with searchText as
     * the column was, so ß, æ, a ligature or a Greek accent typed as written finds it (U16)
     */
    folded?: boolean | undefined;
    ftsTable: string;
    table: SQLiteTable;
};
/**
 * The rows of `source.table` whose FTS entry matches every term; null for an empty query, no row
 * for one only FTS syntax spells
 */
declare function buildFtsWhere(source: FtsSource, query: string | null): SQL | null;
/** `highlight()` of one FTS column, its marker tags bound */
declare function buildFtsHighlight(ftsTableName: string, columnIndex: number, tags: {
    end: string;
    start: string;
}): SQL;
type SpellfixRow = {
    word: string;
};
type SpellfixDb = {
    all: (query: SQL) => Promise<Array<SpellfixRow>>;
};
type SpellfixSearch = FtsSource & {
    /** editdist3 bound below which a word counts as a correction (default 300) */
    maxEditDistance?: number;
    /** corrections taken per term (default 3) */
    maxCorrections?: number;
    spellfixTable: string;
};
/**
 * Like buildFtsWhere, with each term of three or more characters widened by its spellfix1
 * corrections: every term is required, each matched by itself or any of its corrections.
 */
declare function buildFtsWhereWithSpellfix(db: SpellfixDb, search: SpellfixSearch, query: string | null): Promise<SQL | null>;
/**
 * Resolver that converts a raw search string into a LIKE WHERE clause.
 * Returns null when input is empty — callers can skip the clause entirely.
 */
type LikeSearchResolver = (q: string) => SQL | null;
/**
 * The text a `_search` column stores for `parts`, and what likeSearch folds each term to: the
 * present parts joined by a space, compatibility forms folded (ＦＩＬＥ, ﬁ), lowercased, Greek final
 * sigma ς as σ (so a letter folds the same wherever it stands), combining marks removed but those
 * that make a Cyrillic letter (й, ї, ё), the Latin letters with no decomposition spelled in ASCII
 * (ß ss, æ ae, œ oe, ø o, ł l, đ and ð d, þ th, ı i, ĸ k, ŋ n, ŧ t, ħ h), then composed again (NFC),
 * so й and Hangul syllables stay whole. Write the column with it, so the reader and the writer
 * are one function.
 */
declare function searchText(...parts: Array<string | null | undefined>): string;
/**
 * Factory for LIKE-based search on a `_search` column that searchText wrote, or any SQL expression
 * that holds such text. Each whitespace-separated term is folded by searchText and trimmed, and
 * kept only when it holds a letter or digit, as FTS keeps its terms: a mark alone folds to nothing
 * or a space, and would match every row. Its LIKE specials (`%`, `_`, `\`) are escaped, and every
 * term is required. A query whose terms all fold away matches no row, as FTS answers it.
 */
declare function likeSearch(column: SQLiteColumn | SQL): LikeSearchResolver;
export { buildFtsHighlight, buildFtsMatch, buildFtsWhere, buildFtsWhereWithSpellfix, likeSearch, type LikeSearchResolver, sanitizeFtsTerm, searchText, type SpellfixDb, type SpellfixSearch, };

/**
 * A many-to-many relation a filter reaches through a join table: `through.throughSourceKey`

/**
 * Query helper types — DB-agnostic.
 * Shared by parsers, validators, and DB-specific SQL generators.
 */
import type { SQL } from "drizzle-orm";
type FieldSelection = {
    relations: Record<string, FieldSelection | null>;
    scalars: string[];
};
type ParsedFields = {
    root: FieldSelection;
};
type FilterOperator = "eq" | "ne" | "neq" | "gt" | "gte" | "lt" | "lte" | "in" | "nin" | "like" | "ilike" | "is"
/** Parsed for callers that validate and run it themselves (JSON-array membership); comb's own validator and SQL take none */
 | "contains";
type FieldType = "string" | "number" | "boolean" | "date" | "enum";
type FilterCondition = {
    field: string;
    operator: FilterOperator;
    value: unknown;
};
type FilterGroup = {
    conditions: FilterCondition[];
    logic: "and" | "or";
    subgroups: FilterGroup[];
};
type FilterAST = {
    root: FilterGroup;
};
type SortDirection = "asc" | "desc";
type SortNulls = "first" | "last";
type SortField<T extends string = string> = {
    direction: SortDirection;
    field: T;
    nulls?: SortNulls | undefined;
};
type ComputedFilterResolver = (operator: FilterOperator, value: unknown) => SQL | null;
type ComputedSortResolver = (direction: SortDirection) => SQL;
type ListQueryCapabilities = {
    computedFilterFields: Record<string, FieldType>;
    computedSortFields: Set<string>;
    filterFields: Record<string, FieldType>;
    pagination: {
        defaultLimit: number;
        maxLimit: number;
    };
    relationFilterFields: Record<string, FieldType>;
    searchFields: Set<string>;
    sortFields: Set<string>;
};
export type { ComputedFilterResolver, ComputedSortResolver, FieldSelection, FieldType, FilterAST, FilterCondition, FilterGroup, FilterOperator, ListQueryCapabilities, ParsedFields, SortDirection, SortField, SortNulls, };
