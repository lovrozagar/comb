// sqlite/columns.d.ts
/**
 * Unified column factories for SQLite
 *
 * Provides dialect-agnostic column definitions that can be swapped with PostgreSQL
 */
import { type $Type, type HasDefault, type IsPrimaryKey, type NotNull } from "drizzle-orm";
import { type SQLiteBooleanBuilder, type SQLiteIntegerBuilder, type SQLiteRealBuilder, type SQLiteTextBuilder, type SQLiteTextJsonBuilder } from "drizzle-orm/sqlite-core";
import type * as z from "zod";
import type { EnumStates, IntegerConstraints, JsonConstraints, TextConstraints, TimestampConstraints } from "./constraints.ts";
type Text = SQLiteTextBuilder<[string, ...string[]]>;
type CreatedAtName = "createdAt" | "created_at";
type DeletedAtName = "deletedAt" | "deleted_at";
type UpdatedAtName = "updatedAt" | "updated_at";
/** What each `c` factory returns: drizzle's own builder types, spelled out for isolated declarations */
export type SqliteColumns = {
    boolean: (name: string) => SQLiteBooleanBuilder;
    createdAt: (name: CreatedAtName, _constraints?: TimestampConstraints) => HasDefault<NotNull<SQLiteIntegerBuilder>>;
    deletedAt: (name: DeletedAtName, _constraints?: TimestampConstraints) => SQLiteIntegerBuilder;
    enum: <T extends readonly string[]>(name: string, _values: T, _states?: EnumStates<T[number]>) => $Type<Text, T[number]>;
    id: (_prefix: string) => IsPrimaryKey<Text>;
    integer: (name: string, _constraints?: IntegerConstraints) => SQLiteIntegerBuilder;
    json: <TSchema extends z.ZodTypeAny = z.ZodTypeAny>(name: string, _schema: TSchema, _constraints?: JsonConstraints) => $Type<SQLiteTextJsonBuilder, z.infer<TSchema>>;
    real: (name: string, _constraints?: IntegerConstraints) => SQLiteRealBuilder;
    ref: (name: string, _constraints?: TextConstraints) => Text;
    serialId: () => IsPrimaryKey<HasDefault<NotNull<SQLiteIntegerBuilder>>>;
    text: (name: string, _constraints?: TextConstraints) => Text;
    timestamp: (name: string, _constraints?: TimestampConstraints) => SQLiteIntegerBuilder;
    updatedAt: (name: UpdatedAtName, _constraints?: TimestampConstraints) => HasDefault<HasDefault<NotNull<SQLiteIntegerBuilder>>>;
};
export declare const c: SqliteColumns;
export {};

// sqlite/constraints.d.ts
/**
 * Type-safe column constraint definitions
 * These exist only in the type system and are completely erased at runtime
 */
type BaseConstraints = {
    /** Field is auto-generated, optional in create, excluded from update */
    autogenerate?: boolean;
    /** Field cannot be mutated after creation */
    nomutate?: boolean;
    /**
     * This column scopes rows to a tenant.
     *
     * Declared, never inferred: comb can see that a column is a foreign key but
     * not that the table it points at is the tenant, and a wrong tenant boundary
     * is worse than a missing one. Saying so here makes it a hand-reviewed fact.
     * At most one column per table may carry it.
     */
    tenant?: boolean;
};
type StringTransformConstraints = {
    /** Apply lowercase transformation */
    lowercase?: boolean;
    /** Trim whitespace */
    trim?: boolean;
    /** Apply uppercase transformation */
    uppercase?: boolean;
};
type NumericRangeConstraints = {
    /** Maximum value */
    max?: number;
    /** Minimum value */
    min?: number;
};
type StringValidationConstraints = {
    /** Email validation */
    email?: boolean;
    /** Image URI validation (accepts https URLs and data:image/ URIs) */
    imageUri?: boolean;
    /** Human/entity name validation (letters, marks, numbers, spaces, punctuation) */
    name?: boolean;
    /** Password validation (min 8, max 72, printable ASCII) */
    password?: boolean;
    /** Regex pattern validation */
    pattern?: string;
    /** URL validation */
    url?: boolean;
};
/**
 * Text column constraints
 * Closed record - only these properties allowed
 */
export type TextConstraints = BaseConstraints & StringTransformConstraints & NumericRangeConstraints & StringValidationConstraints;
/**
 * Integer/numeric column constraints
 * Closed record - only these properties allowed
 */
export type IntegerConstraints = BaseConstraints & NumericRangeConstraints;
/**
 * JSON column constraints
 * Closed record - only these properties allowed
 */
export type JsonConstraints = BaseConstraints & {
    /** Maximum JSON stringified byte length */
    maxBytes?: number;
};
/**
 * Timestamp column constraints
 * Closed record - only these properties allowed
 */
export type TimestampConstraints = BaseConstraints;
/**
 * State machine declared alongside an enum column's values.
 *
 * Type-level only, like every other constraint here — the analyzer reads it
 * out of the source. Hold the same object as a const and pass it to both
 * `c.enum` and `assertTransition`. See docs/state-machines.md.
 */
export type EnumStates<TState extends string = string> = {
    /** The value a row starts at */
    initial?: TState | undefined;
    /** Values from which no transition is legal */
    terminal?: readonly TState[] | undefined;
    /** Legal moves, `from` → `to[]`. Omitted states are unconstrained. */
    transitions?: Readonly<Partial<Record<TState, readonly TState[]>>> | undefined;
};
export {};
